238inline uint16_t crc16_ccitt(const uint8_t* data, size_t len)
239{
240 uint16_t crc = 0xFFFF;
241 for (size_t i = 0; i < len; ++i) {
242 crc ^= (uint16_t)data[i] << 8;
243 for (int j = 0; j < 8; ++j) {
244 if (crc & 0x8000)
245 crc = (crc << 1) ^ 0x1021;
246 else
247 crc <<= 1;
248 }
249 }
250 return crc;
251}
