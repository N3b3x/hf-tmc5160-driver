176// ------------- CRC16-CCITT FUNCTION -------------
177
178inline uint16_t crc16_ccitt(const uint8_t* data, size_t len)
179{
180 uint16_t crc = 0xFFFF;
181 for (size_t i = 0; i < len; ++i) {
182 crc ^= (uint16_t)data[i] << 8;
183 for (int j = 0; j < 8; ++j) {
184 if (crc & 0x8000)
185 crc = (crc << 1) ^ 0x1021;
186 else
187 crc <<= 1;
188 }
189 }
190 return crc;
191}
