165template <typename CommType>
166int32_t TMC5160<CommType>::speedToInternal(float speed_hz) const noexcept {
167 // v[Hz] = v[5160] * (f_CLK[Hz]/2 / 2^23)
168 // v[5160] = v[Hz] / (f_CLK[Hz]/2 / 2^23)
169 // v[5160] = v[Hz] * 2^24 / f_CLK
170 // Then multiply by microstep count (256)
171 if (speed_hz == 0.0f) {
172 return 0;
173 }
174 float internal =
175 (speed_hz * static_cast<float>(1UL << 24)) / static_cast<float>(f_clk_);
176 internal *= static_cast<float>(Microsteps::USTEP_COUNT);
177 return static_cast<int32_t>(internal);
178}

180template <typename CommType>
181float TMC5160<CommType>::speedFromInternal(
182 int32_t speed_internal) const noexcept {
183 // v[Hz] = v[5160] * (f_CLK[Hz]/2 / 2^23) / microstep_count
184 if (speed_internal == 0) {
185 return 0.0f;
186 }
187 float speed_hz = static_cast<float>(speed_internal) *
188 static_cast<float>(f_clk_) / static_cast<float>(1UL << 24);
189 speed_hz /= static_cast<float>(Microsteps::USTEP_COUNT);
190 return speed_hz;
191}

193template <typename CommType>
194int32_t TMC5160<CommType>::accelToInternal(float accel_hz) const noexcept {
195 // a[Hz/s] = a[5160] * f_CLK[Hz]^2 / (512*256) / 2^24
196 // a[5160] = a[Hz/s] * (512*256) * 2^24 / f_CLK^2
197 // Then multiply by microstep count (256)
198 if (accel_hz == 0.0f) {
199 return 0;
200 }
201 float internal = accel_hz * 512.0f * 256.0f * static_cast<float>(1UL << 24) /
202 (static_cast<float>(f_clk_) * static_cast<float>(f_clk_));
203 internal *= static_cast<float>(Microsteps::USTEP_COUNT);
204 return static_cast<int32_t>(internal);
205}

207template <typename CommType>
208int32_t
209TMC5160<CommType>::thresholdSpeedToTstep(float speed_hz) const noexcept {
210 // TSTEP = f_CLK / (speed * 256)
211 if (speed_hz == 0.0f) {
212 return 0;
213 }
214 float tstep = static_cast<float>(f_clk_) / (speed_hz * 256.0f);
215 tstep = std::max(0.0f, std::min(1048575.0f, tstep));
216 return static_cast<int32_t>(tstep);
217}

263template <typename CommType>
264bool TMC5160<CommType>::RampControl::SetMaxSpeed(float speed) noexcept {
265 int32_t internal = driver_->speedToInternal(std::abs(speed));
266 internal = std::min(
267 internal, static_cast<decltype(internal)>(0x7FFFFF)); // VMAX is 23 bits
268 if (!driver_->comm_.WriteRegister(Registers::VMAX,
269 static_cast<uint32_t>(internal))) {
270 return false;
271 }
272 // If in velocity mode, update direction
273 uint32_t rampmode = 0;
274 if (driver_->comm_.ReadRegister(Registers::RAMPMODE, rampmode)) {
275 if (rampmode == static_cast<uint8_t>(RampMode::VELOCITY_POS) ||
276 rampmode == static_cast<uint8_t>(RampMode::VELOCITY_NEG)) {
277 uint8_t new_mode = (speed < 0.0f)
278 ? static_cast<uint8_t>(RampMode::VELOCITY_NEG)
279 : static_cast<uint8_t>(RampMode::VELOCITY_POS);
280 driver_->comm_.WriteRegister(Registers::RAMPMODE, new_mode);
281 }
282 }
283 return true;
284}

313 int32_t vstart = driver_->speedToInternal(std::abs(start_speed));
314 int32_t vstop = driver_->speedToInternal(std::abs(stop_speed));

321 bool success = true;
322 success &= driver_->comm_.WriteRegister(Registers::VSTART,
323 static_cast<uint32_t>(vstart));
324 success &= driver_->comm_.WriteRegister(Registers::VSTOP,
325 static_cast<uint32_t>(vstop));
326 success &=
327 driver_->comm_.WriteRegister(Registers::V_1, static_cast<uint32_t>(v1));

606template <typename CommType>
607bool TMC5160<CommType>::MotorControl::SetupMotorFromSpec(
608 const MotorSpec &motor_spec,
609 const MechanicalSystem *mechanical_system) noexcept {
610 // Calculate global scaler based on rated current
611 // Typical calculation: global_scaler = (rated_current_ma * 32) / (irun * sense_resistor_current)
612 // For simplicity, we'll use a basic calculation
613 uint16_t global_scaler = 32;
614 if (motor_spec.rated_current_ma > 0) {
615 // Basic calculation: assume 1.5A max current, scale accordingly
616 global_scaler = static_cast<uint16_t>(
617 std::min(256U, std::max(32U, (motor_spec.rated_current_ma * 32U) / 1500U)));
618 }
619
620 // Set global scaler
621 if (!SetGlobalScaler(global_scaler)) {
622 return false;
623 }
624
625 // Calculate irun and ihold from rated current
626 // irun should be between 16-31 for best performance
627 // We'll use 80% of rated current for irun, 30% for ihold
628 uint8_t irun = 16;
629 uint8_t ihold = 0;
630 if (motor_spec.rated_current_ma > 0 && global_scaler > 0) {
631 // Calculate irun: target current = (irun/32) * (global_scaler/32) * sense_resistor_current
632 // Simplified: irun = (target_current * 32) / (global_scaler * sense_resistor_current / 32)
633 // Assuming sense resistor gives ~1.5A at irun=31, global_scaler=32
634 float target_run_current = motor_spec.rated_current_ma * 0.8f;
635 uint32_t irun_calc = static_cast<uint32_t>(
636 (target_run_current * 32.0f) /
637 (static_cast<float>(global_scaler) * 0.046875f));
638 irun = static_cast<uint8_t>(
639 std::min(static_cast<uint32_t>(31U),
640 std::max(static_cast<uint32_t>(16U), irun_calc)));
641 float target_hold_current = motor_spec.rated_current_ma * 0.3f;
642 uint32_t ihold_calc = static_cast<uint32_t>(
643 (target_hold_current * 32.0f) /
644 (static_cast<float>(global_scaler) * 0.046875f));
645 ihold = static_cast<uint8_t>(
646 std::min(static_cast<uint32_t>(31U), ihold_calc));
647 }
648
649 // Set motor current
650 if (!SetCurrent(irun, ihold)) {
651 return false;
652 }
653
654 // Configure chopper based on inductance if available
655 if (motor_spec.winding_inductance_uh > 0) {
656 ChopperConfig chop_config{};
657 // Higher inductance may need different settings
658 // This is a simplified heuristic
659 if (motor_spec.winding_inductance_uh > 3000) {
660 chop_config.tbl = 3; // Longer blank time for high inductance
661 }
662 ConfigureChopper(chop_config);
663 }
664
665 // Note: mechanical_system is stored for unit conversions but not used here
666 // as it's used by the unit conversion functions, not motor setup
667
668 return true;
669}

699template <typename CommType>
700bool TMC5160<CommType>::Encoder::SetResolution(int32_t motor_steps,
701 int32_t enc_resolution,
702 bool inverted) noexcept {
703 // Calculate factor: (motor_steps * microsteps) / enc_resolution
704 float factor = static_cast<float>(motor_steps * Microsteps::USTEP_COUNT) /
705 static_cast<float>(enc_resolution);
706
707 // Check if binary prescaler gives exact match
708 int32_t enc_const_binary = static_cast<int32_t>(factor * 65536.0f);
709 if (enc_const_binary * enc_resolution ==
710 motor_steps * Microsteps::USTEP_COUNT * 65536) {
711 // Use binary mode
712 uint32_t encmode_value = 0;
713 if (!driver_->comm_.ReadRegister(Registers::ENCMODE, encmode_value)) {
714 return false;
715 }
716 ENCMODE_Register encmode{};
717 encmode.value = encmode_value;
718 encmode.bits.enc_sel_decimal = false;
719 if (!driver_->comm_.WriteRegister(Registers::ENCMODE, encmode.value)) {
720 return false;
721 }
722 if (inverted) {
723 enc_const_binary = -enc_const_binary;
724 }
725 return driver_->comm_.WriteRegister(
726 Registers::ENC_CONST, static_cast<uint32_t>(enc_const_binary));
727 } else {
728 // Use decimal mode
729 uint32_t encmode_value = 0;
730 if (!driver_->comm_.ReadRegister(Registers::ENCMODE, encmode_value)) {
731 return false;
732 }
733 ENCMODE_Register encmode{};
734 encmode.value = encmode_value;
735 encmode.bits.enc_sel_decimal = true;
736 if (!driver_->comm_.WriteRegister(Registers::ENCMODE, encmode.value)) {
737 return false;
738 }
739 int integer_part = static_cast<int>(std::floor(factor));
740 int decimal_part = static_cast<int>(
741 (factor - static_cast<float>(integer_part)) * 10000.0f);
742 if (inverted) {
743 integer_part = 65535 - integer_part;
744 decimal_part = 10000 - decimal_part;
745 }
746 int32_t enc_const_decimal = integer_part * 65536 + decimal_part;
747 bool exact_match =
748 (static_cast<int32_t>(factor * 10000.0f) * enc_resolution ==
749 motor_steps * Microsteps::USTEP_COUNT * 10000);
750 driver_->comm_.WriteRegister(Registers::ENC_CONST,
751 static_cast<uint32_t>(enc_const_decimal));
752 return exact_match;
753 }
