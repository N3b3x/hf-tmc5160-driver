199template <typename CommType>
200int32_t TMC5160<CommType>::speedToInternal(float speed_hz) const noexcept {
201 // Datasheet formula: v[Hz] = v[5160] * (f_CLK[Hz]/2 / 2^23)
202 // Rearranged: v[5160] = v[Hz] * 2^24 / f_CLK
203 // Where v[Hz] is in μsteps/s (microsteps per second)
204 // Input speed_hz is in steps/s, so we multiply by microstep count (256) to convert to μsteps/s
205 // Final: v[5160] = (speed_hz * 256) * 2^24 / f_CLK
206 if (speed_hz == 0.0f) {
207 return 0;
208 }
209 float internal =
210 (speed_hz * static_cast<float>(1UL << 24)) / static_cast<float>(f_clk_);
211 internal *= static_cast<float>(Microsteps::USTEP_COUNT);
212 return static_cast<int32_t>(internal);
213}

215template <typename CommType>
216float TMC5160<CommType>::speedFromInternal(
217 int32_t speed_internal) const noexcept {
218 // Datasheet formula: v[Hz] = v[5160] * (f_CLK[Hz]/2 / 2^23)
219 // Where v[Hz] is in μsteps/s (microsteps per second)
220 // Output is in steps/s, so we divide by microstep count (256) to convert from μsteps/s
221 // Final: v[steps/s] = (v[5160] * f_CLK / 2^24) / 256
222 if (speed_internal == 0) {
223 return 0.0f;
224 }
225 float speed_hz = static_cast<float>(speed_internal) *
226 static_cast<float>(f_clk_) / static_cast<float>(1UL << 24);
227 speed_hz /= static_cast<float>(Microsteps::USTEP_COUNT);
228 return speed_hz;
229}

231template <typename CommType>
232int32_t TMC5160<CommType>::accelToInternal(float accel_hz) const noexcept {
233 // Datasheet formula: a[Hz/s] = a[5160] * f_CLK[Hz]^2 / (512*256) / 2^24
234 // Rearranged: a[5160] = a[Hz/s] * (512*256) * 2^24 / f_CLK^2
235 // Where a[Hz/s] is in μsteps/s² (microsteps per second squared)
236 // Input accel_hz is in steps/s², so we multiply by microstep count (256) to convert to μsteps/s²
237 // Final: a[5160] = (accel_hz * 256) * (512*256) * 2^24 / f_CLK^2
238 if (accel_hz == 0.0f) {
239 return 0;
240 }
241 float internal = accel_hz * 512.0f * 256.0f * static_cast<float>(1UL << 24) /
242 (static_cast<float>(f_clk_) * static_cast<float>(f_clk_));
243 internal *= static_cast<float>(Microsteps::USTEP_COUNT);
244 return static_cast<int32_t>(internal);
245}

247template <typename CommType>
248int32_t
249TMC5160<CommType>::thresholdSpeedToTstep(float speed_hz) const noexcept {
250 // Datasheet formula: TSTEP = f_CLK / f256STEP = f_CLK / (fSTEP*256/USC)
251 // Where fSTEP is in μsteps/s, USC is microstep count (normally 256)
252 // For USC=256: TSTEP = f_CLK / fSTEP
253 // Input speed_hz is in steps/s, so fSTEP = speed_hz * 256 (convert to μsteps/s)
254 // Final: TSTEP = f_CLK / (speed_hz * 256)
255 // TSTEP is 20-bit unsigned (max value 0xFFFFF = 1048575)
256 if (speed_hz == 0.0f) {
257 return 0;
258 }
259 float tstep = static_cast<float>(f_clk_) / (speed_hz * 256.0f);
260 tstep = std::max(0.0f, std::min(1048575.0f, tstep));
261 return static_cast<int32_t>(tstep);
262}

308template <typename CommType>
309bool TMC5160<CommType>::RampControl::SetMaxSpeed(float speed) noexcept {
310 int32_t internal = driver_.speedToInternal(std::abs(speed));
311 internal = std::min(
312 internal, static_cast<decltype(internal)>(0x7FFFFF)); // VMAX is 23 bits
313 if (!driver_.comm_.WriteRegister(Registers::VMAX,
314 static_cast<uint32_t>(internal))) {
315 return false;
316 }
317 // If in velocity mode, update direction
318 uint32_t rampmode = 0;
319 if (driver_.comm_.ReadRegister(Registers::RAMPMODE, rampmode)) {
320 if (rampmode == static_cast<uint8_t>(RampMode::VELOCITY_POS) ||
321 rampmode == static_cast<uint8_t>(RampMode::VELOCITY_NEG)) {
322 uint8_t new_mode = (speed < 0.0f)
323 ? static_cast<uint8_t>(RampMode::VELOCITY_NEG)
324 : static_cast<uint8_t>(RampMode::VELOCITY_POS);
325 driver_.comm_.WriteRegister(Registers::RAMPMODE, new_mode, driver_.GetCommAddress());
326 }
327 }
328 return true;
329}

370 static_cast<uint32_t>(vstop));

372 driver_.comm_.WriteRegister(Registers::V_1, static_cast<uint32_t>(v1));

725template <typename CommType>
726bool TMC5160<CommType>::MotorControl::SetupMotorFromSpec(
727 const MotorSpec &motor_spec,
728 const MechanicalSystem *mechanical_system) noexcept {
729 // Calculate global scaler based on rated current
730 // Typical calculation: global_scaler = (rated_current_ma * 32) / (irun * sense_resistor_current)
731 // For simplicity, we'll use a basic calculation
732 uint16_t global_scaler = 32;
733 if (motor_spec.rated_current_ma > 0) {
734 // Basic calculation: assume 1.5A max current, scale accordingly
735 global_scaler = static_cast<uint16_t>(
736 std::min(256U, std::max(32U, (motor_spec.rated_current_ma * 32U) / 1500U)));
737 }
738
739 // Set global scaler
740 if (!SetGlobalScaler(global_scaler)) {
741 return false;
742 }
743
744 // Calculate irun and ihold from rated current
745 // irun should be between 16-31 for best performance
746 // We'll use 80% of rated current for irun, 30% for ihold
747 uint8_t irun = 16;
748 uint8_t ihold = 0;
749 if (motor_spec.rated_current_ma > 0 && global_scaler > 0) {
750 // Calculate irun: target current = (irun/32) * (global_scaler/32) * sense_resistor_current
751 // Simplified: irun = (target_current * 32) / (global_scaler * sense_resistor_current / 32)
752 // Assuming sense resistor gives ~1.5A at irun=31, global_scaler=32
753 float target_run_current = motor_spec.rated_current_ma * 0.8f;
754 uint32_t irun_calc = static_cast<uint32_t>(
755 (target_run_current * 32.0f) /
756 (static_cast<float>(global_scaler) * 0.046875f));
757 irun = static_cast<uint8_t>(
758 std::min(static_cast<uint32_t>(31U),
759 std::max(static_cast<uint32_t>(16U), irun_calc)));
760 float target_hold_current = motor_spec.rated_current_ma * 0.3f;
761 uint32_t ihold_calc = static_cast<uint32_t>(
762 (target_hold_current * 32.0f) /
763 (static_cast<float>(global_scaler) * 0.046875f));
764 ihold = static_cast<uint8_t>(
765 std::min(static_cast<uint32_t>(31U), ihold_calc));
766 }
767
768 // Set motor current
769 if (!SetCurrent(irun, ihold)) {
770 return false;
771 }
772
773 // Configure chopper based on inductance if available
774 if (motor_spec.winding_inductance_uh > 0) {
775 ChopperConfig chop_config{};
776 // Higher inductance may need different settings
777 // This is a simplified heuristic
778 if (motor_spec.winding_inductance_uh > 3000) {
779 chop_config.tbl = 3; // Longer blank time for high inductance
780 }
781 ConfigureChopper(chop_config);
782 }
783
784 // Note: mechanical_system is stored for unit conversions but not used here
785 // as it's used by the unit conversion functions, not motor setup
786
787 return true;
788}

818template <typename CommType>
819bool TMC5160<CommType>::Encoder::SetResolution(int32_t motor_steps,
820 int32_t enc_resolution,
821 bool inverted) noexcept {
822 // Calculate factor: (motor_steps * microsteps) / enc_resolution
823 float factor = static_cast<float>(motor_steps * Microsteps::USTEP_COUNT) /
824 static_cast<float>(enc_resolution);
825
826 // Check if binary prescaler gives exact match
827 int32_t enc_const_binary = static_cast<int32_t>(factor * 65536.0f);
828 if (enc_const_binary * enc_resolution ==
829 motor_steps * Microsteps::USTEP_COUNT * 65536) {
830 // Use binary mode
831 uint32_t encmode_value = 0;
832 if (!driver_.comm_.ReadRegister(Registers::ENCMODE, encmode_value)) {
833 return false;
834 }
835 ENCMODE_Register encmode{};
836 encmode.value = encmode_value;
837 encmode.bits.enc_sel_decimal = false;
838 if (!driver_.comm_.WriteRegister(Registers::ENCMODE, encmode.value)) {
839 return false;
840 }
841 if (inverted) {
842 enc_const_binary = -enc_const_binary;
843 }
844 return driver_.comm_.WriteRegister(
845 Registers::ENC_CONST, static_cast<uint32_t>(enc_const_binary));
846 } else {
847 // Use decimal mode
848 uint32_t encmode_value = 0;
849 if (!driver_.comm_.ReadRegister(Registers::ENCMODE, encmode_value)) {
850 return false;
851 }
852 ENCMODE_Register encmode{};
853 encmode.value = encmode_value;
854 encmode.bits.enc_sel_decimal = true;
855 if (!driver_.comm_.WriteRegister(Registers::ENCMODE, encmode.value)) {
856 return false;
857 }
858 int integer_part = static_cast<int>(std::floor(factor));
859 int decimal_part = static_cast<int>(
860 (factor - static_cast<float>(integer_part)) * 10000.0f);
861 if (inverted) {
862 integer_part = 65535 - integer_part;
863 decimal_part = 10000 - decimal_part;
864 }
865 int32_t enc_const_decimal = integer_part * 65536 + decimal_part;
866 bool exact_match =
867 (static_cast<int32_t>(factor * 10000.0f) * enc_resolution ==
868 motor_steps * Microsteps::USTEP_COUNT * 10000);
869 driver_.comm_.WriteRegister(Registers::ENC_CONST,
870 static_cast<uint32_t>(enc_const_decimal));
871 return exact_match;
872 }
