30// Implementation of chip communication mode control methods
31template <typename CommType>
32bool TMC5160<CommType>::SetChipCommMode(ChipCommMode mode) noexcept {
33 const char* mode_name = (mode == ChipCommMode::SPI_INTERNAL_RAMP) ? "SPI_INTERNAL_RAMP" :
34 (mode == ChipCommMode::SPI_EXTERNAL_STEPDIR) ? "SPI_EXTERNAL_STEPDIR" :
35 (mode == ChipCommMode::UART_INTERNAL_RAMP) ? "UART_INTERNAL_RAMP" : "UNKNOWN";
36 TMC5160_LOG_DEBUG(comm_, 2, "TMC5160", "SetChipCommMode(%s)", mode_name);
37
38 // Map mode to SPI_MODE and SD_MODE pin states
39 GpioSignal spi_mode_signal, sd_mode_signal;
40
41 switch (mode) {
42 case ChipCommMode::SPI_INTERNAL_RAMP:
43 spi_mode_signal = GpioSignal::ACTIVE; // HIGH
44 sd_mode_signal = GpioSignal::INACTIVE; // LOW
45 break;
46 case ChipCommMode::SPI_EXTERNAL_STEPDIR:
47 spi_mode_signal = GpioSignal::ACTIVE; // HIGH
48 sd_mode_signal = GpioSignal::ACTIVE; // HIGH
49 break;
50 case ChipCommMode::UART_INTERNAL_RAMP:
51 spi_mode_signal = GpioSignal::INACTIVE; // LOW
52 sd_mode_signal = GpioSignal::INACTIVE; // LOW
53 break;
54 default:
55 return false;
56 }
57
58 // Set SPI_MODE pin
59 if (!this->comm_.GpioSet(TMC5160CtrlPin::SPI_MODE, spi_mode_signal)) {
60 return false;
61 }
62
63 // Set SD_MODE pin
64 if (!this->comm_.GpioSet(TMC5160CtrlPin::SD_MODE, sd_mode_signal)) {
65 return false;
66 }
67
68 return true;
69}

73 GpioSignal spi_mode_signal, sd_mode_signal;
74
75 // Read SPI_MODE pin
76 if (!this->comm_.GpioRead(TMC5160CtrlPin::SPI_MODE, spi_mode_signal)) {
77 return false;
78 }
79
80 // Read SD_MODE pin
81 if (!this->comm_.GpioRead(TMC5160CtrlPin::SD_MODE, sd_mode_signal)) {
82 return false;
83 }
84
85 // Determine mode from pin states
86 if (spi_mode_signal == GpioSignal::ACTIVE) {
87 // SPI_MODE = HIGH
88 if (sd_mode_signal == GpioSignal::ACTIVE) {
89 mode = ChipCommMode::SPI_EXTERNAL_STEPDIR;
90 } else {
91 mode = ChipCommMode::SPI_INTERNAL_RAMP;
92 }
93 } else {
94 // SPI_MODE = LOW (UART mode)
95 mode = ChipCommMode::UART_INTERNAL_RAMP;
96 }
97
98 return true;
99}

268template <typename CommType>
269int32_t TMC5160<CommType>::speedToInternal(float speed_hz) const noexcept {
270 // Datasheet formula: v[Hz] = v[5160] * (f_CLK[Hz]/2 / 2^23)
271 // Rearranged: v[5160] = v[Hz] * 2^24 / f_CLK
272 // Where v[Hz] is in μsteps/s (microsteps per second)
273 // Input speed_hz is in steps/s, so we multiply by microstep count (256) to convert to μsteps/s
274 // Final: v[5160] = (speed_hz * 256) * 2^24 / f_CLK
275 if (speed_hz == 0.0F) {
276 return 0;
277 }
278 float internal = (speed_hz * static_cast<float>(1UL << 24)) / static_cast<float>(f_clk_);
279 internal *= static_cast<float>(Microsteps::USTEP_COUNT);
280 return static_cast<int32_t>(internal);
281}

283template <typename CommType>
284float TMC5160<CommType>::speedFromInternal(int32_t speed_internal) const noexcept {
285 // Datasheet formula: v[Hz] = v[5160] * (f_CLK[Hz]/2 / 2^23)
286 // Where v[Hz] is in μsteps/s (microsteps per second)
287 // Output is in steps/s, so we divide by microstep count (256) to convert from μsteps/s
288 // Final: v[steps/s] = (v[5160] * f_CLK / 2^24) / 256
289 if (speed_internal == 0) {
290 return 0.0F;
291 }
292 float speed_hz = static_cast<float>(speed_internal) * static_cast<float>(f_clk_) / static_cast<float>(1UL << 24);
293 speed_hz /= static_cast<float>(Microsteps::USTEP_COUNT);
294 return speed_hz;
295}

297template <typename CommType>
298int32_t TMC5160<CommType>::accelToInternal(float accel_hz) const noexcept {
299 // Datasheet formula: a[Hz/s] = a[5160] * f_CLK[Hz]^2 / (512*256) / 2^24
300 // Rearranged: a[5160] = a[Hz/s] * (512*256) * 2^24 / f_CLK^2
301 // Where a[Hz/s] is in μsteps/s² (microsteps per second squared)
302 // Input accel_hz is in steps/s², so we multiply by microstep count (256) to convert to μsteps/s²
303 // Final: a[5160] = (accel_hz * 256) * (512*256) * 2^24 / f_CLK^2
304 if (accel_hz == 0.0F) {
305 return 0;
306 }
307 float internal = accel_hz * 512.0F * 256.0F * static_cast<float>(1UL << 24) /
308 (static_cast<float>(f_clk_) * static_cast<float>(f_clk_));
309 internal *= static_cast<float>(Microsteps::USTEP_COUNT);
310 return static_cast<int32_t>(internal);
311}

313template <typename CommType>
314int32_t TMC5160<CommType>::thresholdSpeedToTstep(float speed_hz) const noexcept {
315 // Datasheet formula: TSTEP = f_CLK / f256STEP = f_CLK / (fSTEP*256/USC)
316 // Where fSTEP is in μsteps/s, USC is microstep count (normally 256)
317 // For USC=256: TSTEP = f_CLK / fSTEP
318 // Input speed_hz is in steps/s, so fSTEP = speed_hz * 256 (convert to μsteps/s)
319 // Final: TSTEP = f_CLK / (speed_hz * 256)
320 // TSTEP is 20-bit unsigned (max value 0xFFFFF = 1048575)
321 if (speed_hz == 0.0F) {
322 return 0;
323 }
324 float tstep = static_cast<float>(f_clk_) / (speed_hz * 256.0F);
325 tstep = std::max(0.0F, std::min(1048575.0F, tstep));
326 return static_cast<int32_t>(tstep);
327}

373template <typename CommType>
374bool TMC5160<CommType>::RampControl::SetMaxSpeed(float speed) noexcept {
375 TMC5160_LOG_DEBUG(driver_.comm_, 2, "TMC5160", "RampControl::SetMaxSpeed(%.2f steps/s)", speed);
376 int32_t internal = driver_.speedToInternal(std::abs(speed));
377 internal = std::min(internal, static_cast<decltype(internal)>(0x7FFFFF)); // VMAX is 23 bits
378 if (!driver_.comm_.WriteRegister(Registers::VMAX, static_cast<uint32_t>(internal))) {
379 return false;
380 }
381 // If in velocity mode, update direction
382 uint32_t rampmode = 0;
383 if (driver_.comm_.ReadRegister(Registers::RAMPMODE, rampmode)) {
384 if (rampmode == static_cast<uint8_t>(RampMode::VELOCITY_POS) ||
385 rampmode == static_cast<uint8_t>(RampMode::VELOCITY_NEG)) {
386 uint8_t new_mode =
387 (speed < 0.0F) ? static_cast<uint8_t>(RampMode::VELOCITY_NEG) : static_cast<uint8_t>(RampMode::VELOCITY_POS);
388 driver_.comm_.WriteRegister(Registers::RAMPMODE, new_mode, driver_.GetCommAddress());
389 }
390 }
391 return true;

691 // VDCMIN is 23-bit register (0...2^22), but only bits 22..8 are used for comparison

706 DCCTRL_Register dcctrl{};
707 dcctrl.bits.dc_time = constrain<decltype(config.dc_time)>(config.dc_time, 0U, 1023U);

771template <typename CommType>
772bool TMC5160<CommType>::MotorControl::SetupMotorFromSpec(const MotorSpec& motor_spec,
773 const MechanicalSystem* mechanical_system) noexcept {
774 // Calculate global scaler based on rated current
775 // Typical calculation: global_scaler = (rated_current_ma * 32) / (irun * sense_resistor_current)
776 // For simplicity, we'll use a basic calculation
777 uint16_t global_scaler = 32;
778 if (motor_spec.rated_current_ma > 0) {
779 // Basic calculation: assume 1.5A max current, scale accordingly
780 global_scaler = static_cast<uint16_t>(std::min(256U, std::max(32U, (motor_spec.rated_current_ma * 32U) / 1500U)));
781 }
782
783 // Set global scaler
784 if (!SetGlobalScaler(global_scaler)) {
785 return false;
786 }
787
788 // Calculate irun and ihold from rated current
789 // irun should be between 16-31 for best performance
790 // We'll use 80% of rated current for irun, 30% for ihold
791 uint8_t irun = 16;
792 uint8_t ihold = 0;
793 if (motor_spec.rated_current_ma > 0 && global_scaler > 0) {
794 // Calculate irun: target current = (irun/32) * (global_scaler/32) * sense_resistor_current
795 // Simplified: irun = (target_current * 32) / (global_scaler * sense_resistor_current / 32)
796 // Assuming sense resistor gives ~1.5A at irun=31, global_scaler=32
797 float target_run_current = static_cast<float>(motor_spec.rated_current_ma) * 0.8F;
798 auto irun_calc =
799 static_cast<uint32_t>((target_run_current * 32.0F) / (static_cast<float>(global_scaler) * 0.046875F));
800 irun = static_cast<uint8_t>(std::min(static_cast<uint32_t>(31U), std::max(static_cast<uint32_t>(16U), irun_calc)));
801 float target_hold_current = static_cast<float>(motor_spec.rated_current_ma) * 0.3F;
802 auto ihold_calc =
803 static_cast<uint32_t>((target_hold_current * 32.0F) / (static_cast<float>(global_scaler) * 0.046875F));
804 ihold = static_cast<uint8_t>(std::min(static_cast<uint32_t>(31U), ihold_calc));
805 }
806
807 // Set motor current
808 if (!SetCurrent(irun, ihold)) {
809 return false;
810 }
811
812 // Configure chopper based on inductance if available
813 if (motor_spec.winding_inductance_uh > 0) {
814 ChopperConfig chop_config{};
815 // Higher inductance may need different settings
816 // This is a simplified heuristic
817 if (motor_spec.winding_inductance_uh > 3000) {
818 chop_config.tbl = 3; // Longer blank time for high inductance
819 }
820 ConfigureChopper(chop_config);
821 }
822
823 // Note: mechanical_system is stored for unit conversions but not used here
824 // as it's used by the unit conversion functions, not motor setup
825
826 return true;
827}

855template <typename CommType>
856bool TMC5160<CommType>::Encoder::SetResolution(int32_t motor_steps, int32_t enc_resolution, bool inverted) noexcept {
857 // Calculate factor: (motor_steps * microsteps) / enc_resolution
858 float factor = static_cast<float>(motor_steps * Microsteps::USTEP_COUNT) / static_cast<float>(enc_resolution);
859
860 // Check if binary prescaler gives exact match
861 auto enc_const_binary = static_cast<int32_t>(factor * 65536.0F);
862 if (enc_const_binary * enc_resolution == motor_steps * Microsteps::USTEP_COUNT * 65536) {
863 // Use binary mode
864 uint32_t encmode_value = 0;
865 if (!driver_.comm_.ReadRegister(Registers::ENCMODE, encmode_value)) {
866 return false;
867 }
868 ENCMODE_Register encmode{};
869 encmode.value = encmode_value;
870 encmode.bits.enc_sel_decimal = false;
871 if (!driver_.comm_.WriteRegister(Registers::ENCMODE, encmode.value)) {
872 return false;
873 }
874 if (inverted) {
875 enc_const_binary = -enc_const_binary;
876 }
877 return driver_.comm_.WriteRegister(Registers::ENC_CONST, static_cast<uint32_t>(enc_const_binary));
878 }
879 // Use decimal mode
880 uint32_t encmode_value = 0;
881 if (!driver_.comm_.ReadRegister(Registers::ENCMODE, encmode_value)) {
882 return false;
883 }
884 ENCMODE_Register encmode{};
885 encmode.value = encmode_value;
886 encmode.bits.enc_sel_decimal = true;
887 if (!driver_.comm_.WriteRegister(Registers::ENCMODE, encmode.value)) {
888 return false;
889 }
890 int integer_part = static_cast<int>(std::floor(factor));
891 int decimal_part = static_cast<int>((factor - static_cast<float>(integer_part)) * 10000.0F);
892 if (inverted) {
893 integer_part = 65535 - integer_part;
894 decimal_part = 10000 - decimal_part;
895 }
896 int32_t enc_const_decimal = (integer_part * 65536) + decimal_part;
897 bool exact_match =
898 ((static_cast<int32_t>(factor * 10000.0F) * enc_resolution) == (motor_steps * Microsteps::USTEP_COUNT * 10000));
899 driver_.comm_.WriteRegister(Registers::ENC_CONST, static_cast<uint32_t>(enc_const_decimal));
900 return exact_match;
901}

1034template <typename CommType>
1035bool TMC5160<CommType>::Diagnostics::PerformSensorlessHoming(bool direction, int8_t stall_threshold, float search_speed,
1036 int32_t& final_position, uint32_t timeout_ms) noexcept {
1037 // Configure StallGuard2 for homing
1038 StallGuardConfig sg_config{};
1039 sg_config.sgt = stall_threshold;
1040 sg_config.sfilt = true; // Enable filter for stable readings
1041 if (!ConfigureStallGuard(sg_config)) {
1042 return false;
1043 }
1044
1045 // Enable StallGuard2 stop in SW_MODE
1046 uint32_t sw_mode_value = 0;
1047 if (!driver_.comm_.ReadRegister(Registers::SW_MODE, sw_mode_value)) {
1048 return false;
1049 }
1050 SW_MODE_Register sw_mode{};
1051 sw_mode.value = sw_mode_value;
1052 sw_mode.bits.sg_stop = true; // Enable stop on stall
1053 sw_mode.bits.en_softstop = false; // Use hard stop for precise homing (per datasheet 12.4)
1054 if (!driver_.comm_.WriteRegister(Registers::SW_MODE, sw_mode.value)) {
1055 return false;
1056 }
1057
1058 // Clear any existing stall flags
1059 uint32_t ramp_stat_dummy = 0;
1060 driver_.comm_.ReadRegister(Registers::RAMP_STAT, ramp_stat_dummy);
1061 // Write back 1 to event_stop_sg to clear it
1062 RAMP_STAT_Register clear_stat{};
1063 clear_stat.bits.event_stop_sg = 1;
1064 driver_.comm_.WriteRegister(Registers::RAMP_STAT, clear_stat.value);
1065
1066 // Set velocity mode and start movement
1067 RampMode mode = direction ? RampMode::VELOCITY_POS : RampMode::VELOCITY_NEG;
1068 if (!driver_.rampControl.SetRampMode(mode)) {
1069 return false;
1070 }
1071 if (!driver_.rampControl.SetMaxSpeed(search_speed)) {
1072 return false;
1073 }
1074
1075 // Wait for stall event
1076 bool stalled = false;
1077 // Simple polling loop with crude timeout using rough cycle estimates
1078 // Assuming ~10us per register read interaction on SPI
1079 uint32_t loops = timeout_ms * 100;
1080
1081 for (uint32_t i = 0; i < loops; i++) {
1082 uint32_t ramp_stat = 0;
1083 if (driver_.comm_.ReadRegister(Registers::RAMP_STAT, ramp_stat)) {
1084 RAMP_STAT_Register status{};
1085 status.value = ramp_stat;
1086
1087 // Check for stall stop event or velocity zero (stop executed)
1088 if (status.bits.event_stop_sg || (status.bits.vzero && status.bits.status_sg)) {
1089 stalled = true;
1090 break;
1091 }
1092 }
1093 // Small delay to relieve bus
1094 // driver_.comm_.DelayUs(100); // Requires DelayUs in Comm interface, assuming it exists
1095 }
1096
1097 // Stop motor (ensure VMAX=0)
1098 driver_.rampControl.Stop();
1099
1100 // Disable StallGuard2 stop to prevent accidental stops later
1101 sw_mode.bits.sg_stop = false;
1102 driver_.comm_.WriteRegister(Registers::SW_MODE, sw_mode.value);
1103
1104 // Read final position
1105 final_position = driver_.rampControl.GetCurrentPosition();
1106
1107 // Clear the stall event flag again to be clean
1108 driver_.comm_.WriteRegister(Registers::RAMP_STAT, clear_stat.value);
1109
1110 return stalled;
1111}

1113template <typename CommType>
1114bool TMC5160<CommType>::Diagnostics::PerformSwitchHoming(bool direction, float search_speed, float switch_speed,
1115 int32_t& final_position, bool use_left_switch, uint32_t timeout_ms) noexcept {
1116 // 1. Activate position latching and motor stop upon switch event
1117 SW_MODE_Register sw_mode{};
1118 // Read current SW_MODE to preserve other settings (like softstop)
1119 uint32_t sw_mode_val = 0;
1120 if (!driver_.comm_.ReadRegister(Registers::SW_MODE, sw_mode_val)) {
1121 return false;
1122 }
1123 sw_mode.value = sw_mode_val;
1124
1125 // Configure latching and stop enable based on switch choice
1126 if (use_left_switch) {
1127 sw_mode.bits.latch_l_active = true;
1128 sw_mode.bits.stop_l_enable = true;
1129 } else {
1130 sw_mode.bits.latch_r_active = true;
1131 sw_mode.bits.stop_r_enable = true;
1132 }
1133 // Use hard stop for precise homing (soft stop can overshoot) - datasheet recommends hard stop for StallGuard,
1134 // but for switches hard stop ensures we don't crash if switch is a hard limit.
1135 // However, datasheet 12.4 says "Or motor can be softly decelerated...".
1136 // Let's assume hard stop for safety during homing search.
1137 sw_mode.bits.en_softstop = false;
1138
1139 if (!driver_.comm_.WriteRegister(Registers::SW_MODE, sw_mode.value)) {
1140 return false;
1141 }
1142
1143 // 2. Start motion ramp into direction of switch
1144 RampMode mode = direction ? RampMode::VELOCITY_POS : RampMode::VELOCITY_NEG;
1145 if (!driver_.rampControl.SetRampMode(mode)) {
1146 return false;
1147 }
1148 if (!driver_.rampControl.SetMaxSpeed(search_speed)) {
1149 return false;
1150 }
1151
1152 // 3. Wait for switch hit (motor stops automatically)
1153 bool switch_hit = false;
1154 uint32_t loops = timeout_ms * 100;
1155 for (uint32_t i = 0; i < loops; i++) {
1156 uint32_t ramp_stat = 0;
1157 if (driver_.comm_.ReadRegister(Registers::RAMP_STAT, ramp_stat)) {
1158 RAMP_STAT_Register status{};
1159 status.value = ramp_stat;
1160
1161 // Check for specific stop event
1162 if ((use_left_switch && status.bits.event_stop_l) ||
1163 (!use_left_switch && status.bits.event_stop_r)) {
1164 switch_hit = true;
1165 break;
1166 }
1167 // Also check vzero if we missed the event flag but stopped
1168 if (status.bits.vzero && ((use_left_switch && status.bits.status_stop_l) || (!use_left_switch && status.bits.status_stop_r))) {
1169 switch_hit = true;
1170 break;
1171 }
1172 }
1173 // driver_.comm_.DelayUs(100);
1174 }
1175
1176 // 4. Stop motor command (VMAX=0) to ensure it stays stopped
1177 driver_.rampControl.Stop();
1178
1179 if (switch_hit) {
1180 // 5. Read latched position
1181 // "Latching of... XACTUAL to XLATCH upon a switch event gives a precise snapshot"
1182 // However, datasheet 12.4 "Implementing a homing procedure" step 5 says:
1183 // "Switch to hold mode... calculate difference between latched and actual... or when using hard stop XACTUAL stops exactly at home position"
1184 // Since we used hard stop, XACTUAL should be valid.
1185 // Let's read XLATCH just in case user wants it, but we return current XACTUAL as "final position".
1186 // Actually, let's just return the current position where it stopped.
1187 final_position = driver_.rampControl.GetCurrentPosition();
1188 }
1189
1190 // Disable stop function to allow moving away
1191 if (use_left_switch) {
1192 sw_mode.bits.stop_l_enable = false;
1193 } else {
1194 sw_mode.bits.stop_r_enable = false;
1195 }
1196 driver_.comm_.WriteRegister(Registers::SW_MODE, sw_mode.value);
1197
1198 return switch_hit;
1199}
