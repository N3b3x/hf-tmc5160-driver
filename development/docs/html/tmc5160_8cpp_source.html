31// Implementation of chip communication mode control methods
32template <typename CommType>
33bool TMC5160<CommType>::SetChipCommMode(ChipCommMode mode) noexcept {
34 const char* mode_name = (mode == ChipCommMode::SPI_INTERNAL_RAMP) ? "SPI_INTERNAL_RAMP" :
35 (mode == ChipCommMode::SPI_EXTERNAL_STEPDIR) ? "SPI_EXTERNAL_STEPDIR" :
36 (mode == ChipCommMode::UART_INTERNAL_RAMP) ? "UART_INTERNAL_RAMP" : "UNKNOWN";
37 TMC5160_LOG_DEBUG(comm_, 2, "TMC5160", "SetChipCommMode(%s)", mode_name);
38
39 // Map mode to SPI_MODE and SD_MODE pin states
40 GpioSignal spi_mode_signal, sd_mode_signal;
41
42 switch (mode) {
43 case ChipCommMode::SPI_INTERNAL_RAMP:
44 spi_mode_signal = GpioSignal::ACTIVE; // HIGH
45 sd_mode_signal = GpioSignal::INACTIVE; // LOW
46 break;
47 case ChipCommMode::SPI_EXTERNAL_STEPDIR:
48 spi_mode_signal = GpioSignal::ACTIVE; // HIGH
49 sd_mode_signal = GpioSignal::ACTIVE; // HIGH
50 break;
51 case ChipCommMode::UART_INTERNAL_RAMP:
52 spi_mode_signal = GpioSignal::INACTIVE; // LOW
53 sd_mode_signal = GpioSignal::INACTIVE; // LOW
54 break;
55 default:
56 return false;
57 }
58
59 // Set SPI_MODE pin
60 if (!this->comm_.GpioSet(TMC5160CtrlPin::SPI_MODE, spi_mode_signal)) {
61 return false;
62 }
63
64 // Set SD_MODE pin
65 if (!this->comm_.GpioSet(TMC5160CtrlPin::SD_MODE, sd_mode_signal)) {
66 return false;
67 }
68
69 return true;
70}

72template <typename CommType>
73bool TMC5160<CommType>::GetChipCommMode(ChipCommMode& mode) const noexcept {
74 GpioSignal spi_mode_signal, sd_mode_signal;
75
76 // Read SPI_MODE pin
77 if (!this->comm_.GpioRead(TMC5160CtrlPin::SPI_MODE, spi_mode_signal)) {
78 return false;
79 }
80
81 // Read SD_MODE pin
82 if (!this->comm_.GpioRead(TMC5160CtrlPin::SD_MODE, sd_mode_signal)) {
83 return false;
84 }
85
86 // Determine mode from pin states
87 if (spi_mode_signal == GpioSignal::ACTIVE) {
88 // SPI_MODE = HIGH
89 if (sd_mode_signal == GpioSignal::ACTIVE) {
90 mode = ChipCommMode::SPI_EXTERNAL_STEPDIR;
91 } else {
92 mode = ChipCommMode::SPI_INTERNAL_RAMP;
93 }
94 } else {
95 // SPI_MODE = LOW (UART mode)
96 mode = ChipCommMode::UART_INTERNAL_RAMP;
97 }
98
99 return true;
100}

114// Implementation of unit conversion helpers
115template <typename CommType>
116float TMC5160<CommType>::convertSpeedToSteps(float value, Unit unit) const noexcept {
117 if (value == 0.0f) return 0.0f;
118
119 // Calculate effective steps per revolution (full steps * microsteps * gear ratio)
120 float effective_steps_per_rev = static_cast<float>(motor_spec_.steps_per_rev) *
121 static_cast<float>(current_microsteps_) *
122 mechanical_system_.gear_ratio;
123
124 switch (unit) {
125 case Unit::Steps:
126 return value;
127 case Unit::RPM:
128 // RPM * steps_per_rev / 60
129 return (value * effective_steps_per_rev) / 60.0f;
130 case Unit::Rad:
131 // rad/s * steps_per_rev / (2*PI)
132 return (value * effective_steps_per_rev) / (2.0f * 3.14159265359f);
133 case Unit::Deg:
134 // deg/s * steps_per_rev / 360
135 return (value * effective_steps_per_rev) / 360.0f;
136 case Unit::Mm:
137 if (mechanical_system_.system_type == MechanicalSystemType::LeadScrew && mechanical_system_.lead_screw_pitch_mm > 0.0f) {
138 // mm/s / pitch * steps_per_rev
139 return (value / mechanical_system_.lead_screw_pitch_mm) * effective_steps_per_rev;
140 } else if (mechanical_system_.system_type == MechanicalSystemType::BeltDrive && mechanical_system_.belt_pitch_mm > 0.0f && mechanical_system_.belt_pulley_teeth > 0) {
141 // mm/s / (pitch * teeth) * steps_per_rev
142 float mm_per_rev = mechanical_system_.belt_pitch_mm * static_cast<float>(mechanical_system_.belt_pulley_teeth);
143 return (value / mm_per_rev) * effective_steps_per_rev;
144 }
145 return 0.0f; // Invalid config for mm
146 default:
147 return value;
148 }

228float TMC5160<CommType>::convertSpeedToUnit(float steps_per_sec, Unit unit) const noexcept {
229 // Logic similar to steps->unit but scaling for time if needed
230 // Steps/s -> RPM: (Steps/s / Steps/rev) * 60
231
232 float effective_steps_per_rev = static_cast<float>(motor_spec_.steps_per_rev) *
233 static_cast<float>(current_microsteps_) *
234 mechanical_system_.gear_ratio;

237
238 switch (unit) {

240 return steps_per_sec;
241 case Unit::RPM:
242 return (steps_per_sec / effective_steps_per_rev) * 60.0f;
243 case Unit::Rad:
244 return (steps_per_sec / effective_steps_per_rev) * (2.0f * 3.14159265359f);
245 case Unit::Deg:
246 return (steps_per_sec / effective_steps_per_rev) * 360.0f;
247 case Unit::Mm:

249 return (steps_per_sec / effective_steps_per_rev) * mechanical_system_.lead_screw_pitch_mm;
250 } else if (mechanical_system_.system_type == MechanicalSystemType::BeltDrive) {
251 float mm_per_rev = mechanical_system_.belt_pitch_mm * static_cast<float>(mechanical_system_.belt_pulley_teeth);
252 return (steps_per_sec / effective_steps_per_rev) * mm_per_rev;
253 }
254 return 0.0f;
255 default:
256 return steps_per_sec;

260template <typename CommType>
261bool TMC5160<CommType>::Initialize(const DriverConfig& config) noexcept {
262 TMC5160_LOG_DEBUG(comm_, 2, "TMC5160", "Initialize(toff=%u, mres=%u)",
263 config.chopper.toff, config.chopper.mres);
264
265 // Store physical configuration
266 motor_spec_ = config.motor_spec;
267 mechanical_system_ = config.mechanical;
268
269 // Calculate initial microsteps from config
270 // mres: 0=256, 1=128, ... 8=fullstep. microsteps = 256 >> mres
271 uint8_t mres = constrain<uint8_t>(config.chopper.mres, 0U, 8U);
272 current_microsteps_ = 256U >> mres;
273
274 // Always calculate motor current settings from physical parameters
275 if (motor_spec_.sense_resistor_mohm > 0 && motor_spec_.supply_voltage_mv > 0) {
276 uint8_t calc_irun = 0;
277 uint8_t calc_ihold = 0;
278 uint16_t calc_scaler = 0;
279
280 uint16_t run_current = motor_spec_.run_current_ma;
281 if (run_current == 0) {
282 run_current = motor_spec_.rated_current_ma;
283 }
284
285 if (CalculateMotorCurrent(motor_spec_, motor_spec_.sense_resistor_mohm,
286 motor_spec_.supply_voltage_mv,
287 run_current, motor_spec_.hold_current_ma,
288 calc_irun, calc_ihold, calc_scaler)) {
289 // Apply percentage adjustments before constraining
290 // GLOBAL_SCALER adjustment
291 if (motor_spec_.scaler_adjustment_percent != 0.0f) {
292 float adjustment_factor = 1.0f + (motor_spec_.scaler_adjustment_percent / 100.0f);
293 calc_scaler = static_cast<uint16_t>(std::round(static_cast<float>(calc_scaler) * adjustment_factor));
294 }
295
296 // IRUN adjustment
297 if (motor_spec_.irun_adjustment_percent != 0.0f) {
298 float adjustment_factor = 1.0f + (motor_spec_.irun_adjustment_percent / 100.0f);
299 calc_irun = static_cast<uint8_t>(std::round(static_cast<float>(calc_irun) * adjustment_factor));
300 }
301
302 // IHOLD adjustment
303 if (motor_spec_.ihold_adjustment_percent != 0.0f) {
304 float adjustment_factor = 1.0f + (motor_spec_.ihold_adjustment_percent / 100.0f);
305 calc_ihold = static_cast<uint8_t>(std::round(static_cast<float>(calc_ihold) * adjustment_factor));
306 }
307
308 // Constrain to valid ranges after adjustments
309 calc_scaler = constrain<uint16_t>(calc_scaler, 32U, 256U);
310 calc_irun = constrain<uint8_t>(calc_irun, 0U, 31U);
311 calc_ihold = constrain<uint8_t>(calc_ihold, 0U, 31U);
312
313 // Ensure IHOLD < IRUN
314 if (calc_ihold >= calc_irun) {
315 calc_ihold = (calc_irun > 0) ? (calc_irun - 1) : 0;
316 }
317
318 // Store calculated values internally (not in motor_spec_)
319 calculated_irun_ = calc_irun;
320 calculated_ihold_ = calc_ihold;
321 calculated_global_scaler_ = calc_scaler;
322
323 TMC5160_LOG_DEBUG(comm_, 1, "TMC5160", "Calculated: IRUN=%u, IHOLD=%u, GLOBAL_SCALER=%u (adjustments: scaler=%.1f%%, irun=%.1f%%, ihold=%.1f%%)",
324 calc_irun, calc_ihold, calc_scaler,
325 motor_spec_.scaler_adjustment_percent,
326 motor_spec_.irun_adjustment_percent,
327 motor_spec_.ihold_adjustment_percent);
328
329 // Validate StealthChop lower limit if resistance is available
330 if (motor_spec_.winding_resistance_mohm > 0) {
331 uint16_t lower_limit = CalculateStealthChopLowerLimit(motor_spec_,
332 motor_spec_.supply_voltage_mv,
333 config.chopper.tbl,
334 config.stealthchop.pwm_freq,
335 f_clk_);
336 if (lower_limit > 0) {
337 float run_current_a = static_cast<float>(run_current) / 1000.0f;
338 float lower_limit_a = static_cast<float>(lower_limit) / 1000.0f;
339 if (run_current_a < lower_limit_a * 1.1f) {
340 TMC5160_LOG_DEBUG(comm_, 0, "TMC5160",
341 "WARNING: Run current (%.1fmA) may be below StealthChop lower limit (%.1fmA)",
342 run_current, lower_limit);
343 }
344 }
345 }
346 } else {
347 TMC5160_LOG_DEBUG(comm_, 0, "TMC5160", "Failed to calculate motor current settings");
348 return false;
349 }
350 } else {
351 TMC5160_LOG_DEBUG(comm_, 0, "TMC5160",
352 "Cannot calculate: sense_resistor_mohm=%u, supply_voltage_mv=%u",
353 motor_spec_.sense_resistor_mohm, motor_spec_.supply_voltage_mv);
354 return false;
355 }
356
357 // Clear reset and error flags
358 GSTAT_Register gstat{};
359 gstat.bits.reset = true;
360 gstat.bits.drv_err = true;
361 gstat.bits.uv_cp = true;
362 if (!this->comm_.WriteRegister(Registers::GSTAT, gstat.value, this->GetCommAddress())) {
363 return false;
364 }
365
366 // Configure power stage (calculate register values from user-friendly parameters)
367 DRV_CONF_Register drv_conf{};
368
369 // Calculate DRVSTRENGTH from MOSFET Miller charge
370 float miller = config.power_stage.mosfet_miller_charge_nc;
371 if (miller < 0.0f) {
372 miller = 0.0f; // Constrain to valid range
373 }
374 if (miller == 0.0f) {
375 // Auto-calculate from default (small MOSFET, lowest driver strength)
376 miller = 10.0f;
377 }
378
379 uint8_t drv_strength = 0;
380 // Auto-calculate from Miller charge (datasheet table 3.3)
381 // Use lowest gate driver strength giving favorable switching slopes
382 if (miller < 10.0f) {
383 drv_strength = 0;
384 } else if (miller < 20.0f) {
385 drv_strength = 0; // Can use 0 or 1, prefer 0 (lowest) for best switching quality
386 } else if (miller < 40.0f) {
387 drv_strength = 1; // Can use 1 or 2, prefer 1 (lower)
388 } else if (miller < 60.0f) {
389 drv_strength = 2; // Can use 2 or 3, prefer 2 (lower)
390 } else {
391 drv_strength = 3; // Maximum for very large MOSFETs
392 }
393 drv_conf.bits.drvstrength = constrain<uint8_t>(drv_strength, 0U, 3U);
394
395 // Calculate BBMTIME and BBMCLKS from nanoseconds
396 uint8_t bbm_time_reg = 0;
397 uint8_t bbm_clks_reg = 0;
398
399 // Auto-calculate from bbm_time_ns
400 uint32_t bbm_ns = config.power_stage.bbm_time_ns;
401
402 if (bbm_ns == 0) {
403 // Auto-calculate from MOSFET: use minimum safe time (100ns for small MOSFETs, lowest setting)
404 // For larger MOSFETs with higher Miller charge, may need more time
405 if (miller > 40.0f) {
406 bbm_ns = 200; // Larger MOSFETs need more BBM time
407 } else {
408 bbm_ns = 100; // Default minimum (lowest, BBMTIME=0)
409 }
410 }
411
412 // Ensure minimum 100ns (lowest possible setting)
413 if (bbm_ns < 100) {
414 bbm_ns = 100;
415 }
416
417 // BBMTIME calculation per datasheet:
418 // Formula: time[ns] ≈ 100ns * 32 / (32 - BBMTIME)
419 // Rearranged: BBMTIME = 32 - (100ns * 32 / time_ns)
420 // BBMTIME range: 0-24 (bits 4..0)
421 // BBMTIME=0 gives 100ns (shortest, reset default)
422 // BBMTIME=16 gives 200ns
423 // BBMTIME=24 gives ~375ns (longest, datasheet says 375ns, formula gives ~400ns)
424 // >24 not recommended, use BBMCLKS instead
425 //
426 // Datasheet hint: Choose the lowest setting safely covering the switching event.
427 // Add roughly 30% of reserve to cover production stray.
428 if (bbm_ns <= 200) {
429 // Use BBMTIME (0-24, corresponds to 100ns to ~375ns)
430 // Add 30% headroom as recommended by datasheet
431 float bbm_ns_with_headroom = static_cast<float>(bbm_ns) * 1.3f;
432 // Calculate BBMTIME: BBMTIME = 32 - (100 * 32 / time_ns)
433 float bbm_time_float = 32.0f - (100.0f * 32.0f / bbm_ns_with_headroom);
434 bbm_time_reg = static_cast<uint8_t>(std::round(bbm_time_float));
435 bbm_time_reg = constrain<uint8_t>(bbm_time_reg, 0U, 24U);
436 bbm_clks_reg = 0; // BBMTIME takes precedence, BBMCLKS=0 (off)
437 } else {
438 // Use BBMCLKS for longer times (>200ns)
439 // BBMCLKS range: 0-15 (bits 11..8), typ. 83ns per clock cycle at 12MHz
440 // The longer setting rules (BBMTIME vs. BBMCLKS)
441 // Add 30% headroom as recommended by datasheet
442 float bbm_ns_with_headroom = static_cast<float>(bbm_ns) * 1.3f;
443 // Calculate clock cycles: cycles = (time_ns * f_clk) / 1e9
444 // At 12MHz: 1 cycle = 83.3ns
445 uint32_t clock_cycles = static_cast<uint32_t>((bbm_ns_with_headroom * static_cast<float>(f_clk_)) / 1000000000.0f);
446 bbm_clks_reg = constrain<uint8_t>(static_cast<uint8_t>(clock_cycles), 0U, 15U);
447 if (bbm_clks_reg == 0 && bbm_ns > 200) {
448 bbm_clks_reg = 1; // Minimum 1 clock cycle if >200ns requested
449 }
450 bbm_time_reg = 0; // BBMCLKS takes precedence when set (longer setting rules)
451 }
452
453 drv_conf.bits.bbmtime = bbm_time_reg;
454 drv_conf.bits.bbmclks = bbm_clks_reg;
455 drv_conf.bits.otselect = static_cast<uint8_t>(config.power_stage.over_temp_protection);
456 drv_conf.bits.filt_isense = static_cast<uint8_t>(config.power_stage.sense_filter);
457
458 TMC5160_LOG_DEBUG(comm_, 2, "TMC5160", "Power stage: DRVSTRENGTH=%u (from %.1fnC), BBMTIME=%u, BBMCLKS=%u, FILT_ISENSE=%u",
459 drv_conf.bits.drvstrength, miller,
460 drv_conf.bits.bbmtime, drv_conf.bits.bbmclks, drv_conf.bits.filt_isense);
461
462 if (!this->comm_.WriteRegister(Registers::DRV_CONF, drv_conf.value, this->GetCommAddress())) {
463 return false;
464 }
465
466 // Configure global scaler (use calculated value)
467 TMC5160_LOG_DEBUG(comm_, 3, "TMC5160", "Initialize: Setting GLOBAL_SCALER=%u", calculated_global_scaler_);
468 if (!this->comm_.WriteRegister(Registers::GLOBAL_SCALER, calculated_global_scaler_, this->GetCommAddress())) {
469 return false;
470 }
471
472 // Configure motor current (use calculated values)
473 IHOLD_IRUN_Register iholdrun{};
474 iholdrun.bits.ihold = calculated_ihold_;
475 iholdrun.bits.irun = calculated_irun_;
476 iholdrun.bits.iholddelay = 7;
477 TMC5160_LOG_DEBUG(comm_, 3, "TMC5160", "Initialize: Setting IHOLD_IRUN(irun=%u, ihold=%u, iholddelay=7)",
478 iholdrun.bits.irun, iholdrun.bits.ihold);
479 if (!this->comm_.WriteRegister(Registers::IHOLD_IRUN, iholdrun.value, this->GetCommAddress())) {
480 return false;
481 }
482
483 // Configure short protection (convert user-friendly values to register values)
484 SHORT_CONF_Register short_conf{};
485
486 // Calculate S2VS_LEVEL from voltage threshold
487 uint16_t s2vs_voltage = config.power_stage.s2vs_voltage_mv;
488 if (s2vs_voltage == 0) {
489 s2vs_voltage = 625; // Default recommended: 625mV (S2VS_LEVEL=6)
490 }
491 uint8_t s2vs_level = CalculateS2VSLevel(s2vs_voltage);
492 if (s2vs_level == 0 || s2vs_level < 4) {
493 s2vs_level = 6; // Fallback to default
494 }
495 short_conf.bits.s2vs_level = constrain<uint8_t>(s2vs_level, 4U, 15U);
496
497 // Calculate S2G_LEVEL from voltage threshold (consider supply voltage for VS>52V check)
498 uint16_t s2g_voltage = config.power_stage.s2g_voltage_mv;
499 if (s2g_voltage == 0) {
500 s2g_voltage = 625; // Default recommended: 625mV (S2G_LEVEL=6)
501 }
502 uint8_t s2g_level = CalculateS2GLevel(s2g_voltage, motor_spec_.supply_voltage_mv);
503 if (s2g_level == 0 || s2g_level < 2) {
504 s2g_level = 6; // Fallback to default
505 }
506 short_conf.bits.s2g_level = constrain<uint8_t>(s2g_level, 2U, 15U);
507
508 short_conf.bits.shortfilter =
509 constrain<decltype(config.power_stage.shortfilter)>(config.power_stage.shortfilter, 0U, 3U);
510
511 // Calculate shortdelay from detection delay time
512 uint8_t delay_us_x10 = config.power_stage.short_detection_delay_us_x10;
513 if (delay_us_x10 == 0) {
514 delay_us_x10 = 8; // Default recommended: 0.8µs (typical 0.85µs, shortdelay=0)
515 }
516 uint8_t shortdelay = CalculateShortDelay(delay_us_x10);
517 short_conf.bits.shortdelay = constrain<uint8_t>(shortdelay, 0U, 1U);
518

520 return false;

535 pwmconf.bits.pwm_reg = constrain<decltype(config.stealthchop.pwm_reg)>(config.stealthchop.pwm_reg, 0U, 15U);
536 pwmconf.bits.pwm_lim = constrain<decltype(config.stealthchop.pwm_lim)>(config.stealthchop.pwm_lim, 0U, 15U);
537 pwmconf.bits.freewheel = static_cast<uint8_t>(config.stealthchop.freewheel);
538 if (!this->comm_.WriteRegister(Registers::PWMCONF, pwmconf.value, this->GetCommAddress())) {
539 return false;
540 }
541
542 // Set ramp mode to positioning
543 TMC5160_LOG_DEBUG(comm_, 3, "TMC5160", "Initialize: Setting ramp mode to POSITIONING");
544 if (!rampControl.SetRampMode(RampMode::POSITIONING)) {
545 return false;
546 }
547
548 // Configure global settings (GCONF)
549 // Read-Modify-Write to preserve reserved bits (bits 18-31)
550 uint32_t gconf_value = 0;
551 if (!this->comm_.ReadRegister(Registers::GCONF, gconf_value, this->GetCommAddress())) {
552 return false;
553 }
554 GCONF_Register gconf{};
555 gconf.value = gconf_value;
556
557 gconf.bits.recalibrate = config.global_config.recalibrate ? 1 : 0;
558 gconf.bits.faststandstill = config.global_config.faststandstill ? 1 : 0;
559 gconf.bits.en_pwm_mode = config.global_config.en_pwm_mode ? 1 : 0;
560 gconf.bits.multistep_filt = config.global_config.multistep_filt ? 1 : 0;
561 // Use direction from config (global_config.shaft can override via ConfigureGlobalConfig if
562 // needed)
563 gconf.bits.shaft = (config.direction == MotorDirection::INVERSE) ? 1 : 0;
564 gconf.bits.diag0_error = config.global_config.diag0_error ? 1 : 0;
565 gconf.bits.diag0_otpw = config.global_config.diag0_otpw ? 1 : 0;
566 gconf.bits.diag0_stall_step = config.global_config.diag0_stall_step ? 1 : 0;
567 gconf.bits.diag1_stall_dir = config.global_config.diag1_stall_dir ? 1 : 0;
568 gconf.bits.diag1_index = config.global_config.diag1_index ? 1 : 0;
569 gconf.bits.diag1_onstate = config.global_config.diag1_onstate ? 1 : 0;
570 gconf.bits.diag1_steps_skipped = config.global_config.diag1_steps_skipped ? 1 : 0;
571 gconf.bits.diag0_int_pushpull = config.global_config.diag0_int_pushpull ? 1 : 0;
572 gconf.bits.diag1_poscomp_pushpull = config.global_config.diag1_poscomp_pushpull ? 1 : 0;
573 gconf.bits.small_hysteresis = config.global_config.small_hysteresis ? 1 : 0;
574 gconf.bits.stop_enable = config.global_config.stop_enable ? 1 : 0;
575 gconf.bits.direct_mode = config.global_config.direct_mode ? 1 : 0;
576 gconf.bits.test_mode = config.global_config.test_mode ? 1 : 0;
577 // Reserved bits (18-31) are preserved from read value
578
579 if (!this->comm_.WriteRegister(Registers::GCONF, gconf.value, this->GetCommAddress())) {
580 return false;
581 }
582
583 // Configure ramp generator parameters
584 // Set timing parameters first (convert from milliseconds to register values)
585 // TPOWERDOWN: time_ms = tpowerdown * 2^18 / fCLK * 1000
586 // Rearranged: tpowerdown = time_ms * fCLK / (2^18 * 1000)
587 if (config.ramp_config.tpowerdown_ms >= 0.0f) {
588 float tpowerdown_reg = (config.ramp_config.tpowerdown_ms * static_cast<float>(f_clk_)) / (262144.0f * 1000.0f);
589 uint8_t tpowerdown = constrain<uint8_t>(static_cast<uint8_t>(std::round(tpowerdown_reg)), 0U, 255U);
590 if (!rampControl.SetPowerDownDelay(tpowerdown)) {
591 return false;
592 }
593 }
594
595 // TZEROWAIT: time_ms = tzerowait * 2^18 / fCLK * 1000
596 // Rearranged: tzerowait = time_ms * fCLK / (2^18 * 1000)
597 if (config.ramp_config.tzerowait_ms >= 0.0f) {
598 float tzerowait_reg = (config.ramp_config.tzerowait_ms * static_cast<float>(f_clk_)) / (262144.0f * 1000.0f);
599 uint16_t tzerowait = constrain<uint16_t>(static_cast<uint16_t>(std::round(tzerowait_reg)), 0U, 65535U);
600 if (!rampControl.SetZeroWaitTime(tzerowait)) {
601 return false;
602 }
603 }
604
605 // Get unit specifications for proper conversion
606 Unit velocity_unit = config.ramp_config.velocity_unit;
607 Unit acceleration_unit = config.ramp_config.acceleration_unit;
608
609 // Set velocity parameters (VSTART, VSTOP, V1) using specified unit
610 // Use configured values or defaults: VSTART=0, VSTOP=10, V1=0 (disabled)
611 float vstart = config.ramp_config.vstart;
612 float vstop = config.ramp_config.vstop;
613 float v1 = config.ramp_config.v1;
614
615 // Ensure VSTOP >= VSTART (datasheet requirement)
616 // Convert to steps for comparison if needed
617 if (vstop < vstart && vstart > 0.0f) {
618 vstop = vstart; // Use VSTART if VSTOP is less
619 } else if (vstop == 0.0f && vstart == 0.0f) {
620 vstop = 10.0f; // Default minimum VSTOP
621 }
622
623 if (!rampControl.SetRampSpeeds(vstart, vstop, v1, velocity_unit)) {
624 return false;
625 }
626
627 // Set maximum velocity if configured (VMAX) using specified unit
628 if (config.ramp_config.vmax > 0.0f) {
629 if (!rampControl.SetMaxSpeed(config.ramp_config.vmax, velocity_unit)) {
630 return false;
631 }
632 }
633
634 // Set acceleration parameters using specified unit
635 // AMAX and DMAX (used above V1)
636 if (config.ramp_config.amax > 0.0f) {
637 float dmax = (config.ramp_config.dmax > 0.0f) ? config.ramp_config.dmax : config.ramp_config.amax;
638 if (!rampControl.SetAccelerations(config.ramp_config.amax, dmax, acceleration_unit)) {
639 return false;
640 }
641 }
642
643 // A1 (first acceleration, used between VSTART and V1) using specified unit
644 if (config.ramp_config.a1 > 0.0f) {
645 if (!rampControl.SetFirstAcceleration(config.ramp_config.a1, acceleration_unit)) {
646 return false;
647 }
648 }
649
650 // D1 (first deceleration, used between VSTOP and V1, must not be 0 in positioning mode) using specified unit
651 if (config.ramp_config.d1 > 0.0f) {
652 if (!rampControl.SetFinalDeceleration(config.ramp_config.d1, acceleration_unit)) {
653 return false;
654 }
655 } else {
656 // Set default D1 if not configured (required for positioning mode)
657 // Use default in steps/s² (100 steps/s²)
658 if (!rampControl.SetFinalDeceleration(100.0f, Unit::Steps)) {
659 return false;
660 }
661 }
662
663 initialized_ = true;
664 return true;
665}

674template <typename CommType>
675int32_t TMC5160<CommType>::speedToInternal(float speed_hz) const noexcept {
676 // Datasheet formula: v[Hz] = v[5160] * (f_CLK[Hz]/2 / 2^23)
677 // Rearranged: v[5160] = v[Hz] * 2^24 / f_CLK
678 // Where v[Hz] is in μsteps/s (microsteps per second)
679 // Input speed_hz is in steps/s, so we multiply by microstep count (USC) to convert to μsteps/s
680 // Final: v[5160] = (speed_hz * USC) * 2^24 / f_CLK
681 // Note: USC (microstep count) is tracked in current_microsteps_ and can vary (256, 128, 64, etc.)
682 if (speed_hz == 0.0F) {
683 return 0;
684 }
685 float internal = (speed_hz * static_cast<float>(1UL << 24)) / static_cast<float>(f_clk_);
686 internal *= static_cast<float>(current_microsteps_);
687 return static_cast<int32_t>(internal);
688}
689
690template <typename CommType>
691float TMC5160<CommType>::speedFromInternal(int32_t speed_internal) const noexcept {
692 // Datasheet formula: v[Hz] = v[5160] * (f_CLK[Hz]/2 / 2^23)
693 // Where v[Hz] is in μsteps/s (microsteps per second)
694 // Output is in steps/s, so we divide by microstep count (USC) to convert from μsteps/s
695 // Final: v[steps/s] = (v[5160] * f_CLK / 2^24) / USC
696 // Note: USC (microstep count) is tracked in current_microsteps_ and can vary (256, 128, 64, etc.)
697 if (speed_internal == 0) {
698 return 0.0F;
699 }
700 float speed_hz = static_cast<float>(speed_internal) * static_cast<float>(f_clk_) / static_cast<float>(1UL << 24);
701 speed_hz /= static_cast<float>(current_microsteps_);
702 return speed_hz;
703}

705template <typename CommType>
706int32_t TMC5160<CommType>::accelToInternal(float accel_hz) const noexcept {
707 // Datasheet formula: a[Hz/s] = a[5160] * f_CLK[Hz]^2 / (512*256) / 2^24
708 // Rearranged: a[5160] = a[Hz/s] * (512*256) * 2^24 / f_CLK^2
709 // Where a[Hz/s] is in μsteps/s² (microsteps per second squared)
710 // Input accel_hz is in steps/s², so we multiply by microstep count (USC) to convert to μsteps/s²
711 // Final: a[5160] = (accel_hz * USC) * (512*256) * 2^24 / f_CLK^2
712 // Note: USC (microstep count) is tracked in current_microsteps_ and can vary (256, 128, 64, etc.)
713 // Note: The 256 in (512*256) is a fixed constant from the datasheet formula, not the microstep count
714 if (accel_hz == 0.0F) {
715 return 0;
716 }
717 float internal = accel_hz * 512.0F * 256.0F * static_cast<float>(1UL << 24) /
718 (static_cast<float>(f_clk_) * static_cast<float>(f_clk_));
719 internal *= static_cast<float>(current_microsteps_);
720 return static_cast<int32_t>(internal);
721}

723template <typename CommType>
724int32_t TMC5160<CommType>::thresholdSpeedToTstep(float speed_hz) const noexcept {
725 // Datasheet formula: TSTEP = f_CLK / f256STEP = f_CLK / (fSTEP*256/USC)
726 // Where fSTEP is in μsteps/s, USC is microstep count (can vary: 256, 128, 64, etc.)
727 // Input speed_hz is in steps/s, so fSTEP = speed_hz * USC (convert to μsteps/s)
728 // Final: TSTEP = f_CLK / (speed_hz * USC)
729 // Note: USC (microstep count) is tracked in current_microsteps_ and can vary
730 // TSTEP is 20-bit unsigned (max value 0xFFFFF = 1048575)
731 if (speed_hz == 0.0F) {
732 return 0;
733 }
734 float tstep = static_cast<float>(f_clk_) / (speed_hz * static_cast<float>(current_microsteps_));
735 tstep = std::max(0.0F, std::min(1048575.0F, tstep));
736 return static_cast<int32_t>(tstep);
737}

951template <typename CommType>
952bool TMC5160<CommType>::RampControl::ConfigureReferenceSwitch(const ReferenceSwitchConfig& config) noexcept {
953 // Read-Modify-Write to preserve other SW_MODE bits (like sg_stop)
954 uint32_t sw_mode_val = 0;
955 if (!driver_.comm_.ReadRegister(Registers::SW_MODE, sw_mode_val)) {
956 return false;
957 }
958 SW_MODE_Register sw_mode{};
959 sw_mode.value = sw_mode_val;
960
961 // Use stop enable flags (independent of active level)
962 // Allows enabling/disabling motor stop in real-time while keeping polarity configured
963 sw_mode.bits.stop_l_enable = config.left_switch_stop_enable ? 1 : 0;
964 sw_mode.bits.stop_r_enable = config.right_switch_stop_enable ? 1 : 0;
965
966 // Compute polarity from active level (ACTIVE_LOW = inverted polarity = true, ACTIVE_HIGH = normal = false)
967 sw_mode.bits.pol_stop_l = (config.left_switch_active == ReferenceSwitchActiveLevel::ACTIVE_LOW) ? 1 : 0;
968 sw_mode.bits.pol_stop_r = (config.right_switch_active == ReferenceSwitchActiveLevel::ACTIVE_LOW) ? 1 : 0;
969
970 sw_mode.bits.swap_lr = config.swap_left_right ? 1 : 0;
971
972 // Compute latching flags from enum
973 sw_mode.bits.latch_l_active = (config.latch_left == ReferenceLatchMode::ACTIVE_EDGE ||
974 config.latch_left == ReferenceLatchMode::BOTH_EDGES) ? 1 : 0;
975 sw_mode.bits.latch_l_inactive = (config.latch_left == ReferenceLatchMode::INACTIVE_EDGE ||
976 config.latch_left == ReferenceLatchMode::BOTH_EDGES) ? 1 : 0;
977 sw_mode.bits.latch_r_active = (config.latch_right == ReferenceLatchMode::ACTIVE_EDGE ||
978 config.latch_right == ReferenceLatchMode::BOTH_EDGES) ? 1 : 0;
979 sw_mode.bits.latch_r_inactive = (config.latch_right == ReferenceLatchMode::INACTIVE_EDGE ||
980 config.latch_right == ReferenceLatchMode::BOTH_EDGES) ? 1 : 0;
981
982 sw_mode.bits.en_latch_encoder = config.en_latch_encoder ? 1 : 0;
983 sw_mode.bits.en_softstop = (config.stop_mode == ReferenceStopMode::SOFT_STOP) ? 1 : 0;
984
985 return driver_.comm_.WriteRegister(Registers::SW_MODE, sw_mode.value, driver_.GetCommAddress());
986}

988template <typename CommType>
989bool TMC5160<CommType>::RampControl::GetReferenceSwitchConfig(ReferenceSwitchConfig& config) noexcept {
990 uint32_t sw_mode_val = 0;
991 if (!driver_.comm_.ReadRegister(Registers::SW_MODE, sw_mode_val)) {
992 return false;
993 }
994 SW_MODE_Register sw_mode{};
995 sw_mode.value = sw_mode_val;
996
997 // Read stop enable flags
998 config.left_switch_stop_enable = (sw_mode.bits.stop_l_enable != 0);
999 config.right_switch_stop_enable = (sw_mode.bits.stop_r_enable != 0);
1000

1346template <typename CommType>
1347bool TMC5160<CommType>::MotorControl::ConfigureCoolStep(const CoolStepConfig& config) noexcept {
1348 uint32_t coolconf_value = 0;
1349 if (!driver_.comm_.ReadRegister(Registers::COOLCONF, coolconf_value)) {
1350 return false;
1351 }
1352 COOLCONF_Register coolconf{};
1353 coolconf.value = coolconf_value;
1354
1355 // Preserve SGT (StallGuard2 threshold) - it's configured separately via StallGuardConfig
1356 // Only update CoolStep-specific fields
1357
1358 // Calculate SEMIN and SEMAX from user-friendly thresholds
1359 uint8_t semin = 0;
1360 uint8_t semax = 0;
1361
1362 if (config.lower_threshold_sg > 0) {
1363 // Convert SG threshold to SEMIN: threshold = SEMIN * 32
1364 // SEMIN = threshold / 32, clamped to 0-15
1365 semin = static_cast<uint8_t>(config.lower_threshold_sg / 32U);
1366 semin = constrain<decltype(semin)>(semin, 0U, 15U);
1367
1368 // Calculate SEMAX from upper threshold
1369 if (config.upper_threshold_sg > 0) {
1370 // Upper threshold = (SEMIN + SEMAX + 1) * 32
1371 // SEMAX = (upper_threshold / 32) - SEMIN - 1
1372 uint16_t upper_semin_equiv = static_cast<uint16_t>(config.upper_threshold_sg / 32U);
1373 if (upper_semin_equiv > semin) {
1374 semax = static_cast<uint8_t>(upper_semin_equiv - semin - 1U);
1375 semax = constrain<decltype(semax)>(semax, 0U, 15U);
1376 } else {
1377 // Upper threshold must be higher than lower threshold
1378 // If invalid, set semax to 0 (will use minimum hysteresis)
1379 semax = 0;
1380 }
1381 } else {
1382 // If upper threshold not specified, use default hysteresis
1383 // Default: semax = 5 (provides reasonable hysteresis gap)
1384 semax = 5;
1385 }
1386 }
1387 // If lower_threshold_sg = 0, semin = 0 (CoolStep disabled)
1388
1389 // Update CoolStep register fields
1390 coolconf.bits.semin = semin; // 0 = CoolStep disabled
1391 coolconf.bits.semax = semax;
1392
1393 // Convert increment step enum to register value
1394 coolconf.bits.seup = static_cast<uint8_t>(config.increment_step);
1395
1396 // Convert decrement speed enum to register value
1397 coolconf.bits.sedn = static_cast<uint8_t>(config.decrement_speed);
1398
1399 // Convert minimum current enum to register value
1400 coolconf.bits.seimin = (config.min_current == CoolStepMinCurrent::QUARTER_IRUN) ? 1 : 0;
1401
1402 // Update filter enable
1403 coolconf.bits.sfilt = config.enable_filter ? 1 : 0;
1404
1405 // Write COOLCONF register
1406 bool success = driver_.comm_.WriteRegister(Registers::COOLCONF, coolconf.value, driver_.GetCommAddress());
1407
1408 // Configure velocity thresholds if provided
1409 if (success && config.min_velocity > 0.0F) {
1410 float steps_per_sec = driver_.convertSpeedToSteps(config.min_velocity, config.velocity_unit);
1411 int32_t tcoolthrs = driver_.thresholdSpeedToTstep(steps_per_sec);
1412 tcoolthrs = std::min(tcoolthrs, static_cast<decltype(tcoolthrs)>(0xFFFFF)); // 20 bits
1413 success &= driver_.comm_.WriteRegister(Registers::TCOOLTHRS, static_cast<uint32_t>(tcoolthrs));
1414 }
1415
1416 if (success && config.max_velocity > 0.0F) {
1417 float steps_per_sec = driver_.convertSpeedToSteps(config.max_velocity, config.velocity_unit);
1418 int32_t thigh = driver_.thresholdSpeedToTstep(steps_per_sec);
1419 thigh = std::min(thigh, static_cast<decltype(thigh)>(0xFFFFF)); // 20 bits
1420 success &= driver_.comm_.WriteRegister(Registers::THIGH, static_cast<uint32_t>(thigh));
1421 }
1422
1423 return success;
1424}

1426template <typename CommType>
1427bool TMC5160<CommType>::MotorControl::ConfigureDcStep(const DcStepConfig& config) noexcept {
1428 // Convert velocity threshold to internal format with unit support
1429 int32_t vdc_min = 0;
1430 if (config.min_velocity > 0.0F) {
1431 float steps_per_sec = driver_.convertSpeedToSteps(config.min_velocity, config.velocity_unit);
1432 vdc_min = driver_.speedToInternal(steps_per_sec);
1433 // VDCMIN is 23-bit register (0...2^22), but only bits 22..8 are used for comparison
1434 // Lower 8 bits (7..0) are ignored/unused by the hardware comparator.
1435 // We mask the value to 0x7FFF00 to explicitly zero out the unused bits,
1436 // ensuring the register value reflects the effective threshold.
1437 // Note: No shift is required; the register compares bits 22..8 of VDCMIN
1438 // against bits 22..8 of VACTUAL.
1439 vdc_min = vdc_min & 0x7FFF00;
1440 }
1441 bool success = driver_.comm_.WriteRegister(Registers::VDCMIN, static_cast<uint32_t>(vdc_min));
1442 if (!success) {
1443 return false;
1444 }
1445
1446 // Calculate DC_TIME from PWM on-time (microseconds) or auto-calculate from blank time
1447 uint16_t dc_time = 0;
1448
1449 if (config.pwm_on_time_us > 0.0F) {
1450 // Convert microseconds to clock cycles: cycles = (time_us * f_clk) / 1e6
1451 float clock_cycles = (config.pwm_on_time_us * static_cast<float>(driver_.f_clk_)) / 1000000.0F;
1452 dc_time = static_cast<uint16_t>(std::round(clock_cycles));
1453 dc_time = constrain<decltype(dc_time)>(dc_time, 0U, 1023U);
1454 } else {
1455 // Auto-calculate from blank time (TBL) if not specified
1456 // Read current CHOPCONF to get TBL value
1457 uint32_t chopconf_value = 0;
1458 if (driver_.comm_.ReadRegister(Registers::CHOPCONF, chopconf_value)) {
1459 CHOPCONF_Register chopconf{};
1460 chopconf.value = chopconf_value;
1461
1462 // TBL values: 0=16 clocks, 1=24 clocks, 2=36 clocks, 3=54 clocks
1463 uint8_t tbl_clocks[] = {16, 24, 36, 54};
1464 uint8_t tbl_index = constrain<uint8_t>(chopconf.bits.tbl, 0U, 3U);
1465 uint8_t blank_time_clocks = tbl_clocks[tbl_index];
1466
1467 // DC_TIME should be set slightly above blank time
1468 // Datasheet: Lower limit = TBL + n (where n = 1-100 for typical motor)
1469 // Use n = 20 as a reasonable default (provides margin without being too conservative)
1470 dc_time = blank_time_clocks + 20;
1471 dc_time = constrain<decltype(dc_time)>(dc_time, 0U, 1023U);
1472 } else {
1473 // If can't read CHOPCONF, use conservative default (36 clocks + 20 = 56)
1474 dc_time = 56;
1475 }
1476 }
1477
1478 // Calculate DC_SG from DC_TIME based on sensitivity
1479 uint8_t dc_sg = 0;
1480 if (config.stall_sensitivity != DcStepStallSensitivity::DISABLED && dc_time > 0) {
1481 // DC_SG should be set slightly higher than DC_TIME/16
1482 // Different sensitivity levels use different multipliers
1483 float dc_sg_float = 0.0F;
1484 switch (config.stall_sensitivity) {
1485 case DcStepStallSensitivity::LOW:
1486 dc_sg_float = static_cast<float>(dc_time) / 20.0F; // Less sensitive
1487 break;
1488 case DcStepStallSensitivity::MODERATE:
1489 dc_sg_float = static_cast<float>(dc_time) / 16.0F; // Recommended (datasheet default)
1490 break;
1491 case DcStepStallSensitivity::HIGH:
1492 dc_sg_float = static_cast<float>(dc_time) / 12.0F; // More sensitive
1493 break;
1494 default:
1495 dc_sg_float = 0.0F;
1496 break;
1497 }
1498 dc_sg = static_cast<uint8_t>(std::round(dc_sg_float));
1499 // Ensure minimum value of 1 if sensitivity is enabled
1500 if (dc_sg == 0 && config.stall_sensitivity != DcStepStallSensitivity::DISABLED) {
1501 dc_sg = 1;
1502 }
1503 dc_sg = constrain<decltype(dc_sg)>(dc_sg, 0U, 255U);
1504 }
1505
1506 // Configure DCCTRL register
1507 DCCTRL_Register dcctrl{};
1508 dcctrl.bits.dc_time = dc_time;
1509 dcctrl.bits.dc_sg = dc_sg;
1510 success &= driver_.comm_.WriteRegister(Registers::DCCTRL, dcctrl.value, driver_.GetCommAddress());
1511
1512 // Configure stop on stall if requested
1513 if (success && config.stop_on_stall && config.stall_sensitivity != DcStepStallSensitivity::DISABLED) {
1514 // Read current SW_MODE register
1515 uint32_t sw_mode_value = 0;
1516 if (driver_.comm_.ReadRegister(Registers::SW_MODE, sw_mode_value)) {
1517 SW_MODE_Register sw_mode{};
1518 sw_mode.value = sw_mode_value;
1519 sw_mode.bits.sg_stop = 1; // Enable stop on stall
1520 success &= driver_.comm_.WriteRegister(Registers::SW_MODE, sw_mode.value, driver_.GetCommAddress());
1521 }
1522 }
1523
1524 // Ensure CHOPCONF.vhighfs and CHOPCONF.vhighchm are set for DcStep
1525 // These flags switch to fullstepping when VDCMIN is exceeded
1526 // Note: These can also be set via ChopperConfig.vhighfs and ChopperConfig.vhighchm
1527 if (success && config.min_velocity > 0.0F) {
1528 uint32_t chopconf_value = 0;
1529 if (driver_.comm_.ReadRegister(Registers::CHOPCONF, chopconf_value)) {
1530 CHOPCONF_Register chopconf{};
1531 chopconf.value = chopconf_value;
1532 chopconf.bits.vhighfs = 1; // Enable fullstepping at high velocity
1533 chopconf.bits.vhighchm = 1; // Enable chopper mode switching at high velocity
1534 success &= driver_.comm_.WriteRegister(Registers::CHOPCONF, chopconf.value, driver_.GetCommAddress());
1535 }
1536 }
1537
1538 return success;
1539}

1764template <typename CommType>
1765bool TMC5160<CommType>::MotorControl::SetupMotorFromSpec(const MotorSpec& motor_spec,
1766 const MechanicalSystem* mechanical_system) noexcept {
1767 // Calculate global scaler based on rated current
1768 // Typical calculation: global_scaler = (rated_current_ma * 32) / (irun * sense_resistor_current)
1769 // For simplicity, we'll use a basic calculation
1770 uint16_t global_scaler = 32;
1771 if (motor_spec.rated_current_ma > 0) {
1772 // Basic calculation: assume 1.5A max current, scale accordingly
1773 global_scaler = static_cast<uint16_t>(std::min(256U, std::max(32U, (motor_spec.rated_current_ma * 32U) / 1500U)));
1774 }
1775
1776 // Set global scaler
1777 if (!SetGlobalScaler(global_scaler)) {
1778 return false;
1779 }
1780
1781 // Calculate irun and ihold from rated current
1782 // irun should be between 16-31 for best performance
1783 // We'll use 80% of rated current for irun, 30% for ihold
1784 uint8_t irun = 16;
1785 uint8_t ihold = 0;
1786 if (motor_spec.rated_current_ma > 0 && global_scaler > 0) {
1787 // Calculate irun: target current = (irun/32) * (global_scaler/32) * sense_resistor_current
1788 // Simplified: irun = (target_current * 32) / (global_scaler * sense_resistor_current / 32)
1789 // Assuming sense resistor gives ~1.5A at irun=31, global_scaler=32
1790 float target_run_current = static_cast<float>(motor_spec.rated_current_ma) * 0.8F;
1791 auto irun_calc =
1792 static_cast<uint32_t>((target_run_current * 32.0F) / (static_cast<float>(global_scaler) * 0.046875F));
1793 irun = static_cast<uint8_t>(std::min(static_cast<uint32_t>(31U), std::max(static_cast<uint32_t>(16U), irun_calc)));
1794 float target_hold_current = static_cast<float>(motor_spec.rated_current_ma) * 0.3F;
1795 auto ihold_calc =
1796 static_cast<uint32_t>((target_hold_current * 32.0F) / (static_cast<float>(global_scaler) * 0.046875F));
1797 ihold = static_cast<uint8_t>(std::min(static_cast<uint32_t>(31U), ihold_calc));
1798 }
1799
1800 // Set motor current
1801 if (!SetCurrent(irun, ihold)) {
1802 return false;
1803 }
1804
1805 // Configure chopper based on inductance if available
1806 if (motor_spec.winding_inductance_uh > 0) {
1807 ChopperConfig chop_config{};
1808 // Higher inductance may need different settings
1809 // This is a simplified heuristic
1810 if (motor_spec.winding_inductance_uh > 3000) {
1811 chop_config.tbl = 3; // Longer blank time for high inductance
1812 }
1813 ConfigureChopper(chop_config);
1814 }
1815
1816 // Note: mechanical_system is stored for unit conversions but not used here
1817 // as it's used by the unit conversion functions, not motor setup
1818
1819 return true;
1820}

1975template <typename CommType>
1976bool TMC5160<CommType>::Encoder::SetResolution(int32_t motor_steps, int32_t enc_resolution, bool inverted) noexcept {
1977 // Calculate factor: (motor_steps * microsteps) / enc_resolution
1978 // Use current microstep setting (may vary: 256, 128, 64, etc.)
1979 float factor = static_cast<float>(motor_steps * driver_.current_microsteps_) / static_cast<float>(enc_resolution);
1980
1981 // Check if binary prescaler gives exact match
1982 auto enc_const_binary = static_cast<int32_t>(factor * 65536.0F);
1983 if (enc_const_binary * enc_resolution == motor_steps * driver_.current_microsteps_ * 65536) {
1984 // Use binary mode
1985 uint32_t encmode_value = 0;
1986 if (!driver_.comm_.ReadRegister(Registers::ENCMODE, encmode_value)) {
1987 return false;

2278 int32_t& final_position, uint32_t timeout_ms) noexcept {

2280 StallGuardConfig sg_config{};

2282 sg_config.enable_filter = true; // Enable filter for stable readings
2283 if (!ConfigureStallGuard(sg_config)) {

2286
2287 // Enable StallGuard2 stop in SW_MODE
2288 uint32_t sw_mode_value = 0;

2290 return false;
2291 }

2310 if (!driver_.rampControl.SetRampMode(mode)) {

2312 }

2315 // Velocity mode requires AMAX > 0 to actually accelerate from VSTART to VMAX
2316 // Use reasonable acceleration: reach VMAX in ~0.1 seconds
2317 float acceleration = std::max(search_speed * 10.0f, 50000.0f); // At least 50k steps/s²
2318 if (!driver_.rampControl.SetAcceleration(acceleration)) {
2319 return false;
2320 }
2321 if (!driver_.rampControl.SetDeceleration(acceleration)) {
2322 return false;
2323 }
2324
2325 // Set VSTART > 0 to actually start motion in velocity mode
2326 // VSTART should be reasonable but less than VMAX - use 10% of search speed or minimum 1000 steps/s
2327 float vstart_speed = std::max(search_speed * 0.1f, 1000.0f);
2328 if (!driver_.rampControl.SetRampSpeeds(vstart_speed, 100.0f, 0.0f, Unit::Steps)) {
2329 return false;
2330 }

2332 return false;

2373template <typename CommType>
2374bool TMC5160<CommType>::Diagnostics::PerformSwitchHoming(bool direction, float search_speed, float switch_speed,
2375 int32_t& final_position, bool use_left_switch, uint32_t timeout_ms) noexcept {
2376 // Complete homing procedure per datasheet section 12.4:
2377 // Step 1: Make sure switch is not pressed (move away from switch)
2378 // This is user responsibility - we assume switch is not pressed at start
2379
2380 // Step 2: Activate position latching and motor stop upon switch event
2381 SW_MODE_Register sw_mode{};
2382 // Read current SW_MODE to preserve other settings (like sg_stop)
2383 uint32_t sw_mode_val = 0;
2384 if (!driver_.comm_.ReadRegister(Registers::SW_MODE, sw_mode_val)) {
2385 return false;
2386 }
2387 sw_mode.value = sw_mode_val;
2388
2389 // Configure latching and stop enable based on switch choice
2390 // Note: This assumes switch is already configured via ConfigureReferenceSwitch
2391 // We're just enabling latching and stop for the homing procedure
2392 if (use_left_switch) {
2393 sw_mode.bits.latch_l_active = true; // Latch on active edge for homing
2394 sw_mode.bits.stop_l_enable = true; // Enable stop on left switch
2395 } else {
2396 sw_mode.bits.latch_r_active = true; // Latch on active edge for homing
2397 sw_mode.bits.stop_r_enable = true; // Enable stop on right switch
2398 }
2399 // Use hard stop for precise homing (per datasheet 12.4 recommendation)
2400 // Hard stop ensures motor stops exactly at switch position (no overshoot)
2401 sw_mode.bits.en_softstop = false;
2402
2403 if (!driver_.comm_.WriteRegister(Registers::SW_MODE, sw_mode.value)) {
2404 return false;
2405 }
2406
2407 // Step 3: Start motion ramp into direction of switch
2408 // Move to a more negative position for left switch, more positive for right switch
2409 RampMode mode = direction ? RampMode::VELOCITY_POS : RampMode::VELOCITY_NEG;
2410 if (!driver_.rampControl.SetRampMode(mode)) {
2411 return false;
2412 }
2413
2414 // Set acceleration before speed (required for velocity mode)
2415 float acceleration = std::max(search_speed * 10.0f, 50000.0f); // At least 50k steps/s²
2416 if (!driver_.rampControl.SetAcceleration(acceleration)) {
2417 return false;
2418 }
2419 if (!driver_.rampControl.SetDeceleration(acceleration)) {
2420 return false;
2421 }
2422
2423 if (!driver_.rampControl.SetMaxSpeed(search_speed)) {
2424 return false;
2425 }
2426
2427 // Step 3 (continued): Wait for switch hit (motor stops automatically)
2428 bool switch_hit = false;
2429 uint32_t loops = timeout_ms * 100; // 10ms per loop
2430 for (uint32_t i = 0; i < loops; i++) {
2431 uint32_t ramp_stat = 0;
2432 if (driver_.comm_.ReadRegister(Registers::RAMP_STAT, ramp_stat)) {
2433 RAMP_STAT_Register status{};
2434 status.value = ramp_stat;
2435
2436 // Check for specific stop event
2437 if ((use_left_switch && status.bits.event_stop_l) ||
2438 (!use_left_switch && status.bits.event_stop_r)) {
2439 switch_hit = true;
2440 break;
2441 }
2442 // Also check vzero if we missed the event flag but stopped
2443 if (status.bits.vzero && ((use_left_switch && status.bits.status_stop_l) || (!use_left_switch && status.bits.status_stop_r))) {
2444 switch_hit = true;
2445 break;
2446 }
2447 }
2448 // Small delay for polling (10ms)
2449 driver_.comm_.DelayMs(10);
2450 }
2451
2452 // Ensure motor is stopped
2453 driver_.rampControl.Stop();
2454
2455 if (!switch_hit) {
2456 // Timeout - disable stop function and return
2457 if (use_left_switch) {
2458 sw_mode.bits.stop_l_enable = false;
2459 } else {
2460 sw_mode.bits.stop_r_enable = false;
2461 }
2462 driver_.comm_.WriteRegister(Registers::SW_MODE, sw_mode.value);
2463 return false;
2464 }
2465
2466 // Step 4: Wait until motor is in standstill (poll VACTUAL or check vzero/standstill flag)
2467 // Per datasheet: "Wait until the motor is in standstill again by polling the actual velocity
2468 // VACTUAL or checking vzero or the standstill flag"
2469 uint32_t standstill_loops = 0;
2470 const uint32_t max_standstill_loops = 1000; // 10 seconds max wait
2471 bool in_standstill = false;
2472
2473 while (standstill_loops < max_standstill_loops) {
2474 uint32_t ramp_stat = 0;
2475 if (driver_.comm_.ReadRegister(Registers::RAMP_STAT, ramp_stat)) {
2476 RAMP_STAT_Register status{};
2477 status.value = ramp_stat;
2478
2479 // Check vzero flag (velocity reached zero)
2480 if (status.bits.vzero) {
2481 // Also verify VACTUAL is near zero
2482 float vactual = driver_.rampControl.GetCurrentSpeed();
2483 if (std::abs(vactual) < 1.0f) { // Less than 1 step/s
2484 in_standstill = true;
2485 break;
2486 }
2487 }
2488 }
2489 standstill_loops++;
2490 // Small delay for polling (10ms)
2491 driver_.comm_.DelayMs(10);
2492 }
2493
2494 if (!in_standstill) {
2495 // Motor didn't reach standstill - disable stop and return error
2496 if (use_left_switch) {
2497 sw_mode.bits.stop_l_enable = false;
2498 } else {
2499 sw_mode.bits.stop_r_enable = false;
2500 }
2501 driver_.comm_.WriteRegister(Registers::SW_MODE, sw_mode.value);
2502 return false;
2503 }
2504
2505 // Step 5: Switch to hold mode and calculate difference between latched and actual position
2506 // Per datasheet: "Switch the ramp generator to hold mode and calculate the difference
2507 // between the latched position and the actual position. For StallGuard based homing or
2508 // when using hard stop, XACTUAL stops exactly at the home position, so there is no difference (0)."
2509
2510 // Switch to hold mode
2511 if (!driver_.rampControl.SetRampMode(RampMode::HOLD)) {
2512 return false;
2513 }
2514
2515 // Read latched position (XLATCH) - captured at switch hit
2516 int32_t latched_position = driver_.rampControl.GetLatchedPosition();
2517
2518 // Read actual position (XACTUAL) - current motor position
2519 int32_t actual_position = driver_.rampControl.GetCurrentPosition();
2520
2521 // Calculate difference
2522 // With hard stop, XACTUAL stops exactly at switch position, so XLATCH ≈ XACTUAL
2523 // The difference represents any offset between latched and actual position
2524 int32_t position_difference = latched_position - actual_position;
2525
2526 // Step 6: Write the calculated difference into the actual position register
2527 // Per datasheet: "Write the calculated difference into the actual position register.
2528 // Now, homing is finished. A move to position 0 will bring back the motor exactly to the switching point."
2529 //
2530 // To make the switch position = home (0):
2531 // - Current: XACTUAL = switch_position, XLATCH = switch_position (captured)
2532 // - Goal: When motor is at switch, XACTUAL = 0
2533 // - Solution: Set XACTUAL = XACTUAL - latched_position
2534 // This makes: new_XACTUAL = switch_position - switch_position = 0
2535 // And latched position becomes: XLATCH - (switch_position - switch_position) = 0
2536 //
2537 // With hard stop, difference should be ~0, so this is effectively setting XACTUAL = 0
2538 int32_t new_position = actual_position - latched_position;
2539 if (!driver_.rampControl.SetCurrentPosition(static_cast<float>(new_position), Unit::Steps, false)) {
2540 return false;
2541 }
2542
2543 // Return final position (should be 0 after homing, representing home position)
2544 final_position = driver_.rampControl.GetCurrentPosition();
2545
2546 // Disable stop function to allow moving away from switch
2547 if (use_left_switch) {
2548 sw_mode.bits.stop_l_enable = false;
2549 sw_mode.bits.latch_l_active = false;
2550 } else {
2551 sw_mode.bits.stop_r_enable = false;
2552 sw_mode.bits.latch_r_active = false;
2553 }
2554 driver_.comm_.WriteRegister(Registers::SW_MODE, sw_mode.value);
2555
2556 return true;
2557}
