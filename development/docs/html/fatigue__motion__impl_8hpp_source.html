213bool FatigueTestMotion::Start() noexcept {
214 uint32_t current_cycles, target_cycles;
215 float min_pos, max_pos, current_pos;
216
217 {
218 TmcMutexGuard guard(mutex_);
219 if (fabsf(local_min_bound_) < 0.01f && fabsf(local_max_bound_) < 0.01f) {
220 ESP_LOGE(TAG_MOTION, "Cannot start: local bounds not set!");
221 return false;
222 }
223
224 if (cycle_complete_) {
225 ESP_LOGW(TAG_MOTION, "Cycle count reached. Reset cycles or set new target to continue.");
226 return false;
227 }
228
229 // Update trajectory before starting
230 RecalculateTrajectory();
231
232 // Configure driver for positioning mode
233 // Convert calculated values from steps/s to higher-level units (RPM for velocity, RevPerSec for acceleration)
234 float vmax_rpm = (calculated_vmax_ / static_cast<float>(steps_per_rev_)) * 60.0f;
235 float amax_rev_s2 = calculated_amax_ / static_cast<float>(steps_per_rev_);
236
237 driver_->rampControl.SetRampMode(tmc51x0::RampMode::POSITIONING);
238 driver_->rampControl.SetMaxSpeed(vmax_rpm, tmc51x0::Unit::RPM);
239 driver_->rampControl.SetAcceleration(amax_rev_s2, tmc51x0::Unit::RevPerSec);
240 driver_->rampControl.SetDeceleration(amax_rev_s2, tmc51x0::Unit::RevPerSec);
241 driver_->rampControl.SetRampSpeeds(30.0f, 3.0f, 0.0f, tmc51x0::Unit::RPM); // ~1000/100 steps/s for 200 steps/rev
242
243 running_ = true;
244 start_time_us_ = esp_timer_get_time();
245 sinusoidal_mode_ = true; // Use sinusoidal mode
246
247 // Determine initial state based on current position
248 auto current_pos_result = driver_->rampControl.GetCurrentPosition(tmc51x0::Unit::Deg);
249 float current_pos_deg = current_pos_result.IsOk() ? current_pos_result.Value() : 0.0f;
250 float min_pos_deg = local_min_bound_;
251 float max_pos_deg = local_max_bound_;
252
253 // Find closest bound or determine direction
254 float dist_to_min = fabsf(current_pos_deg - min_pos_deg);
255 float dist_to_max = fabsf(current_pos_deg - max_pos_deg);
256
257 // Default to moving towards min unless we're already there
258 if (dist_to_min < 1.0f) { // ~1 degree threshold
259 state_ = MotionState::MOVING_TO_MAX;
260 } else {
261 state_ = MotionState::MOVING_TO_MIN;
262 }
263
264 current_cycles = current_cycles_;
265 target_cycles = target_cycles_;
266 }
267
268 ESP_LOGI(TAG_MOTION, "Starting fatigue test (cycles: %lu/%lu)", current_cycles,
269 target_cycles == 0 ? 0xFFFFFFFF : target_cycles);
270 ESP_LOGI(TAG_MOTION, " Motion: Sinusoidal mode, VMAX=%.1f, AMAX=%.1f",
271 calculated_vmax_, calculated_amax_);
272 return true;
273}

292void FatigueTestMotion::RecalculateTrajectory() noexcept {
293 // Calculate total travel distance (one way) in degrees
294 float distance_deg = fabsf(local_max_bound_ - local_min_bound_);
295 if (distance_deg < 0.1f || frequency_hz_ <= 0.0001f) {
296 calculated_vmax_ = 1000.0f;
297 calculated_amax_ = 5000.0f;
298 estimated_frequency_hz_ = 0.0f;
299 return;
300 }
301
302 // Target cycle period (total time for there + back + dwells)
303 float target_period_s = 1.0f / frequency_hz_;
304
305 // Total dwell time per cycle (min + max)
306 float total_dwell_s = (dwell_at_min_ms_ + dwell_at_max_ms_) / 1000.0f;
307
308 // Time available for motion (there + back)
309 float total_move_time_s = target_period_s - total_dwell_s;
310
311 if (total_move_time_s <= 0.0f) {
312 total_move_time_s = 0.1f;
313 ESP_LOGW(TAG_MOTION, "Requested frequency %.2f Hz is impossible with given dwell times!", frequency_hz_);
314 }
315
316 // Time for one leg (one way)
317 float leg_time_s = total_move_time_s / 2.0f;
318
319 // Convert distance from degrees to steps for internal calculations
320 // (calculated_vmax_ and calculated_amax_ are stored in steps/s and steps/s²)
321 float distance_steps = tmc51x0::DegreesToSteps(distance_deg, steps_per_rev_);
322
323 // Calculate VMAX and AMAX for Trapezoidal Profile (in steps/s and steps/s²)
324 calculated_vmax_ = (1.5f * distance_steps) / leg_time_s;
325 calculated_amax_ = calculated_vmax_ / (leg_time_s / 3.0f);
326
327 // Clamp to driver limits
328 if (calculated_vmax_ > 5000000.0f) calculated_vmax_ = 5000000.0f;
329 if (calculated_amax_ > 5000000.0f) calculated_amax_ = 5000000.0f;
330
331 estimated_frequency_hz_ = 1.0f / (2.0f * leg_time_s + total_dwell_s);
332
333 // Calculate values in higher-level units for logging
334 float vmax_rpm = (calculated_vmax_ / static_cast<float>(steps_per_rev_)) * 60.0f;
335 float amax_rev_s2 = calculated_amax_ / static_cast<float>(steps_per_rev_);
336
337 ESP_LOGI(TAG_MOTION, "Trajectory Recalculated: Dist=%.2f° (%.0f steps), LegTime=%.3fs", distance_deg, distance_steps, leg_time_s);
338 ESP_LOGI(TAG_MOTION, " Target Freq=%.2fHz, Est Freq=%.2fHz", frequency_hz_, estimated_frequency_hz_);
339 ESP_LOGI(TAG_MOTION, " VMAX=%.1f steps/s (%.2f RPM), AMAX=%.1f steps/s² (%.3f rev/s²)",
340 calculated_vmax_, vmax_rpm, calculated_amax_, amax_rev_s2);
341}

343void FatigueTestMotion::ClipLocalBoundsToGlobal() noexcept {
344 if (!bounded_)
345 return;
346
347 float old_min = local_min_bound_;
348 float old_max = local_max_bound_;
349
350 // Clip local bounds to global bounds (all in degrees)
351 local_min_bound_ = std::max(local_min_bound_, global_min_bound_);
352 local_max_bound_ = std::min(local_max_bound_, global_max_bound_);
353
354 // Update home position
355 home_position_ = (local_min_bound_ + local_max_bound_) / 2.0f;
356 amplitude_ = (local_max_bound_ - local_min_bound_) / 2.0f;
357
358 if (fabsf(old_min - local_min_bound_) > 0.01f || fabsf(old_max - local_max_bound_) > 0.01f) {
359 ESP_LOGW(TAG_MOTION, "Local bounds clipped to global bounds");
360 ESP_LOGI(TAG_MOTION, "Clipped local bounds: min=%.2f°, max=%.2f°", local_min_bound_, local_max_bound_);
361 }

364void FatigueTestMotion::UpdateSinuousMotion() noexcept {
365 uint64_t elapsed_us;
366 float freq, amp;
367 float home, local_min, local_max;
368 uint32_t target_cycles;
369 bool cycle_started;
370 float last_target_rel;
371 uint32_t dwell_min, dwell_max;
372 float phase_off;
373
374 {
375 TmcMutexGuard guard(mutex_);
376 elapsed_us = esp_timer_get_time() - start_time_us_;
377 freq = frequency_hz_;
378 amp = amplitude_;
379 home = home_position_;
380 local_min = local_min_bound_;
381 local_max = local_max_bound_;
382 target_cycles = target_cycles_;
383 cycle_started = cycle_started_;
384 last_target_rel = last_target_relative_;
385 dwell_min = dwell_at_min_ms_;
386 dwell_max = dwell_at_max_ms_;
387 phase_off = phase_offset_;
388 }
389
390 // Calculate sinusoidal position
391 double elapsed_s = elapsed_us / 1000000.0;
392 double angle = 2.0 * M_PI * freq * elapsed_s + phase_off;
393 double sin_value = sin(angle);
394
395 // Calculate target position in degrees
396 float target_deg = home + static_cast<float>(amp * sin_value);
397
398 // Get current position relative to center for cycle counting
399 auto current_pos_result = driver_->rampControl.GetCurrentPosition(tmc51x0::Unit::Deg);
400 float current_pos_deg = current_pos_result.IsOk() ? current_pos_result.Value() : 0.0f;
401 float target_relative = target_deg - home;
402
403 // Cycle counting: one cycle = center → extreme → center
404 bool currently_negative = (target_relative < 0.0f);
405 bool last_was_negative = (last_target_rel < 0.0f);
406 bool crossing_center =
407 (last_was_negative != currently_negative) && (fabsf(target_relative) < 1.0f) && (fabsf(last_target_rel) < 1.0f);
408
409 // If we've started a cycle (left center) and now crossing back through center
410 if (cycle_started && crossing_center) {
411 // Completed a cycle
412 uint32_t new_cycles;
413 {
414 TmcMutexGuard guard(mutex_);
415 current_cycles_++;
416 cycle_started_ = false;
417 new_cycles = current_cycles_;
418
419 // Check if target reached
420 if (target_cycles > 0 && new_cycles >= target_cycles) {
421 cycle_complete_ = true;
422 running_ = false;
423 state_ = MotionState::STOPPED;
424 guard.unlock();
425 driver_->rampControl.Stop();
426 ESP_LOGI(TAG_MOTION, "Target cycle count reached: %lu cycles. Stopping.", new_cycles);
427 return;
428 }
429 }
430 ESP_LOGI(TAG_MOTION, "Cycle %lu completed at center (target: %lu)", new_cycles,
431 target_cycles == 0 ? 0xFFFFFFFF : target_cycles);
432 } else if (!cycle_started && fabsf(target_relative) > 1.0f) {
433 // We've left center, cycle has started
434 TmcMutexGuard guard(mutex_);
435 cycle_started_ = true;
436 last_was_negative_ = currently_negative;
437 }
438
439 // Update tracking
440 {
441 TmcMutexGuard guard(mutex_);
442 last_target_relative_ = target_relative;
443 if (fabsf(target_relative) > 0.5f) {
444 last_was_negative_ = currently_negative;
445 }
446 }
447
448 // Clamp to local bounds and handle dwell states
449 // Use ABSOLUTE positioning - home is established via bounds finding or SetUnbounded()
450 // target_deg is calculated as home + amplitude * sin(), so it's already absolute
451 if (target_deg <= local_min) {
452 target_deg = local_min;
453 if (dwell_min > 0) {
454 TmcMutexGuard guard(mutex_);
455 state_ = MotionState::DWELL_AT_MIN;
456 dwell_start_time_ms_ = esp_timer_get_time() / 1000;
457 driver_->rampControl.SetTargetPosition(target_deg, tmc51x0::Unit::Deg);
458 return;
459 } else {
460 TmcMutexGuard guard(mutex_);
461 state_ = MotionState::MOVING_TO_MAX;
462 }
463 } else if (target_deg >= local_max) {
464 target_deg = local_max;
465 if (dwell_max > 0) {
466 TmcMutexGuard guard(mutex_);
467 state_ = MotionState::DWELL_AT_MAX;
468 dwell_start_time_ms_ = esp_timer_get_time() / 1000;
469 driver_->rampControl.SetTargetPosition(target_deg, tmc51x0::Unit::Deg);
470 return;
471 } else {
472 TmcMutexGuard guard(mutex_);
473 state_ = MotionState::MOVING_TO_MIN;
474 }
475 } else {
476 TmcMutexGuard guard(mutex_);
477 if (target_relative > 0.0f) {
478 state_ = MotionState::MOVING_TO_MAX;
479 } else {
480 state_ = MotionState::MOVING_TO_MIN;
481 }
482 }
483
484 // Update target position if it changed significantly
485 // Use ABSOLUTE positioning - home is established, target_deg is absolute
486 if (fabsf(target_deg - current_pos_deg) > 0.5f) { // ~0.5 degree threshold
487 driver_->rampControl.SetRampMode(tmc51x0::RampMode::POSITIONING);
488 // Use higher-level units for slow motion (RPM for velocity, RevPerSec for acceleration)
489 float slow_vmax_rpm = (1000.0f / static_cast<float>(steps_per_rev_)) * 60.0f;
490 float slow_amax_rev_s2 = 2000.0f / static_cast<float>(steps_per_rev_);
491
492 driver_->rampControl.SetTargetPosition(target_deg, tmc51x0::Unit::Deg);
493 driver_->rampControl.SetMaxSpeed(slow_vmax_rpm, tmc51x0::Unit::RPM);
494 driver_->rampControl.SetAcceleration(slow_amax_rev_s2, tmc51x0::Unit::RevPerSec);
495 }
496}

498void FatigueTestMotion::Update() noexcept {
499 {
500 TmcMutexGuard guard(mutex_);
501 if (!running_ || state_ == MotionState::STOPPED) {
502 return;
503 }
504
505 // Check if cycle count reached
506 if (target_cycles_ > 0 && current_cycles_ >= target_cycles_) {
507 if (!cycle_complete_) {
508 cycle_complete_ = true;
509 uint32_t cycles = current_cycles_;
510 state_ = MotionState::STOPPED;
511 running_ = false;
512 guard.unlock();
513 driver_->rampControl.Stop();
514 ESP_LOGI(TAG_MOTION, "Target cycle count reached: %lu cycles. Stopping.", cycles);
515 }
516 return;
517 }
518 }
519
520 // Check if we're in sinusoidal mode
521 bool use_sinusoidal;
522 MotionState current_state;
523 {
524 TmcMutexGuard guard(mutex_);
525 use_sinusoidal = sinusoidal_mode_;
526 current_state = state_;
527 }
528
529 // If sinusoidal mode and not in a dwell state, use sinusoidal update
530 if (use_sinusoidal && current_state != MotionState::DWELL_AT_MIN &&
531 current_state != MotionState::DWELL_AT_MAX) {
532 UpdateSinuousMotion();
533 return;
534 }
535
536 // Handle dwell states
537 uint32_t current_time_ms = esp_timer_get_time() / 1000;
538 uint32_t dwell_min, dwell_max, dwell_start;
539 float min_bound, max_bound;
540
541 {
542 TmcMutexGuard guard(mutex_);
543 current_state = state_;
544 dwell_min = dwell_at_min_ms_;
545 dwell_max = dwell_at_max_ms_;
546 dwell_start = dwell_start_time_ms_;
547 min_bound = local_min_bound_; // Already in degrees
548 max_bound = local_max_bound_; // Already in degrees
549 }
550
551 switch (current_state) {
552 case MotionState::DWELL_AT_MIN:
553 if (current_time_ms - dwell_start >= dwell_min) {
554 TmcMutexGuard guard(mutex_);
555 if (!sinusoidal_mode_) {
556 // Convert to higher-level units (RPM for velocity, RevPerSec for acceleration)
557 float vmax_rpm = (calculated_vmax_ / static_cast<float>(steps_per_rev_)) * 60.0f;
558 float amax_rev_s2 = calculated_amax_ / static_cast<float>(steps_per_rev_);
559
560 state_ = MotionState::MOVING_TO_MAX;
561 driver_->rampControl.SetMaxSpeed(vmax_rpm, tmc51x0::Unit::RPM);
562 driver_->rampControl.SetAcceleration(amax_rev_s2, tmc51x0::Unit::RevPerSec);
563 driver_->rampControl.SetDeceleration(amax_rev_s2, tmc51x0::Unit::RevPerSec);
564 driver_->rampControl.SetTargetPosition(max_bound, tmc51x0::Unit::Deg);
565 } else {
566 state_ = MotionState::MOVING_TO_MAX;
567 }
568 }
569 break;
570
571 case MotionState::DWELL_AT_MAX:
572 if (current_time_ms - dwell_start >= dwell_max) {
573 TmcMutexGuard guard(mutex_);
574 if (!sinusoidal_mode_) {
575 // Convert to higher-level units (RPM for velocity, RevPerSec for acceleration)
576 float vmax_rpm = (calculated_vmax_ / static_cast<float>(steps_per_rev_)) * 60.0f;
577 float amax_rev_s2 = calculated_amax_ / static_cast<float>(steps_per_rev_);
578
579 state_ = MotionState::MOVING_TO_MIN;
580 driver_->rampControl.SetMaxSpeed(vmax_rpm, tmc51x0::Unit::RPM);
581 driver_->rampControl.SetAcceleration(amax_rev_s2, tmc51x0::Unit::RevPerSec);
582 driver_->rampControl.SetDeceleration(amax_rev_s2, tmc51x0::Unit::RevPerSec);
583 driver_->rampControl.SetTargetPosition(min_bound, tmc51x0::Unit::Deg);
584 } else {
585 state_ = MotionState::MOVING_TO_MIN;
586 }
587 }
588 break;
589
590 case MotionState::MOVING_TO_MIN:
591 case MotionState::MOVING_TO_MAX:
592 case MotionState::STOPPED:
593 // These are handled by UpdateSinuousMotion or ramp control
