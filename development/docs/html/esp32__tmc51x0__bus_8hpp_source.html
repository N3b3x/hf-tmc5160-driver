340 bool ApplyPinConfig(const tmc51x0::TMC51x0PinConfig& pin_config) noexcept {
341 constexpr gpio_num_t UNMAPPED_PIN = static_cast<gpio_num_t>(-1);
342
343 // Basic control pins
344 if (pin_config.en_pin != -1) {
345 SetPinMapping(tmc51x0::TMC51x0CtrlPin::EN, static_cast<gpio_num_t>(pin_config.en_pin));
346 }
347
348 // DIR/REFR_DIR compound pin (pin 18)
349 gpio_num_t dir_gpio = UNMAPPED_PIN;
350 if (pin_config.dir_pin != -1) {
351 dir_gpio = static_cast<gpio_num_t>(pin_config.dir_pin);
352 } else if (pin_config.ref_right_pin != -1) {
353 dir_gpio = static_cast<gpio_num_t>(pin_config.ref_right_pin);
354 }
355 if (dir_gpio != UNMAPPED_PIN) {
356 SetPinMapping(tmc51x0::TMC51x0CtrlPin::DIR, dir_gpio);
357 SetPinMapping(tmc51x0::TMC51x0CtrlPin::REFR_DIR, dir_gpio); // Same physical pin
358 }
359
360 // STEP/REFL_STEP compound pin (pin 17)
361 gpio_num_t step_gpio = UNMAPPED_PIN;
362 if (pin_config.step_pin != -1) {
363 step_gpio = static_cast<gpio_num_t>(pin_config.step_pin);
364 } else if (pin_config.ref_left_pin != -1) {
365 step_gpio = static_cast<gpio_num_t>(pin_config.ref_left_pin);
366 }
367 if (step_gpio != UNMAPPED_PIN) {
368 SetPinMapping(tmc51x0::TMC51x0CtrlPin::STEP, step_gpio);
369 SetPinMapping(tmc51x0::TMC51x0CtrlPin::REFL_STEP, step_gpio); // Same physical pin
370 }
371
372 // Diagnostic pins
373 if (pin_config.diag0_pin != -1) {
374 SetPinMapping(tmc51x0::TMC51x0CtrlPin::DIAG0, static_cast<gpio_num_t>(pin_config.diag0_pin));
375 }
376 if (pin_config.diag1_pin != -1) {
377 SetPinMapping(tmc51x0::TMC51x0CtrlPin::DIAG1, static_cast<gpio_num_t>(pin_config.diag1_pin));
378 }
379
380 // ENCA/DCIN compound pin (pin 24)
381 gpio_num_t enca_gpio = UNMAPPED_PIN;
382 if (pin_config.enc_a_pin != -1) {
383 enca_gpio = static_cast<gpio_num_t>(pin_config.enc_a_pin);
384 } else if (pin_config.dc_in_pin != -1) {
385 enca_gpio = static_cast<gpio_num_t>(pin_config.dc_in_pin);
386 }
387 if (enca_gpio != UNMAPPED_PIN) {
388 SetPinMapping(tmc51x0::TMC51x0CtrlPin::ENCA, enca_gpio);
389 SetPinMapping(tmc51x0::TMC51x0CtrlPin::DCIN, enca_gpio); // Same physical pin
390 }
391
392 // ENCB/DCEN compound pin (pin 23)
393 gpio_num_t encb_gpio = UNMAPPED_PIN;
394 if (pin_config.enc_b_pin != -1) {
395 encb_gpio = static_cast<gpio_num_t>(pin_config.enc_b_pin);
396 } else if (pin_config.dc_en_pin != -1) {
397 encb_gpio = static_cast<gpio_num_t>(pin_config.dc_en_pin);
398 }
399 if (encb_gpio != UNMAPPED_PIN) {
400 SetPinMapping(tmc51x0::TMC51x0CtrlPin::ENCB, encb_gpio);
401 SetPinMapping(tmc51x0::TMC51x0CtrlPin::DCEN, encb_gpio); // Same physical pin
402 }
403
404 // ENCN/DCO compound pin (pin 25)
405 gpio_num_t encn_gpio = UNMAPPED_PIN;
406 if (pin_config.enc_n_pin != -1) {
407 encn_gpio = static_cast<gpio_num_t>(pin_config.enc_n_pin);
408 } else if (pin_config.dc_out_pin != -1) {
409 encn_gpio = static_cast<gpio_num_t>(pin_config.dc_out_pin);
410 }
411 if (encn_gpio != UNMAPPED_PIN) {
412 SetPinMapping(tmc51x0::TMC51x0CtrlPin::ENCN, encn_gpio);
413 SetPinMapping(tmc51x0::TMC51x0CtrlPin::DCO, encn_gpio); // Same physical pin
414 }
415
416 // Clock pin
417 if (pin_config.clk_pin != -1) {
418 SetPinMapping(tmc51x0::TMC51x0CtrlPin::CLK, static_cast<gpio_num_t>(pin_config.clk_pin));
419 }
420
421 // Mode configuration pins (if available as control pins)
422 // ⚠️ WARNING: These are typically hardwired. Only configure if connected to GPIO.
423 if (pin_config.spi_mode_pin != -1) {
424 SetPinMapping(tmc51x0::TMC51x0CtrlPin::SPI_MODE, static_cast<gpio_num_t>(pin_config.spi_mode_pin));
425 }
426 if (pin_config.sd_mode_pin != -1) {
427 SetPinMapping(tmc51x0::TMC51x0CtrlPin::SD_MODE, static_cast<gpio_num_t>(pin_config.sd_mode_pin));
428 }
429

448 bool SetPinMapping(tmc51x0::TMC51x0CtrlPin pin, gpio_num_t gpio_pin) noexcept {
449 size_t index = static_cast<size_t>(pin);
450 if (index >= sizeof(pin_mapping_) / sizeof(pin_mapping_[0])) {
451 return false;
452 }
453 pin_mapping_[index] = gpio_pin;
454
455 constexpr gpio_num_t UNMAPPED_PIN = static_cast<gpio_num_t>(-1);
456 // Configure GPIO direction based on pin type
457 if (gpio_pin != UNMAPPED_PIN) {
458 // Diagnostic pins are outputs (read-only from MCU perspective)
459 if (pin == tmc51x0::TMC51x0CtrlPin::DIAG0 || pin == tmc51x0::TMC51x0CtrlPin::DIAG1) {
460 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT); // MCU reads TMC51x0 output
461 gpio_set_pull_mode(gpio_pin, GPIO_PULLUP_ONLY); // Diagnostic pins have pullups
462 }
463 // Encoder pins are inputs (when SD_MODE=0)
464 else if (pin == tmc51x0::TMC51x0CtrlPin::ENCA || pin == tmc51x0::TMC51x0CtrlPin::ENCB ||
465 pin == tmc51x0::TMC51x0CtrlPin::ENCN) {
466 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT);
467 gpio_set_pull_mode(gpio_pin, GPIO_PULLUP_ONLY);
468 }
469 // DC Step pins (when SD_MODE=1, SPI_MODE=1)
470 else if (pin == tmc51x0::TMC51x0CtrlPin::DCEN || pin == tmc51x0::TMC51x0CtrlPin::DCIN) {
471 gpio_set_direction(gpio_pin, GPIO_MODE_OUTPUT); // DCEN and DCIN are inputs to TMC51x0
472 } else if (pin == tmc51x0::TMC51x0CtrlPin::DCO) {
473 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT); // DCO is output from TMC51x0
474 gpio_set_pull_mode(gpio_pin, GPIO_PULLUP_ONLY);
475 }
476 // Reference switch pins can be inputs (when used as reference switches, SD_MODE=0)
477 else if (pin == tmc51x0::TMC51x0CtrlPin::REFL_STEP || pin == tmc51x0::TMC51x0CtrlPin::REFR_DIR) {
478 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT);
479 gpio_set_pull_mode(gpio_pin, GPIO_PULLUP_ONLY); // Typically active LOW
480 active_levels_.SetActiveLevel(pin, false); // Active LOW for reference switches
481 }
482 // CLK pin can be output (for external clock) or input (for reading clock state)
483 else if (pin == tmc51x0::TMC51x0CtrlPin::CLK) {
484 // Default to input, user can configure as output if needed for PWM clock
485 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT);
486 }
487 // Mode configuration pins (SPI_MODE, SD_MODE) - INPUT by default (read-only)
488 // These pins are typically hardwired on dev boards. Only set as OUTPUT if user
489 // explicitly calls communication.SetOperatingMode() or GpioSet() to control them.
490 else if (pin == tmc51x0::TMC51x0CtrlPin::SPI_MODE || pin == tmc51x0::TMC51x0CtrlPin::SD_MODE) {
491 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT); // Input to read chip mode (hardwired on dev boards)
492 gpio_set_pull_mode(gpio_pin, GPIO_PULLUP_ONLY); // Pullup for stable reading
493 }
494 // Other pins default to output
495 else {
496 gpio_set_direction(gpio_pin, GPIO_MODE_OUTPUT);
497 }
498 }
499 return true;
500 }

520 tmc51x0::Result<void> Initialize() noexcept {
521 if (initialized_) {
522 return tmc51x0::Result<void>();
523 }
524
525 // Configure GPIO pins that are mapped
526 constexpr gpio_num_t UNMAPPED_PIN = static_cast<gpio_num_t>(-1);
527 if (en_pin_ != UNMAPPED_PIN) {
528 gpio_set_direction(en_pin_, GPIO_MODE_OUTPUT);
529 // Disable by default using signal abstraction (EN is active LOW, so INACTIVE = HIGH = disabled)
530 auto en_result = GpioSet(tmc51x0::TMC51x0CtrlPin::EN, tmc51x0::GpioSignal::INACTIVE);
531 if (!en_result.IsOk()) {
532 ESP_LOGW(BUS_TAG, "Failed to set EN pin during initialization (non-critical)");
533 // Continue anyway - this is not critical for initialization
534 }
535 }
536 if (dir_pin_ != UNMAPPED_PIN) {
537 gpio_set_direction(dir_pin_, GPIO_MODE_OUTPUT);
538 // Set DIR to inactive by default using signal abstraction
539 auto dir_result = GpioSet(tmc51x0::TMC51x0CtrlPin::DIR, tmc51x0::GpioSignal::INACTIVE);
540 if (!dir_result.IsOk()) {
541 ESP_LOGW(BUS_TAG, "Failed to set DIR pin during initialization (non-critical)");
542 // Continue anyway - this is not critical for initialization
543 }
544 }
545 if (step_pin_ != UNMAPPED_PIN) {
546 gpio_set_direction(step_pin_, GPIO_MODE_OUTPUT);
547 // Set STEP to inactive by default using signal abstraction
548 auto step_result = GpioSet(tmc51x0::TMC51x0CtrlPin::STEP, tmc51x0::GpioSignal::INACTIVE);
549 if (!step_result.IsOk()) {
550 ESP_LOGW(BUS_TAG, "Failed to set STEP pin during initialization (non-critical)");
551 // Continue anyway - this is not critical for initialization
552 }
553 }

555 // Configure SPI bus
556 spi_bus_config_t bus_config = {};
557 bus_config.mosi_io_num = mosi_pin_;
558 bus_config.miso_io_num = miso_pin_;
559 bus_config.sclk_io_num = sclk_pin_;
560 bus_config.quadwp_io_num = -1;
561 bus_config.quadhd_io_num = -1;
562 bus_config.max_transfer_sz = 64; // Support up to 64 bytes for daisy-chain auto-detection (8 devices * 8 bytes)

564
565 esp_err_t ret = spi_bus_initialize(host_, &bus_config, SPI_DMA_CH_AUTO);
566 if (ret != ESP_OK) {

568 return tmc51x0::Result<void>(tmc51x0::ErrorCode::COMM_ERROR);
569 }
570
571 // Configure SPI device (Mode 3: CPOL=1, CPHA=1)
572 spi_device_interface_config_t dev_config = {};
573 dev_config.clock_speed_hz = clock_speed_hz_;
574 dev_config.mode = 3; // SPI Mode 3 for TMC51x0
575 dev_config.spics_io_num = cs_pin_;
576 dev_config.queue_size = 1;
577 dev_config.cs_ena_pretrans = 2;
578 dev_config.cs_ena_posttrans = 2;
579
580 ret = spi_bus_add_device(host_, &dev_config, &device_handle_);
581 if (ret != ESP_OK) {
582 ESP_LOGE(BUS_TAG, "Failed to add SPI device: %s", esp_err_to_name(ret));
583 spi_bus_free(host_);

585 }
586
587 initialized_ = true;
588 ESP_LOGI(BUS_TAG, "SPI interface initialized successfully");
589 return tmc51x0::Result<void>();

684 tmc51x0::Result<void> GpioSet(tmc51x0::TMC51x0CtrlPin pin, tmc51x0::GpioSignal signal) noexcept {
685 // Diagnostic pins and DCO are read-only (outputs from TMC51x0)
686 if (pin == tmc51x0::TMC51x0CtrlPin::DIAG0 || pin == tmc51x0::TMC51x0CtrlPin::DIAG1 ||
687 pin == tmc51x0::TMC51x0CtrlPin::DCO) {
688 ESP_LOGW(BUS_TAG, "Pin is read-only (output from TMC51x0)");
689 return tmc51x0::Result<void>(tmc51x0::ErrorCode::INVALID_VALUE);
690 }
691 // Encoder pins are read-only when used as encoder inputs (SD_MODE=0)
692 if (pin == tmc51x0::TMC51x0CtrlPin::ENCA || pin == tmc51x0::TMC51x0CtrlPin::ENCB ||
693 pin == tmc51x0::TMC51x0CtrlPin::ENCN) {
694 ESP_LOGW(BUS_TAG, "Encoder pins are read-only (use DCEN/DCIN for DC Step mode)");
695 return tmc51x0::Result<void>(tmc51x0::ErrorCode::INVALID_VALUE);
696 }
697
698 gpio_num_t gpio_pin = GetPinMapping(pin);
699 constexpr gpio_num_t UNMAPPED_PIN = static_cast<gpio_num_t>(-1);
700 if (gpio_pin == UNMAPPED_PIN) {
701 ESP_LOGW(BUS_TAG, "Pin not mapped: %d", static_cast<int>(pin));
702 return tmc51x0::Result<void>(tmc51x0::ErrorCode::INVALID_VALUE);
703 }
704
705 // If user is trying to set SPI_MODE or SD_MODE pins, configure them as OUTPUT
706 // (they default to INPUT for read-only mode on hardwired dev boards)
707 if (pin == tmc51x0::TMC51x0CtrlPin::SPI_MODE || pin == tmc51x0::TMC51x0CtrlPin::SD_MODE) {
708 gpio_set_direction(gpio_pin, GPIO_MODE_OUTPUT);
709 }
710
711 bool level = SignalToGpioLevel(pin, signal);
712 gpio_set_level(gpio_pin, level ? 1 : 0);
713 return tmc51x0::Result<void>();
714 }

819 tmc51x0::Result<void> SetClkFreq(uint32_t frequency_hz) noexcept {
820 gpio_num_t clk_pin = GetPinMapping(tmc51x0::TMC51x0CtrlPin::CLK);
821 constexpr gpio_num_t UNMAPPED_PIN = static_cast<gpio_num_t>(-1);
822
823 if (clk_pin == UNMAPPED_PIN) {
824 ESP_LOGW(BUS_TAG, "CLK pin not mapped, cannot configure clock");
825 return tmc51x0::Result<void>(tmc51x0::ErrorCode::INVALID_VALUE);
826 }
827
828 if (frequency_hz == 0) {
829 // Internal clock mode: Set CLK pin to GND (LOW)
830 gpio_set_direction(clk_pin, GPIO_MODE_OUTPUT);
831 gpio_set_level(clk_pin, 0); // Drive LOW (GND) for internal oscillator
832 ESP_LOGI(BUS_TAG, "CLK pin set to GND (internal 12 MHz oscillator enabled)");
833 return tmc51x0::Result<void>();
834 } else {
835 // External clock mode: Not implemented via PWM
836 // User must provide external clock signal on CLK pin
837 ESP_LOGW(BUS_TAG, "External clock generation not implemented. Provide external clock signal on CLK pin (frequency: %u Hz)", frequency_hz);
838 return tmc51x0::Result<void>(tmc51x0::ErrorCode::UNSUPPORTED);
839 }
