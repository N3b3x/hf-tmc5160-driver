214 // Basic control pins

229 }
230
231 // STEP/REFL_STEP compound pin (pin 17)
232 gpio_num_t step_gpio = UNMAPPED_PIN;
233 if (pin_config.step_pin != -1) {
234 step_gpio = static_cast<gpio_num_t>(pin_config.step_pin);
235 } else if (pin_config.ref_left_pin != -1) {
236 step_gpio = static_cast<gpio_num_t>(pin_config.ref_left_pin);
237 }
238 if (step_gpio != UNMAPPED_PIN) {
239 SetPinMapping(tmc5160::TMC5160CtrlPin::STEP, step_gpio);
240 SetPinMapping(tmc5160::TMC5160CtrlPin::REFL_STEP, step_gpio); // Same physical pin
241 }
242
243 // Diagnostic pins

245 SetPinMapping(tmc5160::TMC5160CtrlPin::DIAG0, static_cast<gpio_num_t>(pin_config.diag0_pin));

319 bool SetPinMapping(tmc5160::TMC5160CtrlPin pin, gpio_num_t gpio_pin) noexcept {
320 size_t index = static_cast<size_t>(pin);
321 if (index >= sizeof(pin_mapping_) / sizeof(pin_mapping_[0])) {
322 return false;
323 }
324 pin_mapping_[index] = gpio_pin;
325
326 constexpr gpio_num_t UNMAPPED_PIN = static_cast<gpio_num_t>(-1);
327 // Configure GPIO direction based on pin type
328 if (gpio_pin != UNMAPPED_PIN) {
329 // Diagnostic pins are outputs (read-only from MCU perspective)
330 if (pin == tmc5160::TMC5160CtrlPin::DIAG0 || pin == tmc5160::TMC5160CtrlPin::DIAG1) {
331 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT); // MCU reads TMC5160 output
332 gpio_set_pull_mode(gpio_pin, GPIO_PULLUP_ONLY); // Diagnostic pins have pullups
333 }
334 // Encoder pins are inputs (when SD_MODE=0)
335 else if (pin == tmc5160::TMC5160CtrlPin::ENCA || pin == tmc5160::TMC5160CtrlPin::ENCB ||
336 pin == tmc5160::TMC5160CtrlPin::ENCN) {
337 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT);
338 gpio_set_pull_mode(gpio_pin, GPIO_PULLUP_ONLY);
339 }
340 // DC Step pins (when SD_MODE=1, SPI_MODE=1)
341 else if (pin == tmc5160::TMC5160CtrlPin::DCEN || pin == tmc5160::TMC5160CtrlPin::DCIN) {
342 gpio_set_direction(gpio_pin, GPIO_MODE_OUTPUT); // DCEN and DCIN are inputs to TMC5160
343 } else if (pin == tmc5160::TMC5160CtrlPin::DCO) {
344 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT); // DCO is output from TMC5160
345 gpio_set_pull_mode(gpio_pin, GPIO_PULLUP_ONLY);
346 }
347 // Reference switch pins can be inputs (when used as reference switches, SD_MODE=0)
348 else if (pin == tmc5160::TMC5160CtrlPin::REFL_STEP || pin == tmc5160::TMC5160CtrlPin::REFR_DIR) {
349 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT);
350 gpio_set_pull_mode(gpio_pin, GPIO_PULLUP_ONLY); // Typically active LOW
351 SetPinActiveLevel(pin, false); // Active LOW for reference switches
352 }
353 // CLK pin can be output (for external clock) or input (for reading clock state)
354 else if (pin == tmc5160::TMC5160CtrlPin::CLK) {
355 // Default to input, user can configure as output if needed for PWM clock
356 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT);
357 }
358 // Mode configuration pins (SPI_MODE, SD_MODE) - outputs to control chip mode
359 else if (pin == tmc5160::TMC5160CtrlPin::SPI_MODE || pin == tmc5160::TMC5160CtrlPin::SD_MODE) {
360 gpio_set_direction(gpio_pin, GPIO_MODE_OUTPUT); // Output to control chip mode
361 // Set default state based on pin (SPI_MODE typically HIGH, SD_MODE typically LOW for SPI+Internal Ramp)
362 if (pin == tmc5160::TMC5160CtrlPin::SPI_MODE) {
363 gpio_set_level(gpio_pin, 1); // Default: HIGH (SPI mode)
364 } else {
365 gpio_set_level(gpio_pin, 0); // Default: LOW (Internal ramp mode)
366 }
367 }
368 // Other pins default to output
369 else {
370 gpio_set_direction(gpio_pin, GPIO_MODE_OUTPUT);
371 }
372 }
373 return true;
374 }

394 bool Initialize() noexcept {
395 if (initialized_) {
396 return true;
397 }
398
399 // Configure GPIO pins that are mapped
400 constexpr gpio_num_t UNMAPPED_PIN = static_cast<gpio_num_t>(-1);
401 if (en_pin_ != UNMAPPED_PIN) {
402 gpio_set_direction(en_pin_, GPIO_MODE_OUTPUT);
403 gpio_set_level(en_pin_, 1); // Disable by default (EN is active HIGH to disable)
404 }
405 if (dir_pin_ != UNMAPPED_PIN) {
406 gpio_set_direction(dir_pin_, GPIO_MODE_OUTPUT);
407 gpio_set_level(dir_pin_, 0);
408 }
409 if (step_pin_ != UNMAPPED_PIN) {
410 gpio_set_direction(step_pin_, GPIO_MODE_OUTPUT);
411 gpio_set_level(step_pin_, 0);
412 }
413
414 // Configure SPI bus
415 spi_bus_config_t bus_config = {};
416 bus_config.mosi_io_num = mosi_pin_;
417 bus_config.miso_io_num = miso_pin_;
418 bus_config.sclk_io_num = sclk_pin_;
419 bus_config.quadwp_io_num = -1;
420 bus_config.quadhd_io_num = -1;
421 bus_config.max_transfer_sz = 64; // Support up to 64 bytes for daisy-chain auto-detection (8 devices * 8 bytes)
422 bus_config.flags = SPICOMMON_BUSFLAG_MASTER;
423
424 esp_err_t ret = spi_bus_initialize(host_, &bus_config, SPI_DMA_CH_AUTO);
425 if (ret != ESP_OK) {
426 ESP_LOGE(BUS_TAG, "Failed to initialize SPI bus: %s", esp_err_to_name(ret));
427 return false;
428 }
429
430 // Configure SPI device (Mode 3: CPOL=1, CPHA=1)
431 spi_device_interface_config_t dev_config = {};
432 dev_config.clock_speed_hz = clock_speed_hz_;
433 dev_config.mode = 3; // SPI Mode 3 for TMC5160
434 dev_config.spics_io_num = cs_pin_;
435 dev_config.queue_size = 1;
436 dev_config.cs_ena_pretrans = 2;
437 dev_config.cs_ena_posttrans = 2;
438
439 ret = spi_bus_add_device(host_, &dev_config, &device_handle_);
440 if (ret != ESP_OK) {
441 ESP_LOGE(BUS_TAG, "Failed to add SPI device: %s", esp_err_to_name(ret));
442 spi_bus_free(host_);
443 return false;
444 }
445
446 initialized_ = true;
447 ESP_LOGI(BUS_TAG, "SPI interface initialized successfully");
448 return true;
449 }

518 bool GpioSet(tmc5160::TMC5160CtrlPin pin, tmc5160::GpioSignal signal) noexcept {
519 // Diagnostic pins and DCO are read-only (outputs from TMC5160)
520 if (pin == tmc5160::TMC5160CtrlPin::DIAG0 || pin == tmc5160::TMC5160CtrlPin::DIAG1 ||
521 pin == tmc5160::TMC5160CtrlPin::DCO) {
522 ESP_LOGW(BUS_TAG, "Pin is read-only (output from TMC5160)");
523 return false;
524 }
525 // Encoder pins are read-only when used as encoder inputs (SD_MODE=0)
526 if (pin == tmc5160::TMC5160CtrlPin::ENCA || pin == tmc5160::TMC5160CtrlPin::ENCB ||
527 pin == tmc5160::TMC5160CtrlPin::ENCN) {
528 ESP_LOGW(BUS_TAG, "Encoder pins are read-only (use DCEN/DCIN for DC Step mode)");
529 return false;
530 }
531
532 gpio_num_t gpio_pin = GetPinMapping(pin);
533 constexpr gpio_num_t UNMAPPED_PIN = static_cast<gpio_num_t>(-1);
534 if (gpio_pin == UNMAPPED_PIN) {
535 ESP_LOGW(BUS_TAG, "Pin not mapped: %d", static_cast<int>(pin));
536 return false;
537 }
538
539 bool level = SignalToGpioLevel(pin, signal);
540 gpio_set_level(gpio_pin, level ? 1 : 0);
541 return true;
542 }

555 bool GpioRead(tmc5160::TMC5160CtrlPin pin, tmc5160::GpioSignal& signal) noexcept {
556 gpio_num_t gpio_pin = GetPinMapping(pin);

563 signal = GpioLevelToSignal(pin, level != 0);
564 return true;
