166 // Driver Configuration
167 // NOTE: Board has 0.05 Ohm Sense Resistors (1W, low-inductance type required).
168 // Per datasheet table: RSENSE=0.05Ω → Max RMS=4.7A, Max Peak=6.6A (at CS=31, GLOBAL_SCALER=256)
169 //
170 // Current Calculation (datasheet formula):
171 // I_RMS = (GLOBAL_SCALER/256) * ((IRUN+1)/32) * (VFS/RSENSE) * (1/√2)
172 // Where VFS = 0.325V (typical full-scale voltage)
173 //
174 // With GLOBAL_SCALER=160, RSENSE=0.05Ω:
175 // Max Peak Current = 6.5A * (160/256) = 4.06A Peak (at IRUN=31)
176 //
177 // For 17HS4401S motor (1.68A RMS rated):
178 // IRUN=20: I_RMS = 0.625 * 0.65625 * 6.5 * 0.707 = ~1.88A RMS
179 // IRUN=20: I_Peak = 4.06A * 0.65625 = ~2.66A Peak
180 //
181 // Increased to IRUN=20 for better StealthChop calibration:
182 // - IRUN ≥ 8 is minimum for StealthChop automatic tuning
183 // - IRUN 16-31 recommended for best microstep performance
184 // - Current is slightly above motor rating but acceptable for testing
185 constexpr uint8_t GLOBAL_SCALER = 160; // Fine-tunes current range (0=256=full scale)
186 constexpr uint8_t IRUN = 20; // ~1.88A RMS, ~2.66A Peak (improved StealthChop)
187 constexpr uint8_t IHOLD = 10; // ~0.94A RMS, ~1.33A Peak (50% of Run)
188
189 // Microstepping for Maximum Smoothness
190 constexpr uint8_t MRES = 0; // 0 = 256 microsteps (Highest Resolution)
191 constexpr bool INTERPOLATION = true; // Interpolation (always on for smoothness)
192
193 // Chopper Configuration (SpreadCycle default for NEMA 17)
194 constexpr uint8_t TOFF = 5;
195 constexpr uint8_t HEND = 3;
196 constexpr uint8_t HSTRT = 4;
197 constexpr uint8_t TBL = 2; // Blank time 36 clocks
198
199 // StealthChop Configuration
200 constexpr bool STEALTH_AUTOSCALE = true;
201 constexpr bool STEALTH_AUTOGRAD = true;
202 constexpr uint8_t STEALTH_FREQ = 1; // 1 = ~35kHz @ 12MHz clock (Good balance)
203 constexpr uint8_t STEALTH_OFS = 30;
204
205 // Power Stage Configuration (BSC072N08NS5)
