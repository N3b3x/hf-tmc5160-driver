248 // Driver Configuration
249 // NOTE: Board has 0.05 Ohm Sense Resistors (1W, low-inductance type required).
250 // Per datasheet table: RSENSE=0.05Ω → Max RMS=4.7A, Max Peak=6.6A (at CS=31, GLOBAL_SCALER=256)
251 //
252 // Current Calculation (datasheet formula):
253 // I_RMS = (GLOBAL_SCALER/256) * ((IRUN+1)/32) * (VFS/RSENSE) * (1/√2)
254 // Where VFS = 0.325V (typical full-scale voltage)
255 //
256 // With GLOBAL_SCALER=160, RSENSE=0.05Ω:
257 // Max Peak Current = 6.5A * (160/256) = 4.06A Peak (at IRUN=31)
258 //
259 // For 17HS4401S motor (1.68A RMS rated):
260 // IRUN=20: I_RMS = 0.625 * 0.65625 * 6.5 * 0.707 = ~1.88A RMS
261 // IRUN=20: I_Peak = 4.06A * 0.65625 = ~2.66A Peak
262 //
263 // Increased to IRUN=20 for better StealthChop calibration:
264 // - IRUN ≥ 8 is minimum for StealthChop automatic tuning
265 // - IRUN 16-31 recommended for best microstep performance
266 // - Current is slightly above motor rating but acceptable for testing
267 constexpr uint16_t GLOBAL_SCALER = 160; // Fine-tunes current range (0=256=full scale)
268 constexpr uint8_t IRUN = 20; // ~1.88A RMS, ~2.66A Peak (improved StealthChop)
269 constexpr uint8_t IHOLD = 10; // ~0.94A RMS, ~1.33A Peak (50% of Run)
