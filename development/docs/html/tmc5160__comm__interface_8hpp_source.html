716static constexpr uint8_t calculateCrc8(const uint8_t* data, size_t length) noexcept {
717 uint8_t crc = 0; // Initial value is zero per datasheet
718
719 // Process each byte LSB to MSB
720 for (size_t i = 0; i < length; ++i) {
721 uint8_t currentByte = data[i];
722
723 // Process each bit LSB to MSB (j=0 is LSB, j=7 is MSB)
724 for (uint8_t j = 0; j < 8; ++j) {
725 // Check: (CRC >> 7) XOR (currentByte & 0x01)
726 // This XORs the MSB of CRC with the LSB of currentByte
727 if (((crc >> 7) ^ (currentByte & 0x01)) != 0) {
728 crc = (crc << 1) ^ 0x07; // Polynomial 0x07 (CRC8-ATM)
729 } else {
730 crc = (crc << 1);
731 }
732 currentByte = currentByte >> 1; // Shift to next bit (LSB to MSB)
733 }
734 }
735
736 return crc;
737}

928 uart_frame.type = UartFrameType::WriteAccess;
929
930 // Byte 0: Sync nibble (bits 0-3: 1,0,1,0 = 0x0A) + Reserved (bits 4-7) + NODEADDR
931 // Note: Current implementation uses simplified sync (bit 0 = 1) for compatibility
932 // Full sync pattern would be: (node_addr & 0x7F) | 0x0A
933 // Simplified: (node_addr & 0x7F) | 0x01
934 uart_frame.frame.write_fields.sync_nodeaddr = (node_addr & 0x7F) | 0x01;
935
936 // Byte 1: RW bit (bit 7 = 1 for write) + 7-bit register address
937 uart_frame.frame.write_fields.rw_address = (reg_addr & 0x7F) | 0x80;
938
939 // Bytes 2-5: 32-bit data (high byte to low byte, MSB-first)
940 uart_frame.frame.write_fields.data_bytes[0] = static_cast<uint8_t>((value >> 24) & 0xFF);
941 uart_frame.frame.write_fields.data_bytes[1] = static_cast<uint8_t>((value >> 16) & 0xFF);
942 uart_frame.frame.write_fields.data_bytes[2] = static_cast<uint8_t>((value >> 8) & 0xFF);
943 uart_frame.frame.write_fields.data_bytes[3] = static_cast<uint8_t>(value & 0xFF);
944
945 // Bytes 6-7: Reserved (don't cares but included in CRC)
946 uart_frame.frame.write_fields.reserved[0] = 0x00;
947 uart_frame.frame.write_fields.reserved[1] = 0x00;

1488 // Use a read command to a register that's safe to read (GSTAT = 0x00)
1489 // But we'll use a unique address pattern to make it more distinctive
1490 // Actually, let's use a write command with a unique value that we can recognize

1492

1494 // We'll use address 0x73 (last register) which is safe to read
1495 // The command pattern will be: [0x73] [0x00] [0x00] [0x00] [0x00]

1499 // Let's use 0x73 (last register address) - this is distinctive

1504 cmd.GetFrame(cmd_bytes);

1507 // Total transfer: (max_devices+2)*40 bits to ensure we capture loopback
1508 size_t transfer_bytes = static_cast<size_t>(max_devices + 2) * 5;

1530 // The command loops back after n*40 bits, appearing at offset n*5 bytes
