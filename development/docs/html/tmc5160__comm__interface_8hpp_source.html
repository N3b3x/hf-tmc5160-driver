396static constexpr uint8_t calculateCrc8(const uint8_t *data,
397 size_t length) noexcept {
398 uint8_t crc = 0; // Initial value is zero per datasheet
399
400 // Process each byte LSB to MSB
401 for (size_t i = 0; i < length; ++i) {
402 uint8_t currentByte = data[i];
403
404 // Process each bit LSB to MSB (j=0 is LSB, j=7 is MSB)
405 for (uint8_t j = 0; j < 8; ++j) {
406 // Check: (CRC >> 7) XOR (currentByte & 0x01)
407 // This XORs the MSB of CRC with the LSB of currentByte
408 if ((crc >> 7) ^ (currentByte & 0x01)) {
409 crc = (crc << 1) ^ 0x07; // Polynomial 0x07 (CRC8-ATM)
410 } else {
411 crc = (crc << 1);
412 }
413 currentByte = currentByte >> 1; // Shift to next bit (LSB to MSB)
414 }
415 }
416
417 return crc;
418}

604 UartFrame uart_frame{};
605 uart_frame.type = UartFrameType::WriteAccess;
606
607 // Byte 0: Sync nibble (bits 0-3: 1,0,1,0 = 0x0A) + Reserved (bits 4-7) + NODEADDR
608 // Note: Current implementation uses simplified sync (bit 0 = 1) for compatibility
609 // Full sync pattern would be: (node_addr & 0x7F) | 0x0A
610 // Simplified: (node_addr & 0x7F) | 0x01
611 uart_frame.frame.write_fields.sync_nodeaddr = (node_addr & 0x7F) | 0x01;
612
613 // Byte 1: RW bit (bit 7 = 1 for write) + 7-bit register address
614 uart_frame.frame.write_fields.rw_address = (reg_addr & 0x7F) | 0x80;
615
616 // Bytes 2-5: 32-bit data (high byte to low byte, MSB-first)
617 uart_frame.frame.write_fields.data_bytes[0] = static_cast<uint8_t>((value >> 24) & 0xFF);
618 uart_frame.frame.write_fields.data_bytes[1] = static_cast<uint8_t>((value >> 16) & 0xFF);
619 uart_frame.frame.write_fields.data_bytes[2] = static_cast<uint8_t>((value >> 8) & 0xFF);
620 uart_frame.frame.write_fields.data_bytes[3] = static_cast<uint8_t>(value & 0xFF);
621
622 // Bytes 6-7: Reserved (don't cares but included in CRC)
623 uart_frame.frame.write_fields.reserved[0] = 0x00;
624 uart_frame.frame.write_fields.reserved[1] = 0x00;
625
626 // Byte 8: CRC8 checksum (calculated automatically)
627 uart_frame.CalculateCrc();
