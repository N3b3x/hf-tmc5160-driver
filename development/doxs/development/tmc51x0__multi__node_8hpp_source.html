235 bool ProgramSequentially(uint8_t send_delay = 2) noexcept {
236 // Count active devices to determine starting address
237 uint8_t num_devices = 0;
238 for (size_t i = 0; i < MaxDevices; ++i) {
239 if (drivers_[i].has_value()) {
240 num_devices++;
241 }
242 }
243
244 if (num_devices == 0) {
245 return false; // No devices to program
246 }
247
248 // Program devices in forward order (index 0, 1, 2, ...)
249 // but assign addresses in reverse order (254, 253, 252, ...)
250 // First chip: accessible at address 0 (NAI=GND)
251 // After each programming, next chip becomes accessible at address 0
252
253 uint8_t device_index = 0;
254 for (size_t i = 0; i < MaxDevices; ++i) {
255 if (!drivers_[i].has_value()) {
256 continue; // Skip empty slots
257 }
258
259 // Calculate target address: start from 254 and count down
260 // Device at index 0 gets address 254, index 1 gets 253, etc.
261 uint8_t target_address = 254 - device_index;
262
263 // Program device to target address
264 // The device is currently accessible at address 0 (after previous chip's NAO went LOW)
265 // or address 0 for the first chip (NAI=GND)
266 if (!drivers_[i]->uartConfig.ConfigureUartNodeAddress(target_address, send_delay)) {
267 return false; // Failed to program device
268 }
269
270 // Update the driver's node address to the programmed address
271 drivers_[i]->SetUartNodeAddress(target_address);
272
273 // After programming, the chip's NAO automatically goes LOW (per datasheet)
274 // This makes the next chip accessible at address 0
275 // Small delay to ensure NAO settles
276 comm_.DelayMs(10);
277
278 device_index++;
279 }
280
281 return true;
282 }

294 bool ProgramDevice(uint8_t index, uint8_t send_delay = 2) noexcept {
295 if (index >= MaxDevices || !drivers_[index].has_value()) {
296 return false;

325 bool AddDevice(uint8_t index) noexcept {
326 // Validate index
327 if (index < num_onboard_devices_ || index >= MaxDevices) {
328 return false;
329 }
330
331 // Check if slot is already occupied
332 if (drivers_[index].has_value()) {
333 return false;
334 }
335
336 // Enforce sequential indexing: cannot skip indices
337 // All indices before this one (starting from num_onboard_devices) must be filled
338 for (uint8_t i = num_onboard_devices_; i < index; ++i) {
339 if (!drivers_[i].has_value()) {
340 return false; // Cannot skip indices
341 }
342 }
343
344 // Create device instance with initial address 0
345 // The actual address will be programmed via ProgramSequentially() or ProgramDevice()
346 drivers_[index] = std::make_optional<TMC51x0<CommType>>(comm_, f_clk_, 0, 0);
347 num_active_devices_++;
348
349 return true;
350 }

367 bool RemoveDevice(uint8_t index) noexcept {
368 // Cannot remove onboard devices
369 if (index < num_onboard_devices_) {
370 return false;
371 }
372
373 // Validate index

375 return false;
376 }
377
378 // Check if slot is actually active
379 if (!drivers_[index].has_value()) {
380 return false;
381 }
382
383 // Enforce sequential removal: can only remove from the end
384 // Check if there are any devices after this index
385 for (size_t i = index + 1; i < MaxDevices; ++i) {
386 if (drivers_[i].has_value()) {
387 return false; // Cannot remove device in the middle of the chain
388 }
389 }
390
391 // Remove device
392 drivers_[index] = std::nullopt;
