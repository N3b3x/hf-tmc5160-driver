227bool FatigueTestMotion::Start() noexcept {
228 uint32_t current_cycles, target_cycles;
229
230 {
231 TmcMutexGuard guard(driver_mutex_);
232 // CRITICAL: Verify bounds are set before allowing motion to start
233 if (fabsf(local_min_bound_) < 0.01f && fabsf(local_max_bound_) < 0.01f) {
234 ESP_LOGE(TAG_MOTION, "Cannot start: local bounds not set!");
235 return false;
236 }
237
238 if (cycle_complete_) {
239 ESP_LOGW(TAG_MOTION, "Cycle count reached. Reset cycles or set new target to continue.");
240 return false;
241 }
242 }
243
244 // CRITICAL: Verify motor is stopped before starting new motion
245 // This prevents parameter changes during active motion
246 // Check outside mutex to avoid holding lock during driver calls
247 auto standstill_result = driver_->rampControl.IsStandstill();
248 if (!standstill_result || !standstill_result.Value()) {
249 ESP_LOGW(TAG_MOTION, "Motor not at standstill, stopping before start...");
250 driver_->rampControl.Stop();
251 vTaskDelay(pdMS_TO_TICKS(200));
252 // Wait for standstill
253 uint32_t checks = 0;
254 while (checks < 20) {
255 checks++;
256 auto check_result = driver_->rampControl.IsStandstill();
257 if (check_result && check_result.Value()) {
258 break;
259 }
260 vTaskDelay(pdMS_TO_TICKS(100));
261 }
262 }
263
264 {
265 TmcMutexGuard guard(driver_mutex_);
266
267 // Update trajectory before starting
268 RecalculateTrajectory();
269
270 // Configure driver for positioning mode
271 // Values are already in proper units (RPM and rev/s²) - use directly
272 driver_->rampControl.SetRampMode(tmc51x0::RampMode::POSITIONING);
273 driver_->rampControl.SetMaxSpeed(calculated_vmax_rpm_, tmc51x0::Unit::RPM);
274 driver_->rampControl.SetAcceleration(calculated_amax_rev_s2_, tmc51x0::Unit::RevPerSec);
275 driver_->rampControl.SetDeceleration(calculated_amax_rev_s2_, tmc51x0::Unit::RevPerSec);
276 driver_->rampControl.SetRampSpeeds(30.0f, 3.0f, 0.0f, tmc51x0::Unit::RPM);
277
278 running_ = true;
279 start_time_us_ = esp_timer_get_time();
280 sinusoidal_mode_ = true; // Use sinusoidal mode
281
282 // Determine initial state based on current position
283 auto current_pos_result = driver_->rampControl.GetCurrentPosition(tmc51x0::Unit::Deg);
284 float current_pos_deg = current_pos_result.IsOk() ? current_pos_result.Value() : 0.0f;
285 float min_pos_deg = local_min_bound_;
286 float max_pos_deg = local_max_bound_;
287
288 // Find closest bound or determine direction
289 float dist_to_min = fabsf(current_pos_deg - min_pos_deg);
290 float dist_to_max = fabsf(current_pos_deg - max_pos_deg);
291
292 // Choose an initial direction that moves away from the nearest bound.
293 // - If we are already at/near one bound, start by moving toward the opposite bound.
294 // - Otherwise, move toward the nearer bound first.
295 constexpr float NEAR_BOUND_DEG = 1.0f;
296 if (dist_to_min < NEAR_BOUND_DEG) {
297 state_ = MotionState::MOVING_TO_MAX;
298 } else if (dist_to_max < NEAR_BOUND_DEG) {
299 state_ = MotionState::MOVING_TO_MIN;
300 } else {
301 state_ = (dist_to_min <= dist_to_max) ? MotionState::MOVING_TO_MIN : MotionState::MOVING_TO_MAX;
302 }
303
304 current_cycles = current_cycles_;
305 target_cycles = target_cycles_;
306 }
307
308 ESP_LOGI(TAG_MOTION, "Starting fatigue test (cycles: %lu/%lu)", current_cycles,
309 target_cycles == 0 ? 0xFFFFFFFF : target_cycles);
310 ESP_LOGI(TAG_MOTION, " Motion: Sinusoidal mode, VMAX=%.2f RPM, AMAX=%.3f rev/s²",
311 calculated_vmax_rpm_, calculated_amax_rev_s2_);
312 return true;
313}

333void FatigueTestMotion::RecalculateTrajectory() noexcept {
334 // Calculate total travel distance (one way) in degrees
335 float distance_deg = fabsf(local_max_bound_ - local_min_bound_);
336 if (distance_deg < 0.1f || frequency_hz_ <= 0.0001f) {
337 // Default values in proper units
338 calculated_vmax_rpm_ = 30.0f; // 30 RPM default
339 calculated_amax_rev_s2_ = 2.0f; // 2 rev/s² default
340 estimated_frequency_hz_ = 0.0f;
341 return;
342 }
343
344 // Target cycle period (total time for there + back + dwells)
345 float target_period_s = 1.0f / frequency_hz_;
346
347 // Total dwell time per cycle (min + max)
348 float total_dwell_s = (dwell_at_min_ms_ + dwell_at_max_ms_) / 1000.0f;
349
350 // Time available for motion (there + back)
351 float total_move_time_s = target_period_s - total_dwell_s;
352
353 if (total_move_time_s <= 0.0f) {
354 total_move_time_s = 0.1f;
355 ESP_LOGW(TAG_MOTION, "Requested frequency %.2f Hz is impossible with given dwell times!", frequency_hz_);
356 }
357
358 // Time for one leg (one way)
359 float leg_time_s = total_move_time_s / 2.0f;
360
361 // Calculate velocity in degrees per second
362 // Using 1.5x factor for trapezoidal profile (allows for acceleration/deceleration phases)
363 float velocity_deg_per_s = (1.5f * distance_deg) / leg_time_s;
364
365 // Convert to RPM: (deg/s / 360 deg/rev) * 60 s/min = RPM
366 calculated_vmax_rpm_ = (velocity_deg_per_s / 360.0f) * 60.0f;
367
368 // Calculate acceleration in rev/s²
369 // Acceleration = velocity_change / time_to_reach_velocity
370 // Time to reach velocity is approximately leg_time / 3 (for trapezoidal profile)
371 float time_to_reach_velocity = leg_time_s / 3.0f;
372 float velocity_rev_per_s = calculated_vmax_rpm_ / 60.0f;
373 calculated_amax_rev_s2_ = velocity_rev_per_s / time_to_reach_velocity;
374
375 // Clamp to reasonable limits (300 RPM max, 10 rev/s² max)
376 if (calculated_vmax_rpm_ > 300.0f) calculated_vmax_rpm_ = 300.0f;
377 if (calculated_amax_rev_s2_ > 10.0f) calculated_amax_rev_s2_ = 10.0f;
378 if (calculated_vmax_rpm_ < 1.0f) calculated_vmax_rpm_ = 1.0f; // Minimum 1 RPM
379 if (calculated_amax_rev_s2_ < 0.1f) calculated_amax_rev_s2_ = 0.1f; // Minimum 0.1 rev/s²
380
381 estimated_frequency_hz_ = 1.0f / (2.0f * leg_time_s + total_dwell_s);
382
383 ESP_LOGI(TAG_MOTION, "Trajectory Recalculated: Dist=%.2f°, LegTime=%.3fs", distance_deg, leg_time_s);
384 ESP_LOGI(TAG_MOTION, " Target Freq=%.2fHz, Est Freq=%.2fHz", frequency_hz_, estimated_frequency_hz_);
385 ESP_LOGI(TAG_MOTION, " VMAX=%.2f RPM, AMAX=%.3f rev/s²",
386 calculated_vmax_rpm_, calculated_amax_rev_s2_);
387}

389void FatigueTestMotion::ClipLocalBoundsToGlobal() noexcept {
390 if (!bounded_)
391 return;
392
393 float old_min = local_min_bound_;
394 float old_max = local_max_bound_;
395
396 // Clip local bounds to global bounds (all in degrees)
397 local_min_bound_ = std::max(local_min_bound_, global_min_bound_);
398 local_max_bound_ = std::min(local_max_bound_, global_max_bound_);
399
400 // Update home position
401 home_position_ = (local_min_bound_ + local_max_bound_) / 2.0f;
402 amplitude_ = (local_max_bound_ - local_min_bound_) / 2.0f;
403
404 if (fabsf(old_min - local_min_bound_) > 0.01f || fabsf(old_max - local_max_bound_) > 0.01f) {
405 ESP_LOGW(TAG_MOTION, "Local bounds clipped to global bounds");
406 ESP_LOGI(TAG_MOTION, "Clipped local bounds: min=%.2f°, max=%.2f°", local_min_bound_, local_max_bound_);
407 }

410void FatigueTestMotion::UpdateSinuousMotion() noexcept {
411 // CRITICAL SAFETY CHECK: Never run sinusoidal motion if not started properly
412 // This prevents fast oscillation when start_time_us_ is 0 (uninitialized)
413 {
414 TmcMutexGuard guard(driver_mutex_);
415 if (!running_ || start_time_us_ == 0) {
416 return; // Motion not started or not properly initialized
417 }
418 }
419
420 uint64_t elapsed_us;
421 float freq, amp;
422 float home, local_min, local_max;
423 uint32_t target_cycles;
424 bool cycle_started;
425 float last_target_rel;
426 uint32_t dwell_min, dwell_max;
427 float phase_off;
428
429 {
430 TmcMutexGuard guard(driver_mutex_);
431 elapsed_us = esp_timer_get_time() - start_time_us_;
432 freq = frequency_hz_;
433 amp = amplitude_;
434 home = home_position_;
435 local_min = local_min_bound_;
436 local_max = local_max_bound_;
437 target_cycles = target_cycles_;
438 cycle_started = cycle_started_;
439 last_target_rel = last_target_relative_;
440 dwell_min = dwell_at_min_ms_;
441 dwell_max = dwell_at_max_ms_;
442 phase_off = phase_offset_;
443 }
444
445 // Calculate sinusoidal position
446 double elapsed_s = elapsed_us / 1000000.0;
447 double angle = 2.0 * M_PI * freq * elapsed_s + phase_off;
448 double sin_value = sin(angle);
449
450 // Calculate target position in degrees
451 float target_deg = home + static_cast<float>(amp * sin_value);
452
453 // Get current position relative to center for cycle counting
454 float current_pos_deg = 0.0f;
455 {
456 TmcMutexGuard guard(driver_mutex_);
457 auto current_pos_result = driver_->rampControl.GetCurrentPosition(tmc51x0::Unit::Deg);
458 current_pos_deg = current_pos_result.IsOk() ? current_pos_result.Value() : 0.0f;
459 }
460 float target_relative = target_deg - home;
461
462 // Cycle counting: one cycle = center → extreme → center
463 bool currently_negative = (target_relative < 0.0f);
464 bool last_was_negative = (last_target_rel < 0.0f);
465 bool crossing_center =
466 (last_was_negative != currently_negative) && (fabsf(target_relative) < 1.0f) && (fabsf(last_target_rel) < 1.0f);
467
468 // If we've started a cycle (left center) and now crossing back through center
469 if (cycle_started && crossing_center) {
470 // Completed a cycle
471 uint32_t new_cycles;
472 {
473 TmcMutexGuard guard(driver_mutex_);
474 current_cycles_++;
475 cycle_started_ = false;
476 new_cycles = current_cycles_;
477
478 // Check if target reached
479 if (target_cycles > 0 && new_cycles >= target_cycles) {
480 cycle_complete_ = true;
481 running_ = false;
482 state_ = MotionState::STOPPED;
483 // Driver call inside mutex guard
484 driver_->rampControl.Stop();
485 guard.unlock();
486 ESP_LOGI(TAG_MOTION, "Target cycle count reached: %lu cycles. Stopping.", new_cycles);
487 return;
488 }
489 }
490 ESP_LOGI(TAG_MOTION, "Cycle %lu completed at center (target: %lu)", new_cycles,
491 target_cycles == 0 ? 0xFFFFFFFF : target_cycles);
492 } else if (!cycle_started && fabsf(target_relative) > 1.0f) {
493 // We've left center, cycle has started
494 TmcMutexGuard guard(driver_mutex_);
495 cycle_started_ = true;
496 last_was_negative_ = currently_negative;
497 }
498
499 // Update tracking
500 {
501 TmcMutexGuard guard(driver_mutex_);
502 last_target_relative_ = target_relative;
503 if (fabsf(target_relative) > 0.5f) {
504 last_was_negative_ = currently_negative;
505 }
506 }
507
508 // Clamp to local bounds and handle dwell states
509 // Use ABSOLUTE positioning - home is established via bounds finding or SetUnbounded()
510 // target_deg is calculated as home + amplitude * sin(), so it's already absolute
511 if (target_deg <= local_min) {
512 target_deg = local_min;
513 if (dwell_min > 0) {
514 TmcMutexGuard guard(driver_mutex_);
515 state_ = MotionState::DWELL_AT_MIN;
516 dwell_start_time_ms_ = esp_timer_get_time() / 1000;
517 driver_->rampControl.SetTargetPosition(target_deg, tmc51x0::Unit::Deg);
518 return;
519 } else {
520 TmcMutexGuard guard(driver_mutex_);
521 state_ = MotionState::MOVING_TO_MAX;
522 }
523 } else if (target_deg >= local_max) {
524 target_deg = local_max;
525 if (dwell_max > 0) {
526 TmcMutexGuard guard(driver_mutex_);
527 state_ = MotionState::DWELL_AT_MAX;
528 dwell_start_time_ms_ = esp_timer_get_time() / 1000;
529 driver_->rampControl.SetTargetPosition(target_deg, tmc51x0::Unit::Deg);
530 return;
531 } else {
532 TmcMutexGuard guard(driver_mutex_);
533 state_ = MotionState::MOVING_TO_MIN;
534 }
535 } else {
536 TmcMutexGuard guard(driver_mutex_);
537 if (target_relative > 0.0f) {
538 state_ = MotionState::MOVING_TO_MAX;
539 } else {
540 state_ = MotionState::MOVING_TO_MIN;
541 }
542 }
543
544 // Update target position if it changed significantly
545 // Use ABSOLUTE positioning - home is established, target_deg is absolute
546 if (fabsf(target_deg - current_pos_deg) > 0.5f) { // ~0.5 degree threshold
547 TmcMutexGuard guard(driver_mutex_);
548 driver_->rampControl.SetRampMode(tmc51x0::RampMode::POSITIONING);
549 // Use slow speed for position correction (5 RPM, 1 rev/s²)
550 constexpr float SLOW_VMAX_RPM = 5.0f;
551 constexpr float SLOW_AMAX_REV_S2 = 1.0f;
552
553 driver_->rampControl.SetTargetPosition(target_deg, tmc51x0::Unit::Deg);
554 driver_->rampControl.SetMaxSpeed(SLOW_VMAX_RPM, tmc51x0::Unit::RPM);
555 driver_->rampControl.SetAcceleration(SLOW_AMAX_REV_S2, tmc51x0::Unit::RevPerSec);
556 }
557}

559void FatigueTestMotion::Update() noexcept {
560 {
561 TmcMutexGuard guard(driver_mutex_);
562 if (!running_ || state_ == MotionState::STOPPED) {
563 return;
564 }
565
566 // Check if cycle count reached
567 if (target_cycles_ > 0 && current_cycles_ >= target_cycles_) {
568 if (!cycle_complete_) {
569 cycle_complete_ = true;
570 uint32_t cycles = current_cycles_;
571 state_ = MotionState::STOPPED;
572 running_ = false;
573 // Driver call inside mutex guard
574 driver_->rampControl.Stop();
575 guard.unlock();
576 ESP_LOGI(TAG_MOTION, "Target cycle count reached: %lu cycles. Stopping.", cycles);
577 }
578 return;
579 }
580 }
581
582 // Check if we're in sinusoidal mode
583 bool use_sinusoidal;
584 MotionState current_state;
585 {
586 TmcMutexGuard guard(driver_mutex_);
587 use_sinusoidal = sinusoidal_mode_;
588 current_state = state_;
589 }
590
591 // If sinusoidal mode and not in a dwell state, use sinusoidal update
592 if (use_sinusoidal && current_state != MotionState::DWELL_AT_MIN &&
593 current_state != MotionState::DWELL_AT_MAX) {
594 UpdateSinuousMotion();
595 return;
596 }
597
598 // Handle dwell states
599 uint32_t current_time_ms = esp_timer_get_time() / 1000;
600 uint32_t dwell_min, dwell_max, dwell_start;
601 float min_bound, max_bound;
602
603 {
604 TmcMutexGuard guard(driver_mutex_);
605 current_state = state_;
606 dwell_min = dwell_at_min_ms_;
607 dwell_max = dwell_at_max_ms_;
608 dwell_start = dwell_start_time_ms_;
609 min_bound = local_min_bound_; // Already in degrees
610 max_bound = local_max_bound_; // Already in degrees
611 }
612
613 switch (current_state) {
614 case MotionState::DWELL_AT_MIN:
615 if (current_time_ms - dwell_start >= dwell_min) {
616 TmcMutexGuard guard(driver_mutex_);
617 if (!sinusoidal_mode_) {
618 // Values are already in proper units (RPM and rev/s²) - use directly
619 state_ = MotionState::MOVING_TO_MAX;
620 driver_->rampControl.SetMaxSpeed(calculated_vmax_rpm_, tmc51x0::Unit::RPM);
621 driver_->rampControl.SetAcceleration(calculated_amax_rev_s2_, tmc51x0::Unit::RevPerSec);
622 driver_->rampControl.SetDeceleration(calculated_amax_rev_s2_, tmc51x0::Unit::RevPerSec);
623 driver_->rampControl.SetTargetPosition(max_bound, tmc51x0::Unit::Deg);
624 } else {
625 state_ = MotionState::MOVING_TO_MAX;
626 }
627 }
628 break;
629
630 case MotionState::DWELL_AT_MAX:
631 if (current_time_ms - dwell_start >= dwell_max) {
632 TmcMutexGuard guard(driver_mutex_);
633 if (!sinusoidal_mode_) {
634 // Values are already in proper units (RPM and rev/s²) - use directly
635 state_ = MotionState::MOVING_TO_MIN;
636 driver_->rampControl.SetMaxSpeed(calculated_vmax_rpm_, tmc51x0::Unit::RPM);
637 driver_->rampControl.SetAcceleration(calculated_amax_rev_s2_, tmc51x0::Unit::RevPerSec);
638 driver_->rampControl.SetDeceleration(calculated_amax_rev_s2_, tmc51x0::Unit::RevPerSec);
639 driver_->rampControl.SetTargetPosition(min_bound, tmc51x0::Unit::Deg);
640 } else {
641 state_ = MotionState::MOVING_TO_MIN;
642 }
643 }
644 break;
645
646 case MotionState::MOVING_TO_MIN:
647 case MotionState::MOVING_TO_MAX:
648 case MotionState::STOPPED:
649 // These are handled by UpdateSinuousMotion or ramp control
650 break;
651 }
652}
