396static constexpr uint8_t calculateCrc8(const uint8_t *data,
397 size_t length) noexcept {
398 uint8_t crc = 0; // Initial value is zero per datasheet
399
400 // Process each byte LSB to MSB
401 for (size_t i = 0; i < length; ++i) {
402 uint8_t currentByte = data[i];
403
404 // Process each bit LSB to MSB (j=0 is LSB, j=7 is MSB)
405 for (uint8_t j = 0; j < 8; ++j) {
406 // Check: (CRC >> 7) XOR (currentByte & 0x01)
407 // This XORs the MSB of CRC with the LSB of currentByte
408 if ((crc >> 7) ^ (currentByte & 0x01)) {
409 crc = (crc << 1) ^ 0x07; // Polynomial 0x07 (CRC8-ATM)
410 } else {
411 crc = (crc << 1);
412 }
413 currentByte = currentByte >> 1; // Shift to next bit (LSB to MSB)
414 }
415 }
416
417 return crc;
418}

603 static UartFrame Write(uint8_t node_addr, uint8_t reg_addr, uint32_t value) noexcept {
604 UartFrame uart_frame{};
605 uart_frame.type = UartFrameType::WriteAccess;
606
607 // Byte 0: Sync nibble (bits 0-3: 1,0,1,0 = 0x0A) + Reserved (bits 4-7) + NODEADDR
608 // Note: Current implementation uses simplified sync (bit 0 = 1) for compatibility
609 // Full sync pattern would be: (node_addr & 0x7F) | 0x0A
610 // Simplified: (node_addr & 0x7F) | 0x01
611 uart_frame.frame.write_fields.sync_nodeaddr = (node_addr & 0x7F) | 0x01;
612
613 // Byte 1: RW bit (bit 7 = 1 for write) + 7-bit register address
614 uart_frame.frame.write_fields.rw_address = (reg_addr & 0x7F) | 0x80;
615
616 // Bytes 2-5: 32-bit data (high byte to low byte, MSB-first)
617 uart_frame.frame.write_fields.data_bytes[0] = static_cast<uint8_t>((value >> 24) & 0xFF);
618 uart_frame.frame.write_fields.data_bytes[1] = static_cast<uint8_t>((value >> 16) & 0xFF);
619 uart_frame.frame.write_fields.data_bytes[2] = static_cast<uint8_t>((value >> 8) & 0xFF);
620 uart_frame.frame.write_fields.data_bytes[3] = static_cast<uint8_t>(value & 0xFF);
621
622 // Bytes 6-7: Reserved (don't cares but included in CRC)
623 uart_frame.frame.write_fields.reserved[0] = 0x00;
624 uart_frame.frame.write_fields.reserved[1] = 0x00;
625
626 // Byte 8: CRC8 checksum (calculated automatically)
627 uart_frame.CalculateCrc();
628
629 return uart_frame;
630 }

641 uart_frame.type = UartFrameType::ReadRequest;
642
643 // Byte 0: Sync nibble + Reserved + NODEADDR
644 // Note: Current implementation uses simplified sync (bit 0 = 1) for compatibility
645 // Full sync pattern per datasheet: bits 0-3 = 1,0,1,0 (0x0A)
646 uart_frame.frame.read_request_fields.sync_nodeaddr = (node_addr & 0x7F) | 0x01;
647
648 // Byte 1: RW bit (bit 7 = 0 for read) + 7-bit register address
649 uart_frame.frame.read_request_fields.rw_address = reg_addr & 0x7F;
650
651 // Bytes 2-3: Reserved (don't cares but included in CRC)
652 uart_frame.frame.read_request_fields.reserved[0] = 0x00;
653 uart_frame.frame.read_request_fields.reserved[1] = 0x00;
654

656 if (use_9byte_frame) {
657 // Zero out bytes 4-7 (they're not part of the 5-byte frame but may be sent)
658 for (size_t i = 4; i < 8; ++i) {
659 uart_frame.frame.bytes[i] = 0x00;
660 }
661 // Calculate CRC over 8 bytes (bytes 0-7) for 9-byte frame
662 size_t crc_length = 8;
663 uint8_t calculated_crc = calculateCrc8(uart_frame.frame.bytes, crc_length);
664 uart_frame.frame.bytes[8] = calculated_crc;
665 } else {
666 // Standard 5-byte frame per datasheet: CRC over bytes 0-3, CRC in byte 4
667 uart_frame.CalculateCrc();
668 }
669
670 return uart_frame;
671 }

1204 // Calculate transfer size for daisy-chaining
1205 // If total_chain_length_ is known (n > 0), use datasheet formula: 40·(n-k+1) bits
1206 // Otherwise, use simplified approach: (k+1) * 40 bits
1207 size_t transfer_bytes;
1208 size_t response_byte_offset;

1210 if (total_chain_length_ > 0) {
1211 // Use datasheet formula: 40·(n-k+1) bits total for chain of n devices
1212 // This includes: 40·(n-k) bits of padding + 40 bits of command
1213 // Response from device k appears at offset (n-k-1)*5 bytes (reverse order)
1214 uint8_t n = total_chain_length_;

1216 transfer_bytes = static_cast<size_t>(n - k + 1) * 5; // 40·(n-k+1) bits = (n-k+1)*5 bytes

1218 } else {

1220 // Transfer: (k+1) * 40 bits = (k+1) * 5 bytes
1221 // Response: Extract from offset k*5 bytes (end of transfer)

1276

1278 if (status.HasError()) {

1281 "Read register 0x%02X: SPI_STATUS=0x%02X [%s%s%s%s%s%s%s%s] - ERROR DETECTED",
