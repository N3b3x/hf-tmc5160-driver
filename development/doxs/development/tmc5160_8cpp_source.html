30// Helper function to constrain value between min and max
31template <typename T>
32static constexpr T constrain(T value, T min_val, T max_val) noexcept {
33 if (value < min_val) {
34 return min_val;
35 }
36 if (value > max_val) {
37 return max_val;
38 }
39 return value;

188template <typename CommType>
189int32_t TMC5160<CommType>::speedToInternal(float speed_hz) const noexcept {
190 // Datasheet formula: v[Hz] = v[5160] * (f_CLK[Hz]/2 / 2^23)
191 // Rearranged: v[5160] = v[Hz] * 2^24 / f_CLK
192 // Where v[Hz] is in μsteps/s (microsteps per second)
193 // Input speed_hz is in steps/s, so we multiply by microstep count (256) to convert to μsteps/s
194 // Final: v[5160] = (speed_hz * 256) * 2^24 / f_CLK
195 if (speed_hz == 0.0F) {
196 return 0;
197 }
198 float internal = (speed_hz * static_cast<float>(1UL << 24)) / static_cast<float>(f_clk_);
199 internal *= static_cast<float>(Microsteps::USTEP_COUNT);
200 return static_cast<int32_t>(internal);
201}

203template <typename CommType>
204float TMC5160<CommType>::speedFromInternal(int32_t speed_internal) const noexcept {
205 // Datasheet formula: v[Hz] = v[5160] * (f_CLK[Hz]/2 / 2^23)
206 // Where v[Hz] is in μsteps/s (microsteps per second)
207 // Output is in steps/s, so we divide by microstep count (256) to convert from μsteps/s
208 // Final: v[steps/s] = (v[5160] * f_CLK / 2^24) / 256
209 if (speed_internal == 0) {
210 return 0.0F;
211 }
212 float speed_hz = static_cast<float>(speed_internal) * static_cast<float>(f_clk_) / static_cast<float>(1UL << 24);
213 speed_hz /= static_cast<float>(Microsteps::USTEP_COUNT);
214 return speed_hz;
215}

217template <typename CommType>
218int32_t TMC5160<CommType>::accelToInternal(float accel_hz) const noexcept {
219 // Datasheet formula: a[Hz/s] = a[5160] * f_CLK[Hz]^2 / (512*256) / 2^24
220 // Rearranged: a[5160] = a[Hz/s] * (512*256) * 2^24 / f_CLK^2
221 // Where a[Hz/s] is in μsteps/s² (microsteps per second squared)
222 // Input accel_hz is in steps/s², so we multiply by microstep count (256) to convert to μsteps/s²
223 // Final: a[5160] = (accel_hz * 256) * (512*256) * 2^24 / f_CLK^2
224 if (accel_hz == 0.0F) {
225 return 0;
226 }
227 float internal = accel_hz * 512.0F * 256.0F * static_cast<float>(1UL << 24) /
228 (static_cast<float>(f_clk_) * static_cast<float>(f_clk_));
229 internal *= static_cast<float>(Microsteps::USTEP_COUNT);
230 return static_cast<int32_t>(internal);
231}

233template <typename CommType>
234int32_t TMC5160<CommType>::thresholdSpeedToTstep(float speed_hz) const noexcept {
235 // Datasheet formula: TSTEP = f_CLK / f256STEP = f_CLK / (fSTEP*256/USC)
236 // Where fSTEP is in μsteps/s, USC is microstep count (normally 256)
237 // For USC=256: TSTEP = f_CLK / fSTEP
238 // Input speed_hz is in steps/s, so fSTEP = speed_hz * 256 (convert to μsteps/s)
239 // Final: TSTEP = f_CLK / (speed_hz * 256)
240 // TSTEP is 20-bit unsigned (max value 0xFFFFF = 1048575)
241 if (speed_hz == 0.0F) {
242 return 0;
243 }
244 float tstep = static_cast<float>(f_clk_) / (speed_hz * 256.0F);
245 tstep = std::max(0.0F, std::min(1048575.0F, tstep));
246 return static_cast<int32_t>(tstep);
247}

288template <typename CommType>
289bool TMC5160<CommType>::RampControl::SetMaxSpeed(float speed) noexcept {
290 int32_t internal = driver_.speedToInternal(std::abs(speed));
291 internal = std::min(internal, static_cast<decltype(internal)>(0x7FFFFF)); // VMAX is 23 bits
292 if (!driver_.comm_.WriteRegister(Registers::VMAX, static_cast<uint32_t>(internal))) {
293 return false;
294 }
295 // If in velocity mode, update direction
296 uint32_t rampmode = 0;
297 if (driver_.comm_.ReadRegister(Registers::RAMPMODE, rampmode)) {
298 if (rampmode == static_cast<uint8_t>(RampMode::VELOCITY_POS) ||
299 rampmode == static_cast<uint8_t>(RampMode::VELOCITY_NEG)) {
300 uint8_t new_mode =
301 (speed < 0.0F) ? static_cast<uint8_t>(RampMode::VELOCITY_NEG) : static_cast<uint8_t>(RampMode::VELOCITY_POS);
302 driver_.comm_.WriteRegister(Registers::RAMPMODE, new_mode, driver_.GetCommAddress());
303 }
304 }
305 return true;
306}

339 success &= driver_.comm_.WriteRegister(Registers::V_1, static_cast<uint32_t>(v1));

654template <typename CommType>
655bool TMC5160<CommType>::MotorControl::SetupMotorFromSpec(const MotorSpec& motor_spec,
656 const MechanicalSystem* mechanical_system) noexcept {
657 // Calculate global scaler based on rated current
658 // Typical calculation: global_scaler = (rated_current_ma * 32) / (irun * sense_resistor_current)
659 // For simplicity, we'll use a basic calculation
660 uint16_t global_scaler = 32;
661 if (motor_spec.rated_current_ma > 0) {
662 // Basic calculation: assume 1.5A max current, scale accordingly
663 global_scaler = static_cast<uint16_t>(std::min(256U, std::max(32U, (motor_spec.rated_current_ma * 32U) / 1500U)));
664 }
665
666 // Set global scaler
667 if (!SetGlobalScaler(global_scaler)) {
668 return false;
669 }
670
671 // Calculate irun and ihold from rated current
672 // irun should be between 16-31 for best performance
673 // We'll use 80% of rated current for irun, 30% for ihold
674 uint8_t irun = 16;
675 uint8_t ihold = 0;
676 if (motor_spec.rated_current_ma > 0 && global_scaler > 0) {
677 // Calculate irun: target current = (irun/32) * (global_scaler/32) * sense_resistor_current
678 // Simplified: irun = (target_current * 32) / (global_scaler * sense_resistor_current / 32)
679 // Assuming sense resistor gives ~1.5A at irun=31, global_scaler=32
680 float target_run_current = static_cast<float>(motor_spec.rated_current_ma) * 0.8F;
681 auto irun_calc =
682 static_cast<uint32_t>((target_run_current * 32.0F) / (static_cast<float>(global_scaler) * 0.046875F));
683 irun = static_cast<uint8_t>(std::min(static_cast<uint32_t>(31U), std::max(static_cast<uint32_t>(16U), irun_calc)));
684 float target_hold_current = static_cast<float>(motor_spec.rated_current_ma) * 0.3F;
685 auto ihold_calc =
686 static_cast<uint32_t>((target_hold_current * 32.0F) / (static_cast<float>(global_scaler) * 0.046875F));
687 ihold = static_cast<uint8_t>(std::min(static_cast<uint32_t>(31U), ihold_calc));
688 }
689
690 // Set motor current
691 if (!SetCurrent(irun, ihold)) {
692 return false;
693 }
694
695 // Configure chopper based on inductance if available
696 if (motor_spec.winding_inductance_uh > 0) {
697 ChopperConfig chop_config{};
698 // Higher inductance may need different settings
699 // This is a simplified heuristic
700 if (motor_spec.winding_inductance_uh > 3000) {
701 chop_config.tbl = 3; // Longer blank time for high inductance
702 }
703 ConfigureChopper(chop_config);
704 }
705
706 // Note: mechanical_system is stored for unit conversions but not used here
707 // as it's used by the unit conversion functions, not motor setup
708
709 return true;
710}

738template <typename CommType>
739bool TMC5160<CommType>::Encoder::SetResolution(int32_t motor_steps, int32_t enc_resolution, bool inverted) noexcept {
740 // Calculate factor: (motor_steps * microsteps) / enc_resolution
741 float factor = static_cast<float>(motor_steps * Microsteps::USTEP_COUNT) / static_cast<float>(enc_resolution);
742
743 // Check if binary prescaler gives exact match
744 auto enc_const_binary = static_cast<int32_t>(factor * 65536.0F);
745 if (enc_const_binary * enc_resolution == motor_steps * Microsteps::USTEP_COUNT * 65536) {
746 // Use binary mode
747 uint32_t encmode_value = 0;
748 if (!driver_.comm_.ReadRegister(Registers::ENCMODE, encmode_value)) {
749 return false;
750 }
751 ENCMODE_Register encmode{};
752 encmode.value = encmode_value;
753 encmode.bits.enc_sel_decimal = false;
754 if (!driver_.comm_.WriteRegister(Registers::ENCMODE, encmode.value)) {
755 return false;
756 }
757 if (inverted) {
758 enc_const_binary = -enc_const_binary;
759 }
760 return driver_.comm_.WriteRegister(Registers::ENC_CONST, static_cast<uint32_t>(enc_const_binary));
761 }
762 // Use decimal mode
763 uint32_t encmode_value = 0;
764 if (!driver_.comm_.ReadRegister(Registers::ENCMODE, encmode_value)) {
765 return false;
766 }
767 ENCMODE_Register encmode{};
768 encmode.value = encmode_value;
769 encmode.bits.enc_sel_decimal = true;
770 if (!driver_.comm_.WriteRegister(Registers::ENCMODE, encmode.value)) {
771 return false;
772 }
773 int integer_part = static_cast<int>(std::floor(factor));
774 int decimal_part = static_cast<int>((factor - static_cast<float>(integer_part)) * 10000.0F);
775 if (inverted) {
776 integer_part = 65535 - integer_part;
777 decimal_part = 10000 - decimal_part;
778 }
779 int32_t enc_const_decimal = (integer_part * 65536) + decimal_part;
780 bool exact_match =
781 ((static_cast<int32_t>(factor * 10000.0F) * enc_resolution) == (motor_steps * Microsteps::USTEP_COUNT * 10000));
782 driver_.comm_.WriteRegister(Registers::ENC_CONST, static_cast<uint32_t>(enc_const_decimal));
783 return exact_match;
784}

1157 // During sequential programming, the device is accessible at address 0
1158 // (first device: NAI=GND, subsequent devices: previous NAO=LOW)
1159 // We need to use address 0 to communicate, not the target address
1160 uint8_t current_accessible_address = 0;
1161
1162 SLAVECONF_Register slaveconf{};
1163 slaveconf.bits.slaveaddr = slave_address & 0xFF; // Address range is 0-254 (8-bit)
1164 slaveconf.bits.senddelay = constrain<decltype(send_delay)>(send_delay, 0U, 15U);
1165
1166 // Write to SLAVECONF using current accessible address (0)
1167 bool success = driver_->comm_.WriteRegister(Registers::SLAVECONF, slaveconf.value, current_accessible_address);
1168
1169 // Only update the driver's node address after successful programming
1170 if (success) {
1171 driver_->uart_node_address_ = slave_address & 0xFF; // Address range is 0-254
1172 }
