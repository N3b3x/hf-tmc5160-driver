285
286 // Basic control pins

300 SetPinMapping(tmc5160::TMC5160CtrlPin::REFR_DIR, dir_gpio); // Same physical pin
301 }
302
303 // STEP/REFL_STEP compound pin (pin 17)
304 gpio_num_t step_gpio = UNMAPPED_PIN;
305 if (pin_config.step_pin != -1) {
306 step_gpio = static_cast<gpio_num_t>(pin_config.step_pin);
307 } else if (pin_config.ref_left_pin != -1) {
308 step_gpio = static_cast<gpio_num_t>(pin_config.ref_left_pin);
309 }
310 if (step_gpio != UNMAPPED_PIN) {
311 SetPinMapping(tmc5160::TMC5160CtrlPin::STEP, step_gpio);
312 SetPinMapping(tmc5160::TMC5160CtrlPin::REFL_STEP, step_gpio); // Same physical pin
313 }
314
315 // Diagnostic pins
316 if (pin_config.diag0_pin != -1) {
317 SetPinMapping(tmc5160::TMC5160CtrlPin::DIAG0, static_cast<gpio_num_t>(pin_config.diag0_pin));
318 }
319 if (pin_config.diag1_pin != -1) {
320 SetPinMapping(tmc5160::TMC5160CtrlPin::DIAG1, static_cast<gpio_num_t>(pin_config.diag1_pin));
321 }
322
323 // ENCA/DCIN compound pin (pin 24)
324 gpio_num_t enca_gpio = UNMAPPED_PIN;
325 if (pin_config.enc_a_pin != -1) {
326 enca_gpio = static_cast<gpio_num_t>(pin_config.enc_a_pin);

328 enca_gpio = static_cast<gpio_num_t>(pin_config.dc_in_pin);

331 SetPinMapping(tmc5160::TMC5160CtrlPin::ENCA, enca_gpio);

333 }

391 bool SetPinMapping(tmc5160::TMC5160CtrlPin pin, gpio_num_t gpio_pin) noexcept {
392 size_t index = static_cast<size_t>(pin);
393 if (index >= sizeof(pin_mapping_) / sizeof(pin_mapping_[0])) {
394 return false;
395 }
396 pin_mapping_[index] = gpio_pin;
397
398 constexpr gpio_num_t UNMAPPED_PIN = static_cast<gpio_num_t>(-1);
399 // Configure GPIO direction based on pin type
400 if (gpio_pin != UNMAPPED_PIN) {
401 // Diagnostic pins are outputs (read-only from MCU perspective)
402 if (pin == tmc5160::TMC5160CtrlPin::DIAG0 || pin == tmc5160::TMC5160CtrlPin::DIAG1) {
403 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT); // MCU reads TMC5160 output
404 gpio_set_pull_mode(gpio_pin, GPIO_PULLUP_ONLY); // Diagnostic pins have pullups
405 }
406 // Encoder pins are inputs (when SD_MODE=0)
407 else if (pin == tmc5160::TMC5160CtrlPin::ENCA || pin == tmc5160::TMC5160CtrlPin::ENCB ||
408 pin == tmc5160::TMC5160CtrlPin::ENCN) {
409 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT);
410 gpio_set_pull_mode(gpio_pin, GPIO_PULLUP_ONLY);
411 }
412 // DC Step pins (when SD_MODE=1, SPI_MODE=1)
413 else if (pin == tmc5160::TMC5160CtrlPin::DCEN || pin == tmc5160::TMC5160CtrlPin::DCIN) {
414 gpio_set_direction(gpio_pin, GPIO_MODE_OUTPUT); // DCEN and DCIN are inputs to TMC5160
415 } else if (pin == tmc5160::TMC5160CtrlPin::DCO) {
416 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT); // DCO is output from TMC5160
417 gpio_set_pull_mode(gpio_pin, GPIO_PULLUP_ONLY);
418 }
419 // Reference switch pins can be inputs (when used as reference switches, SD_MODE=0)
420 else if (pin == tmc5160::TMC5160CtrlPin::REFL_STEP || pin == tmc5160::TMC5160CtrlPin::REFR_DIR) {
421 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT);
422 gpio_set_pull_mode(gpio_pin, GPIO_PULLUP_ONLY); // Typically active LOW
423 active_levels_.SetActiveLevel(pin, false); // Active LOW for reference switches
424 }
425 // CLK pin can be output (for external clock) or input (for reading clock state)
426 else if (pin == tmc5160::TMC5160CtrlPin::CLK) {
427 // Default to input, user can configure as output if needed for PWM clock
428 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT);
429 }
430 // Mode configuration pins (SPI_MODE, SD_MODE) - INPUT by default (read-only)
431 // These pins are typically hardwired on dev boards. Only set as OUTPUT if user
432 // explicitly calls SetChipCommMode() or GpioSet() to control them.
433 else if (pin == tmc5160::TMC5160CtrlPin::SPI_MODE || pin == tmc5160::TMC5160CtrlPin::SD_MODE) {
434 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT); // Input to read chip mode (hardwired on dev boards)
435 gpio_set_pull_mode(gpio_pin, GPIO_PULLUP_ONLY); // Pullup for stable reading
436 }
437 // Other pins default to output
438 else {
439 gpio_set_direction(gpio_pin, GPIO_MODE_OUTPUT);
440 }
441 }
442 return true;
443 }

590 bool GpioSet(tmc5160::TMC5160CtrlPin pin, tmc5160::GpioSignal signal) noexcept {
591 // Diagnostic pins and DCO are read-only (outputs from TMC5160)
592 if (pin == tmc5160::TMC5160CtrlPin::DIAG0 || pin == tmc5160::TMC5160CtrlPin::DIAG1 ||
593 pin == tmc5160::TMC5160CtrlPin::DCO) {
594 ESP_LOGW(BUS_TAG, "Pin is read-only (output from TMC5160)");
595 return false;
596 }
597 // Encoder pins are read-only when used as encoder inputs (SD_MODE=0)
598 if (pin == tmc5160::TMC5160CtrlPin::ENCA || pin == tmc5160::TMC5160CtrlPin::ENCB ||
599 pin == tmc5160::TMC5160CtrlPin::ENCN) {
600 ESP_LOGW(BUS_TAG, "Encoder pins are read-only (use DCEN/DCIN for DC Step mode)");
601 return false;
602 }
603
604 gpio_num_t gpio_pin = GetPinMapping(pin);
605 constexpr gpio_num_t UNMAPPED_PIN = static_cast<gpio_num_t>(-1);
606 if (gpio_pin == UNMAPPED_PIN) {
607 ESP_LOGW(BUS_TAG, "Pin not mapped: %d", static_cast<int>(pin));
608 return false;
609 }
610
611 // If user is trying to set SPI_MODE or SD_MODE pins, configure them as OUTPUT
612 // (they default to INPUT for read-only mode on hardwired dev boards)
613 if (pin == tmc5160::TMC5160CtrlPin::SPI_MODE || pin == tmc5160::TMC5160CtrlPin::SD_MODE) {
614 gpio_set_direction(gpio_pin, GPIO_MODE_OUTPUT);
615 }
616
617 bool level = SignalToGpioLevel(pin, signal);
618 gpio_set_level(gpio_pin, level ? 1 : 0);
619 return true;
620 }
