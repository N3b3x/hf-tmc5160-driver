107 ret = spi_bus_add_device(host_, &dev_config, &device_handle_);

151 bool SpiTransfer(const uint8_t *tx, uint8_t *rx, size_t length) noexcept {
152 if (!initialized_ || !device_handle_) {
153 ESP_LOGE(BUS_TAG, "SPI interface not initialized");
154 return false;
155 }
156
157 spi_transaction_t trans = {};
158 trans.length = length * 8;
159 trans.tx_buffer = tx;
160 trans.rx_buffer = rx;
161
162 esp_err_t ret = spi_device_transmit(device_handle_, &trans);
163 if (ret != ESP_OK) {
164 ESP_LOGE(BUS_TAG, "SPI transfer failed: %s", esp_err_to_name(ret));
165 return false;
166 }
167
168 return true;
169 }
