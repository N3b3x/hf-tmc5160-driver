148// ------------- CRC16-CCITT FUNCTION -------------
149
150inline uint16_t crc16_ccitt(const uint8_t* data, size_t len)
151{
152 uint16_t crc = 0xFFFF;
153 for (size_t i = 0; i < len; ++i) {
154 crc ^= (uint16_t)data[i] << 8;
155 for (int j = 0; j < 8; ++j) {
156 if (crc & 0x8000)
157 crc = (crc << 1) ^ 0x1021;
158 else
159 crc <<= 1;
160 }
161 }
162 return crc;
163}
