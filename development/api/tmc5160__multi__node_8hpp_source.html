236 bool ProgramSequentially(uint8_t send_delay = 2) noexcept {
237 // Count active devices to determine starting address
238 uint8_t num_devices = 0;
239 for (size_t i = 0; i < MaxDevices; ++i) {
240 if (drivers_[i].has_value()) {
241 num_devices++;
242 }
243 }
244
245 if (num_devices == 0) {
246 return false; // No devices to program
247 }
248
249 // Program devices in forward order (index 0, 1, 2, ...)
250 // but assign addresses in reverse order (254, 253, 252, ...)
251 // First chip: accessible at address 0 (NAI=GND)
252 // After each programming, next chip becomes accessible at address 0
253
254 uint8_t device_index = 0;
255 for (size_t i = 0; i < MaxDevices; ++i) {
256 if (!drivers_[i].has_value()) {
257 continue; // Skip empty slots
258 }
259
260 // Calculate target address: start from 254 and count down
261 // Device at index 0 gets address 254, index 1 gets 253, etc.
262 uint8_t target_address = 254 - device_index;
263
264 // Program device to target address
265 // The device is currently accessible at address 0 (after previous chip's NAO went LOW)
266 // or address 0 for the first chip (NAI=GND)
267 if (!drivers_[i]->uartConfig.ConfigureSlave(target_address, send_delay)) {
268 return false; // Failed to program device
269 }
270
271 // Update the driver's node address to the programmed address
272 drivers_[i]->SetUartNodeAddress(target_address);
273
274 // After programming, the chip's NAO automatically goes LOW (per datasheet)
275 // This makes the next chip accessible at address 0
276 // Small delay to ensure NAO settles
277 comm_.DelayMs(10);
278
279 device_index++;
280 }
281
282 return true;
283 }

295 bool ProgramDevice(uint8_t index, uint8_t send_delay = 2) noexcept {
296 if (index >= MaxDevices || !drivers_[index].has_value()) {
297 return false;
298 }
299
300 // Calculate target address: 254 - index (per datasheet)
301 uint8_t target_address = 254 - index;
302
303 if (!drivers_[index]->uartConfig.ConfigureSlave(target_address, send_delay)) {
304 return false;
305 }
306
307 drivers_[index]->SetUartNodeAddress(target_address);
308 return true;

326 bool AddDevice(uint8_t index) noexcept {
327 // Validate index
328 if (index < num_onboard_devices_ || index >= MaxDevices) {
329 return false;
330 }
331
332 // Check if slot is already occupied
333 if (drivers_[index].has_value()) {
334 return false;
335 }
336
337 // Enforce sequential indexing: cannot skip indices
338 // All indices before this one (starting from num_onboard_devices) must be filled
339 for (uint8_t i = num_onboard_devices_; i < index; ++i) {
340 if (!drivers_[i].has_value()) {
341 return false; // Cannot skip indices
342 }
343 }
344
345 // Create device instance with initial address 0
346 // The actual address will be programmed via ProgramSequentially() or ProgramDevice()
347 drivers_[index] = std::make_optional<TMC5160<CommType>>(comm_, f_clk_, 0, 0);
348 num_active_devices_++;
349
350 return true;
351 }

368 bool RemoveDevice(uint8_t index) noexcept {
369 // Cannot remove onboard devices
370 if (index < num_onboard_devices_) {
371 return false;
372 }
373
374 // Validate index
375 if (index >= MaxDevices) {
376 return false;
377 }
378
379 // Check if slot is actually active
380 if (!drivers_[index].has_value()) {
381 return false;
382 }
383
384 // Enforce sequential removal: can only remove from the end
385 // Check if there are any devices after this index
386 for (size_t i = index + 1; i < MaxDevices; ++i) {
387 if (drivers_[i].has_value()) {
388 return false; // Cannot remove device in the middle of the chain
389 }
390 }
391
392 // Remove device
393 drivers_[index] = std::nullopt;
394 num_active_devices_--;
395
396 return true;
