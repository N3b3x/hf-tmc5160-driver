763static constexpr uint8_t calculateCrc8(const uint8_t *data,
764 size_t length) noexcept {
765 uint8_t crc = 0; // Initial value is zero per datasheet
766
767 // Process each byte LSB to MSB
768 for (size_t i = 0; i < length; ++i) {
769 uint8_t currentByte = data[i];
770
771 // Process each bit LSB to MSB (j=0 is LSB, j=7 is MSB)
772 for (uint8_t j = 0; j < 8; ++j) {
773 // Check: (CRC >> 7) XOR (currentByte & 0x01)
774 // This XORs the MSB of CRC with the LSB of currentByte
775 if (((crc >> 7) ^ (currentByte & 0x01)) != 0) {
776 crc = (crc << 1) ^ 0x07; // Polynomial 0x07 (CRC8-ATM)
777 } else {
778 crc = (crc << 1);
779 }
780 currentByte = currentByte >> 1; // Shift to next bit (LSB to MSB)
781 }
782 }
783
784 return crc;
785}
