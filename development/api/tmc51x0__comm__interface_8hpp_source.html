706static constexpr uint8_t calculateCrc8(const uint8_t* data, size_t length) noexcept {
707 uint8_t crc = 0; // Initial value is zero per datasheet
708
709 // Process each byte LSB to MSB
710 for (size_t i = 0; i < length; ++i) {
711 uint8_t currentByte = data[i];
712
713 // Process each bit LSB to MSB (j=0 is LSB, j=7 is MSB)
714 for (uint8_t j = 0; j < 8; ++j) {
715 // Check: (CRC >> 7) XOR (currentByte & 0x01)
716 // This XORs the MSB of CRC with the LSB of currentByte
717 if (((crc >> 7) ^ (currentByte & 0x01)) != 0) {
718 crc = (crc << 1) ^ 0x07; // Polynomial 0x07 (CRC8-ATM)
719 } else {
720 crc = (crc << 1);
721 }
722 currentByte = currentByte >> 1; // Shift to next bit (LSB to MSB)
723 }
724 }
725
726 return crc;
727}
