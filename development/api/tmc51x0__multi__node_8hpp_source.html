254 Result<void> ProgramSequentially(uint8_t send_delay = 2) noexcept {
255 // Count active devices to determine starting address
256 uint8_t num_devices = 0;
257 for (size_t i = 0; i < MaxDevices; ++i) {
258 if (drivers_[i].has_value()) {
259 num_devices++;
260 }
261 }
262
263 if (num_devices == 0) {
264 return Result<void>(ErrorCode::INVALID_STATE); // No devices to program
265 }
266
267 // Program devices in forward order (index 0, 1, 2, ...)
268 // but assign addresses in reverse order (254, 253, 252, ...)
269 // First chip: accessible at address 0 (NAI=GND)
270 // After each programming, next chip becomes accessible at address 0
271
272 uint8_t device_index = 0;
273 for (size_t i = 0; i < MaxDevices; ++i) {
274 if (!drivers_[i].has_value()) {
275 continue; // Skip empty slots
276 }
277
278 // Calculate target address: start from 254 and count down
279 // Device at index 0 gets address 254, index 1 gets 253, etc.
280 uint8_t target_address = 254 - device_index;
281
282 // Program device to target address
283 // The device is currently accessible at address 0 (after previous chip's
284 // NAO went LOW) or address 0 for the first chip (NAI=GND)
285 if (!drivers_[i]->uartConfig.ConfigureUartNodeAddress(target_address,
286 send_delay)) {
287 return Result<void>(
288 ErrorCode::INVALID_STATE); // Failed to program device
289 }
290
291 // Update the driver's node address to the programmed address
292 drivers_[i]->SetUartNodeAddress(target_address);
293
294 // After programming, the chip's NAO automatically goes LOW (per
295 // datasheet) This makes the next chip accessible at address 0 Small delay
296 // to ensure NAO settles
297 comm_.DelayMs(10);
298
299 device_index++;
300 }
301
302 return Result<void>();
303 }

350 Result<void> AddDevice(uint8_t index) noexcept {
351 // Validate index
352 if (index < num_onboard_devices_ || index >= MaxDevices) {
353 return Result<void>(ErrorCode::INVALID_STATE);
354 }
355
356 // Check if slot is already occupied
357 if (drivers_[index].has_value()) {
358 return Result<void>(ErrorCode::INVALID_STATE);
359 }
360
361 // Enforce sequential indexing: cannot skip indices
362 // All indices before this one (starting from num_onboard_devices) must be
363 // filled
364 for (uint8_t i = num_onboard_devices_; i < index; ++i) {
365 if (!drivers_[i].has_value()) {
366 return Result<void>(ErrorCode::INVALID_STATE); // Cannot skip indices
367 }
368 }
369
370 // Create device instance with initial address 0
371 // The actual address will be programmed via ProgramSequentially() or
372 // ProgramDevice()
373 drivers_[index] =
374 std::make_optional<TMC51x0<CommType>>(comm_, f_clk_, 0, 0);
375 num_active_devices_++;
376
377 return Result<void>();
378 }

395 Result<void> RemoveDevice(uint8_t index) noexcept {
396 // Cannot remove onboard devices
397 if (index < num_onboard_devices_) {
398 return Result<void>(ErrorCode::INVALID_STATE);
399 }
400
401 // Validate index
402 if (index >= MaxDevices) {
403 return Result<void>(ErrorCode::INVALID_STATE);
404 }
405
406 // Check if slot is actually active
407 if (!drivers_[index].has_value()) {
408 return Result<void>(ErrorCode::INVALID_STATE);
409 }
410
411 // Enforce sequential removal: can only remove from the end
412 // Check if there are any devices after this index
413 for (size_t i = index + 1; i < MaxDevices; ++i) {
414 if (drivers_[i].has_value()) {
415 return Result<void>(
416 ErrorCode::INVALID_STATE); // Cannot remove device in the middle of
417 // the chain
418 }
419 }
420
421 // Remove device
422 drivers_[index] = std::nullopt;
423 num_active_devices_--;
424
425 return Result<void>();
