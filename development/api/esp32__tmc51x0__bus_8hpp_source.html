338 bool ApplyPinConfig(const tmc51x0::TMC51x0PinConfig& pin_config) noexcept {
339 constexpr gpio_num_t UNMAPPED_PIN = static_cast<gpio_num_t>(-1);
340
341 // Basic control pins
342 if (pin_config.en_pin != -1) {
343 SetPinMapping(tmc51x0::TMC51x0CtrlPin::EN, static_cast<gpio_num_t>(pin_config.en_pin));
344 }
345
346 // DIR/REFR_DIR compound pin (pin 18)
347 gpio_num_t dir_gpio = UNMAPPED_PIN;
348 if (pin_config.dir_pin != -1) {
349 dir_gpio = static_cast<gpio_num_t>(pin_config.dir_pin);
350 } else if (pin_config.ref_right_pin != -1) {
351 dir_gpio = static_cast<gpio_num_t>(pin_config.ref_right_pin);
352 }
353 if (dir_gpio != UNMAPPED_PIN) {
354 SetPinMapping(tmc51x0::TMC51x0CtrlPin::DIR, dir_gpio);
355 SetPinMapping(tmc51x0::TMC51x0CtrlPin::REFR_DIR, dir_gpio); // Same physical pin
356 }
357
358 // STEP/REFL_STEP compound pin (pin 17)
359 gpio_num_t step_gpio = UNMAPPED_PIN;
360 if (pin_config.step_pin != -1) {
361 step_gpio = static_cast<gpio_num_t>(pin_config.step_pin);
362 } else if (pin_config.ref_left_pin != -1) {
363 step_gpio = static_cast<gpio_num_t>(pin_config.ref_left_pin);
364 }
365 if (step_gpio != UNMAPPED_PIN) {
366 SetPinMapping(tmc51x0::TMC51x0CtrlPin::STEP, step_gpio);
367 SetPinMapping(tmc51x0::TMC51x0CtrlPin::REFL_STEP, step_gpio); // Same physical pin
368 }
369
370 // Diagnostic pins
371 if (pin_config.diag0_pin != -1) {
372 SetPinMapping(tmc51x0::TMC51x0CtrlPin::DIAG0, static_cast<gpio_num_t>(pin_config.diag0_pin));
373 }
374 if (pin_config.diag1_pin != -1) {
375 SetPinMapping(tmc51x0::TMC51x0CtrlPin::DIAG1, static_cast<gpio_num_t>(pin_config.diag1_pin));
376 }
377
378 // ENCA/DCIN compound pin (pin 24)
379 gpio_num_t enca_gpio = UNMAPPED_PIN;
380 if (pin_config.enc_a_pin != -1) {
381 enca_gpio = static_cast<gpio_num_t>(pin_config.enc_a_pin);
382 } else if (pin_config.dc_in_pin != -1) {
383 enca_gpio = static_cast<gpio_num_t>(pin_config.dc_in_pin);
384 }
385 if (enca_gpio != UNMAPPED_PIN) {
386 SetPinMapping(tmc51x0::TMC51x0CtrlPin::ENCA, enca_gpio);
387 SetPinMapping(tmc51x0::TMC51x0CtrlPin::DCIN, enca_gpio); // Same physical pin
388 }
389
390 // ENCB/DCEN compound pin (pin 23)
391 gpio_num_t encb_gpio = UNMAPPED_PIN;
392 if (pin_config.enc_b_pin != -1) {
393 encb_gpio = static_cast<gpio_num_t>(pin_config.enc_b_pin);
394 } else if (pin_config.dc_en_pin != -1) {
395 encb_gpio = static_cast<gpio_num_t>(pin_config.dc_en_pin);
396 }
397 if (encb_gpio != UNMAPPED_PIN) {
398 SetPinMapping(tmc51x0::TMC51x0CtrlPin::ENCB, encb_gpio);

400 }

403 gpio_num_t encn_gpio = UNMAPPED_PIN;

405 encn_gpio = static_cast<gpio_num_t>(pin_config.enc_n_pin);

446 bool SetPinMapping(tmc51x0::TMC51x0CtrlPin pin, gpio_num_t gpio_pin) noexcept {
447 size_t index = static_cast<size_t>(pin);
448 if (index >= sizeof(pin_mapping_) / sizeof(pin_mapping_[0])) {
449 return false;
450 }
451 pin_mapping_[index] = gpio_pin;
452
453 constexpr gpio_num_t UNMAPPED_PIN = static_cast<gpio_num_t>(-1);
454 // Configure GPIO direction based on pin type
455 if (gpio_pin != UNMAPPED_PIN) {
456 // Diagnostic pins are outputs (read-only from MCU perspective)
457 if (pin == tmc51x0::TMC51x0CtrlPin::DIAG0 || pin == tmc51x0::TMC51x0CtrlPin::DIAG1) {
458 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT); // MCU reads TMC51x0 output
459 gpio_set_pull_mode(gpio_pin, GPIO_PULLUP_ONLY); // Diagnostic pins have pullups
460 }
461 // Encoder pins are inputs (when SD_MODE=0)
462 else if (pin == tmc51x0::TMC51x0CtrlPin::ENCA || pin == tmc51x0::TMC51x0CtrlPin::ENCB ||
463 pin == tmc51x0::TMC51x0CtrlPin::ENCN) {
464 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT);
465 gpio_set_pull_mode(gpio_pin, GPIO_PULLUP_ONLY);
466 }
467 // DC Step pins (when SD_MODE=1, SPI_MODE=1)
468 else if (pin == tmc51x0::TMC51x0CtrlPin::DCEN || pin == tmc51x0::TMC51x0CtrlPin::DCIN) {
469 gpio_set_direction(gpio_pin, GPIO_MODE_OUTPUT); // DCEN and DCIN are inputs to TMC51x0
470 } else if (pin == tmc51x0::TMC51x0CtrlPin::DCO) {
471 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT); // DCO is output from TMC51x0
472 gpio_set_pull_mode(gpio_pin, GPIO_PULLUP_ONLY);
473 }
474 // Reference switch pins can be inputs (when used as reference switches, SD_MODE=0)
475 else if (pin == tmc51x0::TMC51x0CtrlPin::REFL_STEP || pin == tmc51x0::TMC51x0CtrlPin::REFR_DIR) {
476 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT);
477 gpio_set_pull_mode(gpio_pin, GPIO_PULLUP_ONLY); // Typically active LOW
478 active_levels_.SetActiveLevel(pin, false); // Active LOW for reference switches
479 }
480 // CLK pin can be output (for external clock) or input (for reading clock state)
481 else if (pin == tmc51x0::TMC51x0CtrlPin::CLK) {
482 // Default to input, user can configure as output if needed for PWM clock
483 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT);
484 }
485 // Mode configuration pins (SPI_MODE, SD_MODE) - INPUT by default (read-only)
486 // These pins are typically hardwired on dev boards. Only set as OUTPUT if user
487 // explicitly calls communication.SetOperatingMode() or GpioSet() to control them.
488 else if (pin == tmc51x0::TMC51x0CtrlPin::SPI_MODE || pin == tmc51x0::TMC51x0CtrlPin::SD_MODE) {
489 gpio_set_direction(gpio_pin, GPIO_MODE_INPUT); // Input to read chip mode (hardwired on dev boards)
490 gpio_set_pull_mode(gpio_pin, GPIO_PULLUP_ONLY); // Pullup for stable reading
491 }
492 // Other pins default to output
493 else {
494 gpio_set_direction(gpio_pin, GPIO_MODE_OUTPUT);
495 }
496 }
497 return true;
498 }

518 tmc51x0::Result<void> Initialize() noexcept {
519 if (initialized_) {
520 return tmc51x0::Result<void>();
521 }
522
523 // Configure GPIO pins that are mapped
524 constexpr gpio_num_t UNMAPPED_PIN = static_cast<gpio_num_t>(-1);
525 if (en_pin_ != UNMAPPED_PIN) {
526 gpio_set_direction(en_pin_, GPIO_MODE_OUTPUT);
527 // Disable by default using signal abstraction (EN is active LOW, so INACTIVE = HIGH = disabled)
528 auto en_result = GpioSet(tmc51x0::TMC51x0CtrlPin::EN, tmc51x0::GpioSignal::INACTIVE);
529 if (!en_result.IsOk()) {
530 ESP_LOGW(BUS_TAG, "Failed to set EN pin during initialization (non-critical)");
531 // Continue anyway - this is not critical for initialization
532 }
533 }
534 if (dir_pin_ != UNMAPPED_PIN) {
535 gpio_set_direction(dir_pin_, GPIO_MODE_OUTPUT);
536 // Set DIR to inactive by default using signal abstraction
537 auto dir_result = GpioSet(tmc51x0::TMC51x0CtrlPin::DIR, tmc51x0::GpioSignal::INACTIVE);
538 if (!dir_result.IsOk()) {
539 ESP_LOGW(BUS_TAG, "Failed to set DIR pin during initialization (non-critical)");
540 // Continue anyway - this is not critical for initialization
541 }
542 }
543 if (step_pin_ != UNMAPPED_PIN) {
544 gpio_set_direction(step_pin_, GPIO_MODE_OUTPUT);
545 // Set STEP to inactive by default using signal abstraction
546 auto step_result = GpioSet(tmc51x0::TMC51x0CtrlPin::STEP, tmc51x0::GpioSignal::INACTIVE);
547 if (!step_result.IsOk()) {
548 ESP_LOGW(BUS_TAG, "Failed to set STEP pin during initialization (non-critical)");
549 // Continue anyway - this is not critical for initialization
550 }
551 }

553 // Configure SPI bus
554 spi_bus_config_t bus_config = {};
555 bus_config.mosi_io_num = mosi_pin_;
556 bus_config.miso_io_num = miso_pin_;
557 bus_config.sclk_io_num = sclk_pin_;
558 bus_config.quadwp_io_num = -1;

563 esp_err_t ret = spi_bus_initialize(host_, &bus_config, SPI_DMA_CH_AUTO);

566 return tmc51x0::Result<void>(tmc51x0::ErrorCode::COMM_ERROR);
567 }

569 // Configure SPI device (Mode 3: CPOL=1, CPHA=1)
570 spi_device_interface_config_t dev_config = {};
571 dev_config.clock_speed_hz = clock_speed_hz_;
572 dev_config.mode = 3; // SPI Mode 3 for TMC51x0
573 dev_config.spics_io_num = cs_pin_;
574 dev_config.queue_size = 1;
575 dev_config.cs_ena_pretrans = 2;
576 dev_config.cs_ena_posttrans = 2;

578 ret = spi_bus_add_device(host_, &dev_config, &device_handle_);
579 if (ret != ESP_OK) {
580 ESP_LOGE(BUS_TAG, "Failed to add SPI device: %s", esp_err_to_name(ret));

585 initialized_ = true;

682 tmc51x0::Result<void> GpioSet(tmc51x0::TMC51x0CtrlPin pin, tmc51x0::GpioSignal signal) noexcept {
683 // Diagnostic pins and DCO are read-only (outputs from TMC51x0)
684 if (pin == tmc51x0::TMC51x0CtrlPin::DIAG0 || pin == tmc51x0::TMC51x0CtrlPin::DIAG1 ||
685 pin == tmc51x0::TMC51x0CtrlPin::DCO) {
686 ESP_LOGW(BUS_TAG, "Pin is read-only (output from TMC51x0)");
687 return tmc51x0::Result<void>(tmc51x0::ErrorCode::INVALID_VALUE);
688 }
689 // Encoder pins are read-only when used as encoder inputs (SD_MODE=0)
690 if (pin == tmc51x0::TMC51x0CtrlPin::ENCA || pin == tmc51x0::TMC51x0CtrlPin::ENCB ||
691 pin == tmc51x0::TMC51x0CtrlPin::ENCN) {
692 ESP_LOGW(BUS_TAG, "Encoder pins are read-only (use DCEN/DCIN for DC Step mode)");
693 return tmc51x0::Result<void>(tmc51x0::ErrorCode::INVALID_VALUE);
694 }
695
696 gpio_num_t gpio_pin = GetPinMapping(pin);
697 constexpr gpio_num_t UNMAPPED_PIN = static_cast<gpio_num_t>(-1);
698 if (gpio_pin == UNMAPPED_PIN) {
699 ESP_LOGW(BUS_TAG, "Pin not mapped: %d", static_cast<int>(pin));
700 return tmc51x0::Result<void>(tmc51x0::ErrorCode::INVALID_VALUE);
701 }
702
703 // If user is trying to set SPI_MODE or SD_MODE pins, configure them as OUTPUT
704 // (they default to INPUT for read-only mode on hardwired dev boards)
705 if (pin == tmc51x0::TMC51x0CtrlPin::SPI_MODE || pin == tmc51x0::TMC51x0CtrlPin::SD_MODE) {
706 gpio_set_direction(gpio_pin, GPIO_MODE_OUTPUT);
707 }
708
709 bool level = SignalToGpioLevel(pin, signal);
710 gpio_set_level(gpio_pin, level ? 1 : 0);
711 return tmc51x0::Result<void>();
712 }

796 tmc51x0::Result<void> SetClkFreq(uint32_t frequency_hz) noexcept {
797 gpio_num_t clk_pin = GetPinMapping(tmc51x0::TMC51x0CtrlPin::CLK);
798 constexpr gpio_num_t UNMAPPED_PIN = static_cast<gpio_num_t>(-1);
799
800 if (clk_pin == UNMAPPED_PIN) {
801 ESP_LOGW(BUS_TAG, "CLK pin not mapped, cannot configure clock");
802 return tmc51x0::Result<void>(tmc51x0::ErrorCode::INVALID_VALUE);
803 }
804
805 if (frequency_hz == 0) {
806 // Internal clock mode: Set CLK pin to GND (LOW)
807 gpio_set_direction(clk_pin, GPIO_MODE_OUTPUT);
808 gpio_set_level(clk_pin, 0); // Drive LOW (GND) for internal oscillator
809 ESP_LOGI(BUS_TAG, "CLK pin set to GND (internal 12 MHz oscillator enabled)");
810 return tmc51x0::Result<void>();
811 } else {
812 // External clock mode: Not implemented via PWM
813 // User must provide external clock signal on CLK pin
814 ESP_LOGW(BUS_TAG, "External clock generation not implemented. Provide external clock signal on CLK pin (frequency: %u Hz)", frequency_hz);
815 return tmc51x0::Result<void>(tmc51x0::ErrorCode::UNSUPPORTED);
816 }
