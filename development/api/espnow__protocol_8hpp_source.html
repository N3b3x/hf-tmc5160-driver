375inline uint16_t Crc16Ccitt(const uint8_t* data, size_t len) noexcept
376{
377 uint16_t crc = 0xFFFF;
378 for (size_t i = 0; i < len; ++i) {
379 crc ^= static_cast<uint16_t>(data[i]) << 8;
380 for (int j = 0; j < 8; ++j) {
381 if (crc & 0x8000) {
382 crc = (crc << 1) ^ 0x1021;
383 } else {
384 crc <<= 1;
385 }
386 }
387 }
388 return crc;
389}
