126 explicit TMC51x0DaisyChain(CommType& comm, uint8_t num_onboard_devices,
127 uint32_t f_clk = ClockFreq::DEFAULT_F_CLK) noexcept
128 : comm_(comm), num_onboard_devices_(num_onboard_devices), num_active_devices_(num_onboard_devices),
129 f_clk_(f_clk) {
130 // Validate num_onboard_devices
131 if (num_onboard_devices == 0 || num_onboard_devices > MaxDevices) {
132 num_onboard_devices_ = 1;
133 num_active_devices_ = 1;
134 }
135
136 // Create onboard TMC51x0 instances, one per device
137 // Each device has its position (0, 1, 2, ...) set in constructor
138 for (uint8_t i = 0; i < num_onboard_devices_; ++i) {
139 drivers_[i] = std::make_optional<TMC51x0<CommType>>(comm_, f_clk_, i);
140 }
141
142 // Initialize extra device slots as empty
143 for (size_t i = num_onboard_devices_; i < MaxDevices; ++i) {
144 drivers_[i] = std::nullopt;
145 }
146
147 // CRITICAL: Set the total chain length on the SpiCommInterface
148 // This enables proper response extraction using datasheet formula 40·(n-k+1)
149 UpdateChainLength();
150 }

204 bool AddDevice(uint8_t position) noexcept {
205 // Validate position
206 if (position < num_onboard_devices_ || position >= MaxDevices) {
207 return false;
208 }
209
210 // Check if slot is already occupied
211 if (drivers_[position].has_value()) {
212 return false;
213 }
214
215 // Enforce sequential positioning: cannot skip positions
216 // All positions before this one (starting from num_onboard_devices) must be filled
217 for (uint8_t i = num_onboard_devices_; i < position; ++i) {
218 if (!drivers_[i].has_value()) {
219 return false; // Cannot skip positions in daisy chain
220 }
221 }
222
223 // Create device instance at specified position
224 drivers_[position] = std::make_optional<TMC51x0<CommType>>(comm_, f_clk_, position);
225 num_active_devices_++;
226
227 // Update chain length on SpiCommInterface
228 UpdateChainLength();
229
230 return true;
231 }

249 bool RemoveDevice(uint8_t position) noexcept {
250 // Cannot remove onboard devices
251 if (position < num_onboard_devices_) {
252 return false;
253 }
254
255 // Validate position
256 if (position >= MaxDevices) {
257 return false;
258 }
259
260 // Check if slot is actually active
261 if (!drivers_[position].has_value()) {
262 return false;
263 }
264
265 // Enforce sequential removal: can only remove from the end
266 // Check if there are any devices after this position
267 for (size_t i = position + 1; i < MaxDevices; ++i) {
268 if (drivers_[i].has_value()) {
269 return false; // Cannot remove device in the middle of the chain
270 }
271 }
272
273 // Remove device
274 drivers_[position] = std::nullopt;
275 num_active_devices_--;
276
277 // Update chain length on SpiCommInterface
278 UpdateChainLength();
279
280 return true;
281 }

330 void UpdateChainLength() noexcept {
331 // Find the highest active device position
332 uint8_t max_position = 0;
333 for (size_t i = 0; i < MaxDevices; ++i) {
334 if (drivers_[i].has_value()) {
335 max_position = static_cast<uint8_t>(i);
336 }
337 }
338
339 // Total chain length is max_position + 1 (0-indexed to 1-indexed)
340 uint8_t total_length = max_position + 1;
341 comm_.SetDaisyChainLength(total_length);
342 }
