139 explicit TMC51x0DaisyChain(CommType &comm, uint8_t num_onboard_devices,
140 uint32_t f_clk = ClockFreq::DEFAULT_F_CLK) noexcept
141 : comm_(comm), num_onboard_devices_(num_onboard_devices),
142 num_active_devices_(num_onboard_devices), f_clk_(f_clk) {
143 // Validate num_onboard_devices
144 if (num_onboard_devices == 0 || num_onboard_devices > MaxDevices) {
145 num_onboard_devices_ = 1;
146 num_active_devices_ = 1;
147 }
148
149 // Create onboard TMC51x0 instances, one per device
150 // Each device has its position (0, 1, 2, ...) set in constructor
151 for (uint8_t i = 0; i < num_onboard_devices_; ++i) {
152 drivers_[i] = std::make_optional<TMC51x0<CommType>>(comm_, f_clk_, i);
153 }
154
155 // Initialize extra device slots as empty
156 for (size_t i = num_onboard_devices_; i < MaxDevices; ++i) {
157 drivers_[i] = std::nullopt;
158 }
159
160 // CRITICAL: Set the total chain length on the SpiCommInterface
161 // This enables proper response extraction using datasheet formula
162 // 40·(n-k+1)
163 UpdateChainLength();
164 }

219 Result<void> AddDevice(uint8_t position) noexcept {
220 // Validate position
221 if (position < num_onboard_devices_ || position >= MaxDevices) {
222 return Result<void>(ErrorCode::INVALID_VALUE);
223 }
224
225 // Check if slot is already occupied
226 if (drivers_[position].has_value()) {
227 return Result<void>(ErrorCode::INVALID_STATE);
228 }
229
230 // Enforce sequential positioning: cannot skip positions
231 // All positions before this one (starting from num_onboard_devices) must be
232 // filled
233 for (uint8_t i = num_onboard_devices_; i < position; ++i) {
234 if (!drivers_[i].has_value()) {
235 return Result<void>(
236 ErrorCode::INVALID_STATE); // Cannot skip positions in daisy chain
237 }
238 }
239
240 // Create device instance at specified position
241 drivers_[position] =
242 std::make_optional<TMC51x0<CommType>>(comm_, f_clk_, position);
243 num_active_devices_++;
244
245 // Update chain length on SpiCommInterface
246 UpdateChainLength();
247
248 return Result<void>();
249 }

267 Result<void> RemoveDevice(uint8_t position) noexcept {
268 // Cannot remove onboard devices
269 if (position < num_onboard_devices_) {
270 return Result<void>(ErrorCode::INVALID_VALUE);
271 }
272
273 // Validate position
274 if (position >= MaxDevices) {
275 return Result<void>(ErrorCode::INVALID_VALUE);
276 }
277
278 // Check if slot is actually active
279 if (!drivers_[position].has_value()) {
280 return Result<void>(ErrorCode::INVALID_STATE);
281 }
282
283 // Enforce sequential removal: can only remove from the end
284 // Check if there are any devices after this position
285 for (size_t i = position + 1; i < MaxDevices; ++i) {
286 if (drivers_[i].has_value()) {
287 return Result<void>(
288 ErrorCode::INVALID_STATE); // Cannot remove device in the middle of
289 // the chain
290 }
291 }
292
293 // Remove device
294 drivers_[position] = std::nullopt;
295 num_active_devices_--;
296
297 // Update chain length on SpiCommInterface
298 UpdateChainLength();
299
300 return Result<void>();
301 }

352 void UpdateChainLength() noexcept {
353 // Find the highest active device position
354 uint8_t max_position = 0;
355 for (size_t i = 0; i < MaxDevices; ++i) {
356 if (drivers_[i].has_value()) {
357 max_position = static_cast<uint8_t>(i);
358 }
359 }
360
361 // Total chain length is max_position + 1 (0-indexed to 1-indexed)
362 uint8_t total_length = max_position + 1;
363 comm_.SetDaisyChainLength(total_length);
364 }
