31// Implementation of chip communication mode control methods
32template <typename CommType>
33bool TMC5160<CommType>::SetChipCommMode(ChipCommMode mode) noexcept {
34 const char* mode_name = (mode == ChipCommMode::SPI_INTERNAL_RAMP) ? "SPI_INTERNAL_RAMP" :
35 (mode == ChipCommMode::SPI_EXTERNAL_STEPDIR) ? "SPI_EXTERNAL_STEPDIR" :
36 (mode == ChipCommMode::UART_INTERNAL_RAMP) ? "UART_INTERNAL_RAMP" : "UNKNOWN";
37 TMC5160_LOG_DEBUG(comm_, 2, "TMC5160", "SetChipCommMode(%s)", mode_name);
38
39 // Map mode to SPI_MODE and SD_MODE pin states
40 GpioSignal spi_mode_signal, sd_mode_signal;
41
42 switch (mode) {
43 case ChipCommMode::SPI_INTERNAL_RAMP:
44 spi_mode_signal = GpioSignal::ACTIVE; // HIGH
45 sd_mode_signal = GpioSignal::INACTIVE; // LOW
46 break;
47 case ChipCommMode::SPI_EXTERNAL_STEPDIR:
48 spi_mode_signal = GpioSignal::ACTIVE; // HIGH
49 sd_mode_signal = GpioSignal::ACTIVE; // HIGH
50 break;
51 case ChipCommMode::UART_INTERNAL_RAMP:
52 spi_mode_signal = GpioSignal::INACTIVE; // LOW
53 sd_mode_signal = GpioSignal::INACTIVE; // LOW
54 break;
55 default:
56 return false;
57 }
58
59 // Set SPI_MODE pin
60 if (!this->comm_.GpioSet(TMC5160CtrlPin::SPI_MODE, spi_mode_signal)) {
61 return false;
62 }
63
64 // Set SD_MODE pin
65 if (!this->comm_.GpioSet(TMC5160CtrlPin::SD_MODE, sd_mode_signal)) {
66 return false;
67 }
68
69 return true;
70}

72template <typename CommType>
73bool TMC5160<CommType>::GetChipCommMode(ChipCommMode& mode) const noexcept {
74 GpioSignal spi_mode_signal, sd_mode_signal;
75
76 // Read SPI_MODE pin
77 if (!this->comm_.GpioRead(TMC5160CtrlPin::SPI_MODE, spi_mode_signal)) {
78 return false;
79 }
80
81 // Read SD_MODE pin
82 if (!this->comm_.GpioRead(TMC5160CtrlPin::SD_MODE, sd_mode_signal)) {
83 return false;
84 }
85
86 // Determine mode from pin states
87 if (spi_mode_signal == GpioSignal::ACTIVE) {
88 // SPI_MODE = HIGH
89 if (sd_mode_signal == GpioSignal::ACTIVE) {
90 mode = ChipCommMode::SPI_EXTERNAL_STEPDIR;
91 } else {
92 mode = ChipCommMode::SPI_INTERNAL_RAMP;
93 }
94 } else {
95 // SPI_MODE = LOW (UART mode)
96 mode = ChipCommMode::UART_INTERNAL_RAMP;
97 }
98
99 return true;
100}

114// Implementation of unit conversion helpers
115template <typename CommType>
116float TMC5160<CommType>::convertSpeedToSteps(float value, Unit unit) const noexcept {
117 if (value == 0.0f) return 0.0f;
118
119 // Calculate effective steps per revolution (full steps * microsteps * gear ratio)
120 float effective_steps_per_rev = static_cast<float>(motor_spec_.steps_per_rev) *
121 static_cast<float>(current_microsteps_) *
122 mechanical_system_.gear_ratio;
123
124 switch (unit) {
125 case Unit::Steps:
126 return value;
127 case Unit::RPM:
128 // RPM * steps_per_rev / 60
129 return (value * effective_steps_per_rev) / 60.0f;
130 case Unit::Rad:
131 // rad/s * steps_per_rev / (2*PI)
132 return (value * effective_steps_per_rev) / (2.0f * 3.14159265359f);
133 case Unit::Deg:
134 // deg/s * steps_per_rev / 360
135 return (value * effective_steps_per_rev) / 360.0f;
136 case Unit::Mm:
137 if (mechanical_system_.system_type == MechanicalSystemType::LeadScrew && mechanical_system_.lead_screw_pitch_mm > 0.0f) {
138 // mm/s / pitch * steps_per_rev
139 return (value / mechanical_system_.lead_screw_pitch_mm) * effective_steps_per_rev;
140 } else if (mechanical_system_.system_type == MechanicalSystemType::BeltDrive && mechanical_system_.belt_pitch_mm > 0.0f && mechanical_system_.belt_pulley_teeth > 0) {
141 // mm/s / (pitch * teeth) * steps_per_rev
142 float mm_per_rev = mechanical_system_.belt_pitch_mm * static_cast<float>(mechanical_system_.belt_pulley_teeth);
143 return (value / mm_per_rev) * effective_steps_per_rev;
144 }
145 return 0.0f; // Invalid config for mm
146 default:
147 return value;
148 }

228float TMC5160<CommType>::convertSpeedToUnit(float steps_per_sec, Unit unit) const noexcept {
229 // Logic similar to steps->unit but scaling for time if needed
230 // Steps/s -> RPM: (Steps/s / Steps/rev) * 60
231
232 float effective_steps_per_rev = static_cast<float>(motor_spec_.steps_per_rev) *
233 static_cast<float>(current_microsteps_) *
234 mechanical_system_.gear_ratio;

240 return steps_per_sec;
241 case Unit::RPM:
242 return (steps_per_sec / effective_steps_per_rev) * 60.0f;
243 case Unit::Rad:
244 return (steps_per_sec / effective_steps_per_rev) * (2.0f * 3.14159265359f);
245 case Unit::Deg:

247 case Unit::Mm:

249 return (steps_per_sec / effective_steps_per_rev) * mechanical_system_.lead_screw_pitch_mm;
250 } else if (mechanical_system_.system_type == MechanicalSystemType::BeltDrive) {
251 float mm_per_rev = mechanical_system_.belt_pitch_mm * static_cast<float>(mechanical_system_.belt_pulley_teeth);
252 return (steps_per_sec / effective_steps_per_rev) * mm_per_rev;
253 }
254 return 0.0f;
255 default:
256 return steps_per_sec;

260template <typename CommType>
261bool TMC5160<CommType>::Initialize(const DriverConfig& config) noexcept {
262 TMC5160_LOG_DEBUG(comm_, 2, "TMC5160", "Initialize(toff=%u, mres=%u)",
263 config.chopper.toff, config.chopper.mres);
264
265 // Store physical configuration
266 motor_spec_ = config.motor_spec;
267 mechanical_system_ = config.mechanical;
268
269 // Calculate initial microsteps from config
270 // mres: 0=256, 1=128, ... 8=fullstep. microsteps = 256 >> mres
271 uint8_t mres = constrain<uint8_t>(config.chopper.mres, 0U, 8U);
272 current_microsteps_ = 256U >> mres;
273
274 // Always calculate motor current settings from physical parameters
275 if (motor_spec_.sense_resistor_mohm > 0 && motor_spec_.supply_voltage_mv > 0) {
276 uint8_t calc_irun = 0;
277 uint8_t calc_ihold = 0;
278 uint16_t calc_scaler = 0;
279
280 uint16_t run_current = motor_spec_.run_current_ma;
281 if (run_current == 0) {
282 run_current = motor_spec_.rated_current_ma;
283 }
284
285 if (CalculateMotorCurrent(motor_spec_, motor_spec_.sense_resistor_mohm,
286 motor_spec_.supply_voltage_mv,
287 run_current, motor_spec_.hold_current_ma,
288 calc_irun, calc_ihold, calc_scaler)) {
289 // Apply percentage adjustments before constraining
290 // GLOBAL_SCALER adjustment
291 if (motor_spec_.scaler_adjustment_percent != 0.0f) {
292 float adjustment_factor = 1.0f + (motor_spec_.scaler_adjustment_percent / 100.0f);
293 calc_scaler = static_cast<uint16_t>(std::round(static_cast<float>(calc_scaler) * adjustment_factor));
294 }
295
296 // IRUN adjustment
297 if (motor_spec_.irun_adjustment_percent != 0.0f) {
298 float adjustment_factor = 1.0f + (motor_spec_.irun_adjustment_percent / 100.0f);
299 calc_irun = static_cast<uint8_t>(std::round(static_cast<float>(calc_irun) * adjustment_factor));
300 }
301
302 // IHOLD adjustment
303 if (motor_spec_.ihold_adjustment_percent != 0.0f) {
304 float adjustment_factor = 1.0f + (motor_spec_.ihold_adjustment_percent / 100.0f);
305 calc_ihold = static_cast<uint8_t>(std::round(static_cast<float>(calc_ihold) * adjustment_factor));
306 }
307
308 // Constrain to valid ranges after adjustments
309 calc_scaler = constrain<uint16_t>(calc_scaler, 32U, 256U);
310 calc_irun = constrain<uint8_t>(calc_irun, 0U, 31U);
311 calc_ihold = constrain<uint8_t>(calc_ihold, 0U, 31U);
312
313 // Ensure IHOLD < IRUN
314 if (calc_ihold >= calc_irun) {
315 calc_ihold = (calc_irun > 0) ? (calc_irun - 1) : 0;
316 }
317
318 // Store calculated values internally (not in motor_spec_)
319 calculated_irun_ = calc_irun;
320 calculated_ihold_ = calc_ihold;
321 calculated_global_scaler_ = calc_scaler;
322
323 TMC5160_LOG_DEBUG(comm_, 1, "TMC5160", "Calculated: IRUN=%u, IHOLD=%u, GLOBAL_SCALER=%u (adjustments: scaler=%.1f%%, irun=%.1f%%, ihold=%.1f%%)",
324 calc_irun, calc_ihold, calc_scaler,
325 motor_spec_.scaler_adjustment_percent,
326 motor_spec_.irun_adjustment_percent,
327 motor_spec_.ihold_adjustment_percent);
328
329 // Validate StealthChop lower limit if resistance is available
330 if (motor_spec_.winding_resistance_mohm > 0) {
331 uint16_t lower_limit = CalculateStealthChopLowerLimit(motor_spec_,
332 motor_spec_.supply_voltage_mv,
333 config.chopper.tbl,
334 config.stealthchop.pwm_freq,
335 f_clk_);
336 if (lower_limit > 0) {
337 float run_current_a = static_cast<float>(run_current) / 1000.0f;
338 float lower_limit_a = static_cast<float>(lower_limit) / 1000.0f;
339 if (run_current_a < lower_limit_a * 1.1f) {
340 TMC5160_LOG_DEBUG(comm_, 0, "TMC5160",
341 "WARNING: Run current (%.1fmA) may be below StealthChop lower limit (%.1fmA)",
342 run_current, lower_limit);
343 }
344 }
345 }
346 } else {
347 TMC5160_LOG_DEBUG(comm_, 0, "TMC5160", "Failed to calculate motor current settings");
348 return false;
349 }
350 } else {
351 TMC5160_LOG_DEBUG(comm_, 0, "TMC5160",
352 "Cannot calculate: sense_resistor_mohm=%u, supply_voltage_mv=%u",
353 motor_spec_.sense_resistor_mohm, motor_spec_.supply_voltage_mv);
354 return false;
355 }
356
357 // Clear reset and error flags
358 GSTAT_Register gstat{};
359 gstat.bits.reset = true;
360 gstat.bits.drv_err = true;
361 gstat.bits.uv_cp = true;
362 if (!this->comm_.WriteRegister(Registers::GSTAT, gstat.value, this->GetCommAddress())) {
363 return false;
364 }
365
366 // Configure power stage (calculate register values from user-friendly parameters)
367 DRV_CONF_Register drv_conf{};
368
369 // Calculate DRVSTRENGTH from MOSFET Miller charge
370 float miller = config.power_stage.mosfet_miller_charge_nc;
371 if (miller < 0.0f) {
372 miller = 0.0f; // Constrain to valid range
373 }
374 if (miller == 0.0f) {
375 // Auto-calculate from default (small MOSFET, lowest driver strength)
376 miller = 10.0f;
377 }
378
379 uint8_t drv_strength = 0;
380 // Auto-calculate from Miller charge (datasheet table 3.3)
381 // Use lowest gate driver strength giving favorable switching slopes
382 if (miller < 10.0f) {
383 drv_strength = 0;
384 } else if (miller < 20.0f) {
385 drv_strength = 0; // Can use 0 or 1, prefer 0 (lowest) for best switching quality
386 } else if (miller < 40.0f) {
387 drv_strength = 1; // Can use 1 or 2, prefer 1 (lower)
388 } else if (miller < 60.0f) {
389 drv_strength = 2; // Can use 2 or 3, prefer 2 (lower)
390 } else {
391 drv_strength = 3; // Maximum for very large MOSFETs
392 }
393 drv_conf.bits.drvstrength = constrain<uint8_t>(drv_strength, 0U, 3U);
394
395 // Calculate BBMTIME and BBMCLKS from nanoseconds
396 uint8_t bbm_time_reg = 0;
397 uint8_t bbm_clks_reg = 0;
398
399 // Auto-calculate from bbm_time_ns
400 uint32_t bbm_ns = config.power_stage.bbm_time_ns;
401
402 if (bbm_ns == 0) {
403 // Auto-calculate from MOSFET: use minimum safe time (100ns for small MOSFETs, lowest setting)
404 // For larger MOSFETs with higher Miller charge, may need more time
405 if (miller > 40.0f) {
406 bbm_ns = 200; // Larger MOSFETs need more BBM time
407 } else {
408 bbm_ns = 100; // Default minimum (lowest, BBMTIME=0)
409 }
410 }
411
412 // Ensure minimum 100ns (lowest possible setting)
413 if (bbm_ns < 100) {
414 bbm_ns = 100;
415 }
416
417 // BBMTIME calculation per datasheet:
418 // Formula: time[ns] ≈ 100ns * 32 / (32 - BBMTIME)
419 // Rearranged: BBMTIME = 32 - (100ns * 32 / time_ns)
420 // BBMTIME range: 0-24 (bits 4..0)
421 // BBMTIME=0 gives 100ns (shortest, reset default)
422 // BBMTIME=16 gives 200ns
423 // BBMTIME=24 gives ~375ns (longest, datasheet says 375ns, formula gives ~400ns)
424 // >24 not recommended, use BBMCLKS instead
425 //
426 // Datasheet hint: Choose the lowest setting safely covering the switching event.
427 // Add roughly 30% of reserve to cover production stray.
428 if (bbm_ns <= 200) {
429 // Use BBMTIME (0-24, corresponds to 100ns to ~375ns)
430 // Add 30% headroom as recommended by datasheet
431 float bbm_ns_with_headroom = static_cast<float>(bbm_ns) * 1.3f;
432 // Calculate BBMTIME: BBMTIME = 32 - (100 * 32 / time_ns)
433 float bbm_time_float = 32.0f - (100.0f * 32.0f / bbm_ns_with_headroom);
434 bbm_time_reg = static_cast<uint8_t>(std::round(bbm_time_float));
435 bbm_time_reg = constrain<uint8_t>(bbm_time_reg, 0U, 24U);
436 bbm_clks_reg = 0; // BBMTIME takes precedence, BBMCLKS=0 (off)
437 } else {
438 // Use BBMCLKS for longer times (>200ns)
439 // BBMCLKS range: 0-15 (bits 11..8), typ. 83ns per clock cycle at 12MHz
440 // The longer setting rules (BBMTIME vs. BBMCLKS)
441 // Add 30% headroom as recommended by datasheet
442 float bbm_ns_with_headroom = static_cast<float>(bbm_ns) * 1.3f;
443 // Calculate clock cycles: cycles = (time_ns * f_clk) / 1e9
444 // At 12MHz: 1 cycle = 83.3ns
445 uint32_t clock_cycles = static_cast<uint32_t>((bbm_ns_with_headroom * static_cast<float>(f_clk_)) / 1000000000.0f);
446 bbm_clks_reg = constrain<uint8_t>(static_cast<uint8_t>(clock_cycles), 0U, 15U);
447 if (bbm_clks_reg == 0 && bbm_ns > 200) {
448 bbm_clks_reg = 1; // Minimum 1 clock cycle if >200ns requested
449 }
450 bbm_time_reg = 0; // BBMCLKS takes precedence when set (longer setting rules)
451 }
452
453 drv_conf.bits.bbmtime = bbm_time_reg;
454 drv_conf.bits.bbmclks = bbm_clks_reg;
455 drv_conf.bits.otselect = static_cast<uint8_t>(config.power_stage.over_temp_protection);
456 drv_conf.bits.filt_isense = static_cast<uint8_t>(config.power_stage.sense_filter);
457
458 TMC5160_LOG_DEBUG(comm_, 2, "TMC5160", "Power stage: DRVSTRENGTH=%u (from %.1fnC), BBMTIME=%u, BBMCLKS=%u, FILT_ISENSE=%u",
459 drv_conf.bits.drvstrength, miller,
460 drv_conf.bits.bbmtime, drv_conf.bits.bbmclks, drv_conf.bits.filt_isense);
461
462 if (!this->comm_.WriteRegister(Registers::DRV_CONF, drv_conf.value, this->GetCommAddress())) {
463 return false;
464 }
465
466 // Configure global scaler (use calculated value)
467 TMC5160_LOG_DEBUG(comm_, 3, "TMC5160", "Initialize: Setting GLOBAL_SCALER=%u", calculated_global_scaler_);
468 if (!this->comm_.WriteRegister(Registers::GLOBAL_SCALER, calculated_global_scaler_, this->GetCommAddress())) {
469 return false;
470 }
471
472 // Configure motor current (use calculated values)
473 IHOLD_IRUN_Register iholdrun{};
474 iholdrun.bits.ihold = calculated_ihold_;
475 iholdrun.bits.irun = calculated_irun_;
476 iholdrun.bits.iholddelay = 7;
477 TMC5160_LOG_DEBUG(comm_, 3, "TMC5160", "Initialize: Setting IHOLD_IRUN(irun=%u, ihold=%u, iholddelay=7)",
478 iholdrun.bits.irun, iholdrun.bits.ihold);
479 if (!this->comm_.WriteRegister(Registers::IHOLD_IRUN, iholdrun.value, this->GetCommAddress())) {
480 return false;
481 }
482
483 // Configure short protection (convert user-friendly values to register values)
484 SHORT_CONF_Register short_conf{};
485
486 // Calculate S2VS_LEVEL from voltage threshold
487 uint16_t s2vs_voltage = config.power_stage.s2vs_voltage_mv;
488 if (s2vs_voltage == 0) {
489 s2vs_voltage = 625; // Default recommended: 625mV (S2VS_LEVEL=6)
490 }
491 uint8_t s2vs_level = CalculateS2VSLevel(s2vs_voltage);
492 if (s2vs_level == 0 || s2vs_level < 4) {
493 s2vs_level = 6; // Fallback to default
494 }
495 short_conf.bits.s2vs_level = constrain<uint8_t>(s2vs_level, 4U, 15U);
496
497 // Calculate S2G_LEVEL from voltage threshold (consider supply voltage for VS>52V check)
498 uint16_t s2g_voltage = config.power_stage.s2g_voltage_mv;
499 if (s2g_voltage == 0) {
500 s2g_voltage = 625; // Default recommended: 625mV (S2G_LEVEL=6)
501 }
502 uint8_t s2g_level = CalculateS2GLevel(s2g_voltage, motor_spec_.supply_voltage_mv);
503 if (s2g_level == 0 || s2g_level < 2) {
504 s2g_level = 6; // Fallback to default
505 }
506 short_conf.bits.s2g_level = constrain<uint8_t>(s2g_level, 2U, 15U);
507
508 short_conf.bits.shortfilter =
509 constrain<decltype(config.power_stage.shortfilter)>(config.power_stage.shortfilter, 0U, 3U);
510
511 // Calculate shortdelay from detection delay time
512 uint8_t delay_us_x10 = config.power_stage.short_detection_delay_us_x10;
513 if (delay_us_x10 == 0) {
514 delay_us_x10 = 8; // Default recommended: 0.8µs (typical 0.85µs, shortdelay=0)
515 }
516 uint8_t shortdelay = CalculateShortDelay(delay_us_x10);
517 short_conf.bits.shortdelay = constrain<uint8_t>(shortdelay, 0U, 1U);
518
519 if (!this->comm_.WriteRegister(Registers::SHORT_CONF, short_conf.value, this->GetCommAddress())) {
520 return false;
521 }

523 // Configure chopper
524 CHOPCONF_Register chopconf{};
525 chopconf.bits.toff = constrain<decltype(config.chopper.toff)>(config.chopper.toff, 0U, 15U);
526 chopconf.bits.hstrt_tfd = constrain<decltype(config.chopper.hstrt)>(config.chopper.hstrt, 0U, 7U);
527 chopconf.bits.hend_offset = constrain<decltype(config.chopper.hend)>(config.chopper.hend, 0U, 15U);
528 chopconf.bits.tbl = constrain<decltype(config.chopper.tbl)>(config.chopper.tbl, 0U, 3U);
529 // Note: Bit 17 (vsense) is reserved per datasheet, ignoring config.chopper.vsense
530 chopconf.bits.mres = constrain<decltype(config.chopper.mres)>(config.chopper.mres, 0U, 8U);
531 chopconf.bits.intpol = config.chopper.intpol ? 1 : 0;
532 chopconf.bits.dedge = config.chopper.dedge ? 1 : 0;
533 chopconf.bits.chm = config.chopper.chm ? 1 : 0;
534 if (!this->comm_.WriteRegister(Registers::CHOPCONF, chopconf.value, this->GetCommAddress())) {
535 return false;

621template <typename CommType>
622int32_t TMC5160<CommType>::speedToInternal(float speed_hz) const noexcept {
623 // Datasheet formula: v[Hz] = v[5160] * (f_CLK[Hz]/2 / 2^23)
624 // Rearranged: v[5160] = v[Hz] * 2^24 / f_CLK
625 // Where v[Hz] is in μsteps/s (microsteps per second)
626 // Input speed_hz is in steps/s, so we multiply by microstep count (256) to convert to μsteps/s
627 // Final: v[5160] = (speed_hz * 256) * 2^24 / f_CLK
628 if (speed_hz == 0.0F) {
629 return 0;
630 }
631 float internal = (speed_hz * static_cast<float>(1UL << 24)) / static_cast<float>(f_clk_);
632 internal *= static_cast<float>(Microsteps::USTEP_COUNT);
633 return static_cast<int32_t>(internal);
634}
635
636template <typename CommType>
637float TMC5160<CommType>::speedFromInternal(int32_t speed_internal) const noexcept {
638 // Datasheet formula: v[Hz] = v[5160] * (f_CLK[Hz]/2 / 2^23)
639 // Where v[Hz] is in μsteps/s (microsteps per second)
640 // Output is in steps/s, so we divide by microstep count (256) to convert from μsteps/s
641 // Final: v[steps/s] = (v[5160] * f_CLK / 2^24) / 256
642 if (speed_internal == 0) {
643 return 0.0F;
644 }
645 float speed_hz = static_cast<float>(speed_internal) * static_cast<float>(f_clk_) / static_cast<float>(1UL << 24);
646 speed_hz /= static_cast<float>(Microsteps::USTEP_COUNT);
647 return speed_hz;
648}

650template <typename CommType>
651int32_t TMC5160<CommType>::accelToInternal(float accel_hz) const noexcept {
652 // Datasheet formula: a[Hz/s] = a[5160] * f_CLK[Hz]^2 / (512*256) / 2^24
653 // Rearranged: a[5160] = a[Hz/s] * (512*256) * 2^24 / f_CLK^2
654 // Where a[Hz/s] is in μsteps/s² (microsteps per second squared)
655 // Input accel_hz is in steps/s², so we multiply by microstep count (256) to convert to μsteps/s²
656 // Final: a[5160] = (accel_hz * 256) * (512*256) * 2^24 / f_CLK^2
657 if (accel_hz == 0.0F) {
658 return 0;
659 }
660 float internal = accel_hz * 512.0F * 256.0F * static_cast<float>(1UL << 24) /
661 (static_cast<float>(f_clk_) * static_cast<float>(f_clk_));
662 internal *= static_cast<float>(Microsteps::USTEP_COUNT);
663 return static_cast<int32_t>(internal);
664}

666template <typename CommType>
667int32_t TMC5160<CommType>::thresholdSpeedToTstep(float speed_hz) const noexcept {
668 // Datasheet formula: TSTEP = f_CLK / f256STEP = f_CLK / (fSTEP*256/USC)
669 // Where fSTEP is in μsteps/s, USC is microstep count (normally 256)
670 // For USC=256: TSTEP = f_CLK / fSTEP
671 // Input speed_hz is in steps/s, so fSTEP = speed_hz * 256 (convert to μsteps/s)
672 // Final: TSTEP = f_CLK / (speed_hz * 256)
673 // TSTEP is 20-bit unsigned (max value 0xFFFFF = 1048575)
674 if (speed_hz == 0.0F) {
675 return 0;
676 }
677 float tstep = static_cast<float>(f_clk_) / (speed_hz * 256.0F);
678 tstep = std::max(0.0F, std::min(1048575.0F, tstep));
679 return static_cast<int32_t>(tstep);
680}

1004 // Read-Modify-Write to preserve iholddelay

1008 IHOLD_IRUN_Register iholdrun{};

1344template <typename CommType>
1345bool TMC5160<CommType>::MotorControl::SetupMotorFromSpec(const MotorSpec& motor_spec,
1346 const MechanicalSystem* mechanical_system) noexcept {
1347 // Calculate global scaler based on rated current
1348 // Typical calculation: global_scaler = (rated_current_ma * 32) / (irun * sense_resistor_current)
1349 // For simplicity, we'll use a basic calculation
1350 uint16_t global_scaler = 32;
1351 if (motor_spec.rated_current_ma > 0) {
1352 // Basic calculation: assume 1.5A max current, scale accordingly
1353 global_scaler = static_cast<uint16_t>(std::min(256U, std::max(32U, (motor_spec.rated_current_ma * 32U) / 1500U)));
1354 }
1355
1356 // Set global scaler
1357 if (!SetGlobalScaler(global_scaler)) {
1358 return false;
1359 }
1360
1361 // Calculate irun and ihold from rated current
1362 // irun should be between 16-31 for best performance
1363 // We'll use 80% of rated current for irun, 30% for ihold
1364 uint8_t irun = 16;
1365 uint8_t ihold = 0;
1366 if (motor_spec.rated_current_ma > 0 && global_scaler > 0) {
1367 // Calculate irun: target current = (irun/32) * (global_scaler/32) * sense_resistor_current
1368 // Simplified: irun = (target_current * 32) / (global_scaler * sense_resistor_current / 32)
1369 // Assuming sense resistor gives ~1.5A at irun=31, global_scaler=32
1370 float target_run_current = static_cast<float>(motor_spec.rated_current_ma) * 0.8F;
1371 auto irun_calc =
1372 static_cast<uint32_t>((target_run_current * 32.0F) / (static_cast<float>(global_scaler) * 0.046875F));
1373 irun = static_cast<uint8_t>(std::min(static_cast<uint32_t>(31U), std::max(static_cast<uint32_t>(16U), irun_calc)));
1374 float target_hold_current = static_cast<float>(motor_spec.rated_current_ma) * 0.3F;
1375 auto ihold_calc =
1376 static_cast<uint32_t>((target_hold_current * 32.0F) / (static_cast<float>(global_scaler) * 0.046875F));
1377 ihold = static_cast<uint8_t>(std::min(static_cast<uint32_t>(31U), ihold_calc));
1378 }
1379
1380 // Set motor current
1381 if (!SetCurrent(irun, ihold)) {
1382 return false;
1383 }
1384
1385 // Configure chopper based on inductance if available
1386 if (motor_spec.winding_inductance_uh > 0) {
1387 ChopperConfig chop_config{};
1388 // Higher inductance may need different settings
1389 // This is a simplified heuristic
1390 if (motor_spec.winding_inductance_uh > 3000) {
1391 chop_config.tbl = 3; // Longer blank time for high inductance
1392 }
1393 ConfigureChopper(chop_config);
1394 }
1395
1396 // Note: mechanical_system is stored for unit conversions but not used here
1397 // as it's used by the unit conversion functions, not motor setup
1398
1399 return true;
1400}

1428template <typename CommType>
1429bool TMC5160<CommType>::Encoder::SetResolution(int32_t motor_steps, int32_t enc_resolution, bool inverted) noexcept {
1430 // Calculate factor: (motor_steps * microsteps) / enc_resolution
1431 float factor = static_cast<float>(motor_steps * Microsteps::USTEP_COUNT) / static_cast<float>(enc_resolution);
1432
1433 // Check if binary prescaler gives exact match
1434 auto enc_const_binary = static_cast<int32_t>(factor * 65536.0F);
1435 if (enc_const_binary * enc_resolution == motor_steps * Microsteps::USTEP_COUNT * 65536) {
1436 // Use binary mode
1437 uint32_t encmode_value = 0;
1438 if (!driver_.comm_.ReadRegister(Registers::ENCMODE, encmode_value)) {
1439 return false;
1440 }
1441 ENCMODE_Register encmode{};
1442 encmode.value = encmode_value;
1443 encmode.bits.enc_sel_decimal = false;
1444 if (!driver_.comm_.WriteRegister(Registers::ENCMODE, encmode.value)) {
1445 return false;
1446 }
1447 if (inverted) {
1448 enc_const_binary = -enc_const_binary;
1449 }
1450 return driver_.comm_.WriteRegister(Registers::ENC_CONST, static_cast<uint32_t>(enc_const_binary));
1451 }
1452 // Use decimal mode
1453 uint32_t encmode_value = 0;

1650template <typename CommType>
1651bool TMC5160<CommType>::Diagnostics::PerformSensorlessHoming(bool direction, int8_t stall_threshold, float search_speed,
1652 int32_t& final_position, uint32_t timeout_ms) noexcept {
1653 // Configure StallGuard2 for homing
1654 StallGuardConfig sg_config{};
1655 sg_config.sgt = stall_threshold;
1656 sg_config.sfilt = true; // Enable filter for stable readings
1657 if (!ConfigureStallGuard(sg_config)) {
1658 return false;
1659 }
1660
1661 // Enable StallGuard2 stop in SW_MODE
1662 uint32_t sw_mode_value = 0;
1663 if (!driver_.comm_.ReadRegister(Registers::SW_MODE, sw_mode_value)) {
1664 return false;
1665 }
1666 SW_MODE_Register sw_mode{};
1667 sw_mode.value = sw_mode_value;
1668 sw_mode.bits.sg_stop = true; // Enable stop on stall
1669 sw_mode.bits.en_softstop = false; // Use hard stop for precise homing (per datasheet 12.4)
1670 if (!driver_.comm_.WriteRegister(Registers::SW_MODE, sw_mode.value)) {
1671 return false;
1672 }
1673
1674 // Clear any existing stall flags
1675 uint32_t ramp_stat_dummy = 0;
1676 driver_.comm_.ReadRegister(Registers::RAMP_STAT, ramp_stat_dummy);
1677 // Write back 1 to event_stop_sg to clear it
1678 RAMP_STAT_Register clear_stat{};
1679 clear_stat.bits.event_stop_sg = 1;
1680 driver_.comm_.WriteRegister(Registers::RAMP_STAT, clear_stat.value);
1681
1682 // Set velocity mode and start movement
1683 RampMode mode = direction ? RampMode::VELOCITY_POS : RampMode::VELOCITY_NEG;
1684 if (!driver_.rampControl.SetRampMode(mode)) {
1685 return false;
1686 }
1687
1688 // CRITICAL: Set acceleration/deceleration BEFORE setting speeds
1689 // Velocity mode requires AMAX > 0 to actually accelerate from VSTART to VMAX
1690 // Use reasonable acceleration: reach VMAX in ~0.1 seconds
1691 float acceleration = std::max(search_speed * 10.0f, 50000.0f); // At least 50k steps/s²
1692 if (!driver_.rampControl.SetAcceleration(acceleration)) {
1693 return false;
1694 }
1695 if (!driver_.rampControl.SetDeceleration(acceleration)) {
1696 return false;
1697 }
1698
1699 // Set VSTART > 0 to actually start motion in velocity mode
1700 // VSTART should be reasonable but less than VMAX - use 10% of search speed or minimum 1000 steps/s
1701 float vstart_speed = std::max(search_speed * 0.1f, 1000.0f);
1702 if (!driver_.rampControl.SetRampSpeeds(vstart_speed, 100.0f, 0.0f, Unit::Steps)) {
1703 return false;
1704 }
1705 if (!driver_.rampControl.SetMaxSpeed(search_speed)) {
1706 return false;
1707 }
1708
1709 // Wait for stall event
1710 bool stalled = false;
1711 // Simple polling loop with crude timeout using rough cycle estimates
1712 // Assuming ~10us per register read interaction on SPI
1713 uint32_t loops = timeout_ms * 100;
1714
1715 for (uint32_t i = 0; i < loops; i++) {
1716 uint32_t ramp_stat = 0;
1717 if (driver_.comm_.ReadRegister(Registers::RAMP_STAT, ramp_stat)) {
1718 RAMP_STAT_Register status{};
1719 status.value = ramp_stat;
1720
1721 // Check for stall stop event or velocity zero (stop executed)
1722 if (status.bits.event_stop_sg || (status.bits.vzero && status.bits.status_sg)) {
1723 stalled = true;
1724 break;
1725 }
1726 }
1727 // Small delay to relieve bus
1728 // driver_.comm_.DelayUs(100); // Requires DelayUs in Comm interface, assuming it exists
1729 }
1730
1731 // Stop motor (ensure VMAX=0)
1732 driver_.rampControl.Stop();
1733
1734 // Disable StallGuard2 stop to prevent accidental stops later
1735 sw_mode.bits.sg_stop = false;
1736 driver_.comm_.WriteRegister(Registers::SW_MODE, sw_mode.value);
1737
1738 // Read final position
1739 final_position = driver_.rampControl.GetCurrentPosition();
1740
1741 // Clear the stall event flag again to be clean
1742 driver_.comm_.WriteRegister(Registers::RAMP_STAT, clear_stat.value);
1743
1744 return stalled;
1745}

1747template <typename CommType>
1748bool TMC5160<CommType>::Diagnostics::PerformSwitchHoming(bool direction, float search_speed, float switch_speed,
1749 int32_t& final_position, bool use_left_switch, uint32_t timeout_ms) noexcept {
1750 // 1. Activate position latching and motor stop upon switch event
1751 SW_MODE_Register sw_mode{};
1752 // Read current SW_MODE to preserve other settings (like softstop)
1753 uint32_t sw_mode_val = 0;
1754 if (!driver_.comm_.ReadRegister(Registers::SW_MODE, sw_mode_val)) {
1755 return false;
1756 }
1757 sw_mode.value = sw_mode_val;
1758
1759 // Configure latching and stop enable based on switch choice
1760 if (use_left_switch) {
1761 sw_mode.bits.latch_l_active = true;
1762 sw_mode.bits.stop_l_enable = true;
1763 } else {
1764 sw_mode.bits.latch_r_active = true;
1765 sw_mode.bits.stop_r_enable = true;
1766 }
1767 // Use hard stop for precise homing (soft stop can overshoot) - datasheet recommends hard stop for StallGuard,
1768 // but for switches hard stop ensures we don't crash if switch is a hard limit.
1769 // However, datasheet 12.4 says "Or motor can be softly decelerated...".
1770 // Let's assume hard stop for safety during homing search.
1771 sw_mode.bits.en_softstop = false;
1772
1773 if (!driver_.comm_.WriteRegister(Registers::SW_MODE, sw_mode.value)) {
1774 return false;
1775 }
1776
1777 // 2. Start motion ramp into direction of switch
1778 RampMode mode = direction ? RampMode::VELOCITY_POS : RampMode::VELOCITY_NEG;
1779 if (!driver_.rampControl.SetRampMode(mode)) {
1780 return false;
1781 }
1782 if (!driver_.rampControl.SetMaxSpeed(search_speed)) {
1783 return false;
1784 }
1785
1786 // 3. Wait for switch hit (motor stops automatically)
1787 bool switch_hit = false;
1788 uint32_t loops = timeout_ms * 100;
1789 for (uint32_t i = 0; i < loops; i++) {
1790 uint32_t ramp_stat = 0;
1791 if (driver_.comm_.ReadRegister(Registers::RAMP_STAT, ramp_stat)) {
1792 RAMP_STAT_Register status{};
1793 status.value = ramp_stat;
1794
1795 // Check for specific stop event
1796 if ((use_left_switch && status.bits.event_stop_l) ||
1797 (!use_left_switch && status.bits.event_stop_r)) {
1798 switch_hit = true;
1799 break;
1800 }
1801 // Also check vzero if we missed the event flag but stopped
1802 if (status.bits.vzero && ((use_left_switch && status.bits.status_stop_l) || (!use_left_switch && status.bits.status_stop_r))) {
1803 switch_hit = true;
1804 break;
1805 }
1806 }
1807 // driver_.comm_.DelayUs(100);
1808 }
1809
1810 // 4. Stop motor command (VMAX=0) to ensure it stays stopped
1811 driver_.rampControl.Stop();
1812
1813 if (switch_hit) {
1814 // 5. Read latched position
1815 // "Latching of... XACTUAL to XLATCH upon a switch event gives a precise snapshot"
1816 // However, datasheet 12.4 "Implementing a homing procedure" step 5 says:
1817 // "Switch to hold mode... calculate difference between latched and actual... or when using hard stop XACTUAL stops exactly at home position"
1818 // Since we used hard stop, XACTUAL should be valid.
1819 // Let's read XLATCH just in case user wants it, but we return current XACTUAL as "final position".
1820 // Actually, let's just return the current position where it stopped.
1821 final_position = driver_.rampControl.GetCurrentPosition();
1822 }
1823
1824 // Disable stop function to allow moving away
1825 if (use_left_switch) {
1826 sw_mode.bits.stop_l_enable = false;
1827 } else {
1828 sw_mode.bits.stop_r_enable = false;
1829 }
1830 driver_.comm_.WriteRegister(Registers::SW_MODE, sw_mode.value);
1831
1832 return switch_hit;
1833}
