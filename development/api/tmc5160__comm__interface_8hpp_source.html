122static constexpr uint8_t calculateCrc8(const uint8_t *data,
123 size_t length) noexcept {
124 uint8_t crc = 0;
125 for (size_t i = 0; i < length; ++i) {
126 crc ^= data[i];
127 for (uint8_t j = 0; j < 8; ++j) {
128 if (crc & 0x80) {
129 crc = (crc << 1) ^ 0x07;
130 } else {
131 crc <<= 1;
132 }
133 }
134 }
135 return crc;
136}

470 bool ReadRegister(uint8_t address, uint32_t &value,
471 uint8_t csn_pin_index = 0) noexcept {
472 // Select chip if multi-chip setup
473 if (csn_pin_index > 0) {
474 SetChipSelect(csn_pin_index, true); // Assert CSN
475 }
476
477 std::array<uint8_t, 8> tx_buf{};
478 std::array<uint8_t, 8> rx_buf{};
479
480 // First transaction: Send read address, receive dummy data
481 tx_buf[0] = address & 0x7F; // Clear write bit
482 for (size_t i = 1; i < 8; ++i) {
483 tx_buf[i] = 0x00;
484 }
485
486 TMC5160_LOG_DEBUG(
487 *static_cast<Derived *>(this), 3, "SPI",
488 "Read register 0x%02X (CSN=%u): TX %02X %02X %02X %02X %02X %02X %02X %02X",
489 address, csn_pin_index, tx_buf[0], tx_buf[1], tx_buf[2], tx_buf[3],
490 tx_buf[4], tx_buf[5], tx_buf[6], tx_buf[7]);
491
492 if (!SpiTransfer(tx_buf.data(), rx_buf.data(), 8)) {
493 if (csn_pin_index > 0) {
494 SetChipSelect(csn_pin_index, false); // Deassert CSN on error
495 }
496 return false;
497 }
498
499 // Minimum CSN high time: 2*tclk + 10ns (typically ~176ns with 12MHz clock)
500 // Use 1us delay for safety
501 this->DelayUs(1);
502
503 // Second transaction: Send address again, receive actual data
504 if (!SpiTransfer(tx_buf.data(), rx_buf.data(), 8)) {
505 if (csn_pin_index > 0) {
506 SetChipSelect(csn_pin_index, false); // Deassert CSN on error
507 }
508 return false;
509 }
510
511 // Deassert CSN if multi-chip setup
512 if (csn_pin_index > 0) {
513 SetChipSelect(csn_pin_index, false);
514 }
515
516 TMC5160_LOG_DEBUG(
517 *static_cast<Derived *>(this), 3, "SPI",
518 "Read register 0x%02X: RX %02X %02X %02X %02X %02X %02X %02X %02X",
519 address, rx_buf[0], rx_buf[1], rx_buf[2], rx_buf[3], rx_buf[4],
520 rx_buf[5], rx_buf[6], rx_buf[7]);
521
522 // Extract 32-bit value from bytes 1-4 (MSB-first)
523 value = (static_cast<uint32_t>(rx_buf[1]) << 24) |
524 (static_cast<uint32_t>(rx_buf[2]) << 16) |
525 (static_cast<uint32_t>(rx_buf[3]) << 8) |
526 static_cast<uint32_t>(rx_buf[4]);
527
528 return true;
529 }

538 bool WriteRegister(uint8_t address, uint32_t value,
539 uint8_t csn_pin_index = 0) noexcept {
540 // Select chip if multi-chip setup
541 if (csn_pin_index > 0) {
542 SetChipSelect(csn_pin_index, true); // Assert CSN
543 }
544
545 std::array<uint8_t, 8> tx_buf{};
546 std::array<uint8_t, 8> rx_buf{};
547
548 // First transaction: Send write command, receive dummy/previous data
549 tx_buf[0] = (address & 0x7F) | 0x80; // Set write bit
550 tx_buf[1] = static_cast<uint8_t>((value >> 24) & 0xFF);
551 tx_buf[2] = static_cast<uint8_t>((value >> 16) & 0xFF);
552 tx_buf[3] = static_cast<uint8_t>((value >> 8) & 0xFF);
553 tx_buf[4] = static_cast<uint8_t>(value & 0xFF);
554 for (size_t i = 5; i < 8; ++i) {
555 tx_buf[i] = 0x00;
556 }
557
558 TMC5160_LOG_DEBUG(*static_cast<Derived *>(this), 3, "SPI",
559 "Write register 0x%02X = 0x%08X (CSN=%u): TX %02X %02X %02X %02X "
560 "%02X %02X %02X %02X",
561 address, value, csn_pin_index, tx_buf[0], tx_buf[1],
562 tx_buf[2], tx_buf[3], tx_buf[4], tx_buf[5], tx_buf[6],
563 tx_buf[7]);
564
565 if (!SpiTransfer(tx_buf.data(), rx_buf.data(), 8)) {
566 if (csn_pin_index > 0) {
567 SetChipSelect(csn_pin_index, false); // Deassert CSN on error
568 }
569 return false;
570 }
571
572 TMC5160_LOG_DEBUG(
573 *static_cast<Derived *>(this), 3, "SPI",
574 "Write register 0x%02X (TX1): RX %02X %02X %02X %02X %02X %02X %02X %02X",
575 address, rx_buf[0], rx_buf[1], rx_buf[2], rx_buf[3], rx_buf[4],
576 rx_buf[5], rx_buf[6], rx_buf[7]);
577
578 // Minimum CSN high time: 2*tclk + 10ns (typically ~176ns with 12MHz clock)
579 // Use 1us delay for safety
580 this->DelayUs(1);
581
582 // Second transaction: Send dummy read to receive write confirmation/status
583 // The response from the second transaction contains the status/confirmation
584 // for the write command sent in the first transaction
585 tx_buf[0] = address & 0x7F; // Read address (clear write bit)
586 for (size_t i = 1; i < 8; ++i) {
587 tx_buf[i] = 0x00;
588 }
589
590 if (!SpiTransfer(tx_buf.data(), rx_buf.data(), 8)) {
591 if (csn_pin_index > 0) {
592 SetChipSelect(csn_pin_index, false); // Deassert CSN on error
593 }
594 return false;
595 }
596
597 // Deassert CSN if multi-chip setup
598 if (csn_pin_index > 0) {
599 SetChipSelect(csn_pin_index, false);
600 }
601
602 TMC5160_LOG_DEBUG(
603 *static_cast<Derived *>(this), 3, "SPI",
604 "Write register 0x%02X (TX2): RX %02X %02X %02X %02X %02X %02X %02X %02X",
605 address, rx_buf[0], rx_buf[1], rx_buf[2], rx_buf[3], rx_buf[4],
606 rx_buf[5], rx_buf[6], rx_buf[7]);
607
608 return true;
609 }

753 bool ReadRegister(uint8_t address, uint32_t &value) noexcept {
754 std::array<uint8_t, 9> tx_buf{};
755 std::array<uint8_t, 9> rx_buf{};
756
757 // Build UART frame for read
758 tx_buf[0] = (slaveAddress_ & 0x7F) | 0x01; // Sync bit + address
759 tx_buf[1] = address & 0x7F; // Register address (read)
760 tx_buf[2] = 0x00;
761 tx_buf[3] = 0x00;
762 tx_buf[4] = 0x00;
763 tx_buf[5] = 0x00;
764 tx_buf[6] = 0x00;
765 tx_buf[7] = 0x00;
766 tx_buf[8] = calculateCrc8(tx_buf.data(), 8);
767
768 TMC5160_LOG_DEBUG(
769 *static_cast<Derived *>(this), 3, "UART",
770 "Read register 0x%02X: TX %02X %02X %02X %02X %02X %02X %02X %02X %02X",
771 address, tx_buf[0], tx_buf[1], tx_buf[2], tx_buf[3], tx_buf[4],
772 tx_buf[5], tx_buf[6], tx_buf[7], tx_buf[8]);
773
774 if (!UartSend(tx_buf.data(), 9)) {
775 return false;
776 }
777
778 if (!UartReceive(rx_buf.data(), 9)) {
779 return false;
780 }
781
782 TMC5160_LOG_DEBUG(
783 *static_cast<Derived *>(this), 3, "UART",
784 "Read register 0x%02X: RX %02X %02X %02X %02X %02X %02X %02X %02X %02X",
785 address, rx_buf[0], rx_buf[1], rx_buf[2], rx_buf[3], rx_buf[4],
786 rx_buf[5], rx_buf[6], rx_buf[7], rx_buf[8]);
787
788 // Verify CRC8
789 uint8_t calculated_crc = calculateCrc8(rx_buf.data(), 8);
790 if (calculated_crc != rx_buf[8]) {
791 TMC5160_LOG_DEBUG(*static_cast<Derived *>(this), 1, "UART",
792 "CRC8 mismatch: calculated 0x%02X, received 0x%02X",
793 calculated_crc, rx_buf[8]);
794 return false;
795 }
796
797 // Extract 32-bit value from bytes 2-5 (MSB-first)
798 value = (static_cast<uint32_t>(rx_buf[2]) << 24) |
799 (static_cast<uint32_t>(rx_buf[3]) << 16) |
800 (static_cast<uint32_t>(rx_buf[4]) << 8) |
801 static_cast<uint32_t>(rx_buf[5]);
