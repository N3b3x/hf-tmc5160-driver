518static constexpr uint8_t calculateCrc8(const uint8_t* data, size_t length) noexcept {
519 uint8_t crc = 0; // Initial value is zero per datasheet
520
521 // Process each byte LSB to MSB
522 for (size_t i = 0; i < length; ++i) {
523 uint8_t currentByte = data[i];
524
525 // Process each bit LSB to MSB (j=0 is LSB, j=7 is MSB)
526 for (uint8_t j = 0; j < 8; ++j) {
527 // Check: (CRC >> 7) XOR (currentByte & 0x01)
528 // This XORs the MSB of CRC with the LSB of currentByte
529 if (((crc >> 7) ^ (currentByte & 0x01)) != 0) {
530 crc = (crc << 1) ^ 0x07; // Polynomial 0x07 (CRC8-ATM)
531 } else {
532 crc = (crc << 1);
533 }
534 currentByte = currentByte >> 1; // Shift to next bit (LSB to MSB)
535 }
536 }
537
538 return crc;
539}

1417 detected_length = n;
1418 TMC5160_LOG_DEBUG(*static_cast<Derived*>(this), 2, "SPI",
1419 "AutoDetectChainLength: Found EXACT command pattern match at offset "
1420 "%zu (expected for n=%u), chain length = %u",
1421 loopback_offset, n, detected_length);
1422 break;
1423 }
1424 }
1425 }
1426
1427 // If exact match failed, log debug info to help diagnose
1428 if (detected_length == 0) {

1430 "AutoDetectChainLength: Exact command pattern not found in received data");

1432 "AutoDetectChainLength: Expected pattern: %02X %02X %02X %02X %02X", cmd_bytes[0], cmd_bytes[1],
1433 cmd_bytes[2], cmd_bytes[3], cmd_bytes[4]);
1434
1435 // Log first few potential loopback positions for debugging
1436 for (uint8_t n = 1; n <= 3 && n <= max_devices; ++n) {
1437 size_t loopback_offset = static_cast<size_t>(n + 1) * 5;
1438 if (loopback_offset + 4 < rx_buf.size()) {
1439 TMC5160_LOG_DEBUG(*static_cast<Derived*>(this), 3, "SPI",
1440 "AutoDetectChainLength: At offset %zu (n=%u): %02X %02X %02X %02X %02X", loopback_offset, n,
1441 rx_buf[loopback_offset], rx_buf[loopback_offset + 1], rx_buf[loopback_offset + 2],
1442 rx_buf[loopback_offset + 3], rx_buf[loopback_offset + 4]);
1443 }
1444 }
1445 }
1446
1447 // If we detected a length, set it (but don't overwrite user-specified value yet)
1448 // The caller will handle verification and update
1449 if (detected_length > 0) {
1450 TMC5160_LOG_DEBUG(*static_cast<Derived*>(this), 2, "SPI", "AutoDetectChainLength: Detected chain length = %u",
1451 detected_length);
1452 // Only update if not user-specified, or if user-specified value matches
1453 if (user_specified_chain_length_ == 0 || user_specified_chain_length_ == detected_length) {
1454 total_chain_length_ = detected_length;
1455 }
1456 } else {
1457 TMC5160_LOG_DEBUG(*static_cast<Derived*>(this), 1, "SPI",
1458 "AutoDetectChainLength: Command pattern not found, assuming single chip");
1459 if (user_specified_chain_length_ == 0) {
1460 total_chain_length_ = 0; // Assume single chip only if not user-specified
1461 }
1462 }
1463
1464 return detected_length;
1465 }

1550 SpiCommand cmd = SpiCommand::Read(address);
1551
1552 // CRITICAL: Chain length MUST be known at this point
1553 // If daisy_chain_position > 0, total_chain_length_ must be > 0 (detected or set)
1554 // If daisy_chain_position == 0, total_chain_length_ should be 1 (single chip)
1555 if (daisy_chain_position > 0 && total_chain_length_ == 0) {

1557 "ReadRegister: Chain length unknown for daisy_chain_position=%u. "
1558 "Cannot proceed without chain length.",
1559 daisy_chain_position);
1560 return false;

1563 // Calculate transfer size and response offset using datasheet formula
1564 // Transfer size: max((k+1)*5, (n-k+1)*5) bytes
1565 // - Sending: (k+1)*5 bytes needed to address device k
1566 // - Receiving: (n-k+1)*5 bytes needed to shift response back (datasheet formula)
1567 // - Use max() to ensure both requirements are met
1568 // Response offset: (n-k-1)*5 bytes (reverse order: last device first)
1569 uint8_t n = total_chain_length_;
1570 uint8_t k = daisy_chain_position;
1571
1572 // Validate: k must be < n (device position must be less than total chain length)
1573 if (k >= n) {
1574 TMC5160_LOG_DEBUG(*static_cast<Derived*>(this), 1, "SPI",
1575 "ReadRegister: Invalid daisy_chain_position=%u for chain length=%u. "
1576 "Position must be < chain length.",
1577 k, n);
1578 return false;
1579 }
1580
1581 // Calculate transfer size: max of sending and receiving requirements
1582 size_t sending_bytes = static_cast<size_t>(k + 1) * 5; // Command must reach device k
1583 size_t receiving_bytes = static_cast<size_t>(n - k + 1) * 5; // Response extraction (datasheet formula)
1584 size_t transfer_bytes = std::max(sending_bytes, receiving_bytes);
1585
1586 // Response offset: (n-k-1)*5 bytes (based on reverse order of devices)
1587 size_t response_byte_offset = static_cast<size_t>(n - k - 1) * 5;
1588
1589 // Validate transfer size meets receiving requirement
1590 if (transfer_bytes < receiving_bytes) {
1591 TMC5160_LOG_DEBUG(*static_cast<Derived*>(this), 1, "SPI",
1592 "ReadRegister: Transfer size %zu bytes < receiving requirement %zu bytes. "
1593 "Response extraction may fail.",
1594 transfer_bytes, receiving_bytes);
1595 return false;
1596 }
1597
1598 // Validate response offset is within buffer bounds
1599 if (response_byte_offset + 4 >= transfer_bytes) {
1600 TMC5160_LOG_DEBUG(*static_cast<Derived*>(this), 1, "SPI",
1601 "ReadRegister: Response offset %zu + 4 >= transfer size %zu. "
1602 "Cannot read full 5-byte response.",
1603 response_byte_offset, transfer_bytes);
1604 return false;
1605 }

1608 std::vector<uint8_t> rx_buf(transfer_bytes, 0);

1611 // For daisy-chain position k, the command is placed first, then padding (zeros)
1612 // Padding structure:
1613 // - Bytes 5 to (k+1)*5-1: Padding to shift command to device k
1614 // - Bytes (k+1)*5 to transfer_bytes-1: Additional padding for full-duplex response extraction
1615 // (only present if transfer_bytes > (k+1)*5, i.e., when (n-k+1)*5 > (k+1)*5)

1620 TMC5160_LOG_DEBUG(*static_cast<Derived*>(this), 3, "SPI",

1623 tx_buf[4], (transfer_bytes > 5) ? tx_buf[5] : 0, (transfer_bytes > 6) ? tx_buf[6] : 0,
1624 (transfer_bytes > 7) ? tx_buf[7] : 0);
1625
1626 if (!SpiTransfer(tx_buf.data(), rx_buf.data(), transfer_bytes)) {
1627 return false;
1628 }

1631 // Use 1us delay for safety
1632 this->DelayUs(1);
1633
1634 // Second transaction: Send address again, receive actual data (pipelined read)
1635 // Per datasheet: Read data is transferred back with the subsequent access
1636 // Use same transfer size for daisy-chaining consistency
1637 if (!SpiTransfer(tx_buf.data(), rx_buf.data(), transfer_bytes)) {
1638 return false;
1639 }

1641 // Extract response data based on daisy-chain position
1642 // IMPORTANT: Responses come back in REVERSE order (last device first, first device last)
1643 // Response offset is calculated above based on whether total_chain_length_ is known
1644 // - If total_chain_length_ > 0: Use datasheet formula, response at (n-k-1)*5 bytes
1645 // - If total_chain_length_ == 0: Use simplified approach, response at k*5 bytes (end of
1646 // transfer)
1647

1649 // For daisy-chaining, this is at the calculated offset
1650 if (response_byte_offset >= rx_buf.size()) {
1651 TMC5160_LOG_DEBUG(*static_cast<Derived*>(this), 1, "SPI",
1652 "Read register 0x%02X: Response offset %zu exceeds buffer size %zu", address,
1653 response_byte_offset, rx_buf.size());
1654 return false;

1661 "Read register 0x%02X: SPI_STATUS=0x%02X [%s%s%s%s%s%s%s%s] - ERROR DETECTED", address,

1663 status.StallGuard2() ? "SG2 " : "", status.Standstill() ? "STST " : "",
1664 status.VelocityReached() ? "VEL " : "", status.PositionReached() ? "POS " : "",
1665 status.StopLeft() ? "STOP_L " : "", status.StopRight() ? "STOP_R " : "");
1666 } else {
1667 // Log response bytes (first 8 or all if less)
1668 size_t log_rx_bytes = (rx_buf.size() < 8) ? rx_buf.size() : 8;
1669 TMC5160_LOG_DEBUG(*static_cast<Derived*>(this), 3, "SPI",
1670 "Read register 0x%02X: RX[0..%zu] %02X %02X %02X %02X %02X %02X %02X %02X "
1671 "| SPI_STATUS=0x%02X [%s%s%s%s%s%s%s%s]",
1672 address, log_rx_bytes - 1, rx_buf[0], (log_rx_bytes > 1) ? rx_buf[1] : 0,
1673 (log_rx_bytes > 2) ? rx_buf[2] : 0, (log_rx_bytes > 3) ? rx_buf[3] : 0,
1674 (log_rx_bytes > 4) ? rx_buf[4] : 0, (log_rx_bytes > 5) ? rx_buf[5] : 0,

1679 status.StopLeft() ? "STOP_L " : "", status.StopRight() ? "STOP_R " : "");

1683 // Byte (response_byte_offset+0) contains SPI_STATUS (bits 39-32)

1693 (static_cast<uint32_t>(rx_buf[response_byte_offset + 3]) << 8) |
1694 static_cast<uint32_t>(rx_buf[response_byte_offset + 4]);

1696 // Return false if critical errors detected (but still extract value)
1697 return !(status.ResetFlag() || status.DriverError());
