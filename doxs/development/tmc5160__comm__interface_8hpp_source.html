122static constexpr uint8_t calculateCrc8(const uint8_t *data,
123 size_t length) noexcept {
124 uint8_t crc = 0;
125 for (size_t i = 0; i < length; ++i) {
126 crc ^= data[i];
127 for (uint8_t j = 0; j < 8; ++j) {
128 if (crc & 0x80) {
129 crc = (crc << 1) ^ 0x07;
130 } else {
131 crc <<= 1;
132 }
133 }
134 }
135 return crc;
136}

444 bool ReadRegister(uint8_t address, uint32_t &value) noexcept {
445 std::array<uint8_t, 8> tx_buf{};
446 std::array<uint8_t, 8> rx_buf{};
447
448 // First transaction: Send read address, receive dummy data
449 tx_buf[0] = address & 0x7F; // Clear write bit
450 for (size_t i = 1; i < 8; ++i) {
451 tx_buf[i] = 0x00;
452 }
453
454 TMC5160_LOG_DEBUG(
455 *static_cast<Derived *>(this), 3, "SPI",
456 "Read register 0x%02X: TX %02X %02X %02X %02X %02X %02X %02X %02X",
457 address, tx_buf[0], tx_buf[1], tx_buf[2], tx_buf[3], tx_buf[4],
458 tx_buf[5], tx_buf[6], tx_buf[7]);
459
460 if (!SpiTransfer(tx_buf.data(), rx_buf.data(), 8)) {
461 return false;
462 }
463
464 // Minimum CSN high time: 2*tclk + 10ns (typically ~176ns with 12MHz clock)
465 // Use 1us delay for safety
466 this->DelayUs(1);
467
468 // Second transaction: Send address again, receive actual data
469 if (!SpiTransfer(tx_buf.data(), rx_buf.data(), 8)) {
470 return false;
471 }
472
473 TMC5160_LOG_DEBUG(
474 *static_cast<Derived *>(this), 3, "SPI",
475 "Read register 0x%02X: RX %02X %02X %02X %02X %02X %02X %02X %02X",
476 address, rx_buf[0], rx_buf[1], rx_buf[2], rx_buf[3], rx_buf[4],
477 rx_buf[5], rx_buf[6], rx_buf[7]);
478
479 // Extract 32-bit value from bytes 1-4 (MSB-first)
480 value = (static_cast<uint32_t>(rx_buf[1]) << 24) |
481 (static_cast<uint32_t>(rx_buf[2]) << 16) |
482 (static_cast<uint32_t>(rx_buf[3]) << 8) |
483 static_cast<uint32_t>(rx_buf[4]);
484
485 return true;
486 }
487
494 bool WriteRegister(uint8_t address, uint32_t value) noexcept {
495 std::array<uint8_t, 8> tx_buf{};
496 std::array<uint8_t, 8> rx_buf{};
497
498 // SPI write: address | 0x80, then 4 bytes of data
499 tx_buf[0] = (address & 0x7F) | 0x80; // Set write bit
500 tx_buf[1] = static_cast<uint8_t>((value >> 24) & 0xFF);
501 tx_buf[2] = static_cast<uint8_t>((value >> 16) & 0xFF);
502 tx_buf[3] = static_cast<uint8_t>((value >> 8) & 0xFF);
503 tx_buf[4] = static_cast<uint8_t>(value & 0xFF);
504 for (size_t i = 5; i < 8; ++i) {
505 tx_buf[i] = 0x00;
506 }
507
508 TMC5160_LOG_DEBUG(*static_cast<Derived *>(this), 3, "SPI",
509 "Write register 0x%02X = 0x%08X: TX %02X %02X %02X %02X "
510 "%02X %02X %02X %02X",
511 address, value, tx_buf[0], tx_buf[1], tx_buf[2],
512 tx_buf[3], tx_buf[4], tx_buf[5], tx_buf[6], tx_buf[7]);
513
514 if (!SpiTransfer(tx_buf.data(), rx_buf.data(), 8)) {
515 return false;
516 }
517
518 TMC5160_LOG_DEBUG(
519 *static_cast<Derived *>(this), 3, "SPI",
520 "Write register 0x%02X: RX %02X %02X %02X %02X %02X %02X %02X %02X",
521 address, rx_buf[0], rx_buf[1], rx_buf[2], rx_buf[3], rx_buf[4],
522 rx_buf[5], rx_buf[6], rx_buf[7]);
523
524 return true;
525 }

639 bool ReadRegister(uint8_t address, uint32_t &value) noexcept {
640 std::array<uint8_t, 9> tx_buf{};
641 std::array<uint8_t, 9> rx_buf{};
642
643 // Build UART frame for read
644 tx_buf[0] = (slaveAddress_ & 0x7F) | 0x01; // Sync bit + address
645 tx_buf[1] = address & 0x7F; // Register address (read)
646 tx_buf[2] = 0x00;
647 tx_buf[3] = 0x00;
648 tx_buf[4] = 0x00;
649 tx_buf[5] = 0x00;
650 tx_buf[6] = 0x00;
651 tx_buf[7] = 0x00;
652 tx_buf[8] = calculateCrc8(tx_buf.data(), 8);
653
654 TMC5160_LOG_DEBUG(
655 *static_cast<Derived *>(this), 3, "UART",
656 "Read register 0x%02X: TX %02X %02X %02X %02X %02X %02X %02X %02X %02X",
657 address, tx_buf[0], tx_buf[1], tx_buf[2], tx_buf[3], tx_buf[4],
658 tx_buf[5], tx_buf[6], tx_buf[7], tx_buf[8]);
659
660 if (!UartSend(tx_buf.data(), 9)) {
661 return false;
662 }
663
664 if (!UartReceive(rx_buf.data(), 9)) {
665 return false;
666 }
667
668 TMC5160_LOG_DEBUG(
669 *static_cast<Derived *>(this), 3, "UART",
670 "Read register 0x%02X: RX %02X %02X %02X %02X %02X %02X %02X %02X %02X",
671 address, rx_buf[0], rx_buf[1], rx_buf[2], rx_buf[3], rx_buf[4],
672 rx_buf[5], rx_buf[6], rx_buf[7], rx_buf[8]);
673
674 // Verify CRC8
675 uint8_t calculated_crc = calculateCrc8(rx_buf.data(), 8);
676 if (calculated_crc != rx_buf[8]) {
677 TMC5160_LOG_DEBUG(*static_cast<Derived *>(this), 1, "UART",
678 "CRC8 mismatch: calculated 0x%02X, received 0x%02X",
679 calculated_crc, rx_buf[8]);
680 return false;
681 }
682
683 // Extract 32-bit value from bytes 2-5 (MSB-first)
684 value = (static_cast<uint32_t>(rx_buf[2]) << 24) |
685 (static_cast<uint32_t>(rx_buf[3]) << 16) |
686 (static_cast<uint32_t>(rx_buf[4]) << 8) |
687 static_cast<uint32_t>(rx_buf[5]);
688
689 return true;
