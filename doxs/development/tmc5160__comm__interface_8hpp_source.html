709static constexpr uint8_t calculateCrc8(const uint8_t* data, size_t length) noexcept {
710 uint8_t crc = 0; // Initial value is zero per datasheet
711
712 // Process each byte LSB to MSB
713 for (size_t i = 0; i < length; ++i) {
714 uint8_t currentByte = data[i];
715
716 // Process each bit LSB to MSB (j=0 is LSB, j=7 is MSB)
717 for (uint8_t j = 0; j < 8; ++j) {
718 // Check: (CRC >> 7) XOR (currentByte & 0x01)
719 // This XORs the MSB of CRC with the LSB of currentByte
720 if (((crc >> 7) ^ (currentByte & 0x01)) != 0) {
721 crc = (crc << 1) ^ 0x07; // Polynomial 0x07 (CRC8-ATM)
722 } else {
723 crc = (crc << 1);
724 }
725 currentByte = currentByte >> 1; // Shift to next bit (LSB to MSB)
726 }
727 }
728
729 return crc;
730}

1522 cmd_bytes[2], cmd_bytes[3], cmd_bytes[4]);

1526 size_t loopback_offset = static_cast<size_t>(n) * 5;

1533 }
1534 }

1537 // The caller will handle verification and update

1543 total_chain_length_ = detected_length;
1544 }
1545 } else {

1547 "AutoDetectChainLength: Command pattern not found, assuming single chip");

1549 // Only reset if it was 0 and user hasn't specified a length
