606static constexpr uint8_t calculateCrc8(const uint8_t* data, size_t length) noexcept {
607 uint8_t crc = 0; // Initial value is zero per datasheet
608
609 // Process each byte LSB to MSB
610 for (size_t i = 0; i < length; ++i) {
611 uint8_t currentByte = data[i];
612
613 // Process each bit LSB to MSB (j=0 is LSB, j=7 is MSB)
614 for (uint8_t j = 0; j < 8; ++j) {
615 // Check: (CRC >> 7) XOR (currentByte & 0x01)
616 // This XORs the MSB of CRC with the LSB of currentByte
617 if (((crc >> 7) ^ (currentByte & 0x01)) != 0) {
618 crc = (crc << 1) ^ 0x07; // Polynomial 0x07 (CRC8-ATM)
619 } else {
620 crc = (crc << 1);
621 }
622 currentByte = currentByte >> 1; // Shift to next bit (LSB to MSB)
623 }
624 }
625
626 return crc;
627}

816 static UartFrame Write(uint8_t node_addr, uint8_t reg_addr, uint32_t value) noexcept {
817 UartFrame uart_frame{};
818 uart_frame.type = UartFrameType::WriteAccess;
819
820 // Byte 0: Sync nibble (bits 0-3: 1,0,1,0 = 0x0A) + Reserved (bits 4-7) + NODEADDR
821 // Note: Current implementation uses simplified sync (bit 0 = 1) for compatibility
822 // Full sync pattern would be: (node_addr & 0x7F) | 0x0A
823 // Simplified: (node_addr & 0x7F) | 0x01
824 uart_frame.frame.write_fields.sync_nodeaddr = (node_addr & 0x7F) | 0x01;
825
826 // Byte 1: RW bit (bit 7 = 1 for write) + 7-bit register address
827 uart_frame.frame.write_fields.rw_address = (reg_addr & 0x7F) | 0x80;
828
829 // Bytes 2-5: 32-bit data (high byte to low byte, MSB-first)
830 uart_frame.frame.write_fields.data_bytes[0] = static_cast<uint8_t>((value >> 24) & 0xFF);
831 uart_frame.frame.write_fields.data_bytes[1] = static_cast<uint8_t>((value >> 16) & 0xFF);
832 uart_frame.frame.write_fields.data_bytes[2] = static_cast<uint8_t>((value >> 8) & 0xFF);
833 uart_frame.frame.write_fields.data_bytes[3] = static_cast<uint8_t>(value & 0xFF);
834
835 // Bytes 6-7: Reserved (don't cares but included in CRC)
836 uart_frame.frame.write_fields.reserved[0] = 0x00;
837 uart_frame.frame.write_fields.reserved[1] = 0x00;
838
839 // Byte 8: CRC8 checksum (calculated automatically)
840 uart_frame.CalculateCrc();
841
842 return uart_frame;

1452 uint8_t cmd_bytes[5];
1453 cmd.GetFrame(cmd_bytes);

1456 // Total transfer: (max_devices+2)*40 bits to ensure we capture loopback
1457 size_t transfer_bytes = static_cast<size_t>(max_devices + 2) * 5;
1458

1460 std::vector<uint8_t> rx_buf(transfer_bytes, 0);
1461
1462 // Place command at the beginning (bytes 0-4)
1463 cmd.GetFrame(tx_buf.data());
1464 // Rest is padding (zeros) - already initialized to 0

1467 "AutoDetectChainLength: Probing up to %u devices, transfer_bytes=%zu, "
1468 "cmd_pattern=%02X %02X %02X %02X %02X",
1469 max_devices, transfer_bytes, cmd_bytes[0], cmd_bytes[1], cmd_bytes[2], cmd_bytes[3],
1470 cmd_bytes[4]);
1471
1472 // Perform SPI transfer
1473 if (!SpiTransfer(tx_buf.data(), rx_buf.data(), transfer_bytes)) {

1475 return 0;
1476 }
1477
1478 // Search for our exact command pattern in the received data
1479 // The command loops back after n*40 bits, appearing at offset n*5 bytes
1480 // Since each device delays data by 40 clocks, our command should appear unmodified
1481 // at the loopback point
1482
1483 uint8_t detected_length = 0;
1484
1485 // Search backwards from max_devices to find where our command appears
1486 // For n devices, our command appears at offset n*5 after looping back
1487 // We require an EXACT match of all 5 bytes to confirm loopback
1488 for (uint8_t n = max_devices; n >= 1; --n) {
1489 size_t loopback_offset = static_cast<size_t>(n) * 5;
1490
1491 if (loopback_offset + 4 < rx_buf.size()) {
1492 // Check if the 5-byte chunk at loopback_offset matches our command pattern EXACTLY
1493 // This is the only reliable way to confirm the command looped back correctly
1494 bool exact_match = true;
1495
1496 for (uint8_t i = 0; i < 5; ++i) {
1497 if (rx_buf[loopback_offset + i] != cmd_bytes[i]) {
1498 exact_match = false;
1499 break;
1500 }
1501 }
1502
1503 if (exact_match) {

1505 detected_length = n;
1506 TMC5160_LOG_DEBUG(*static_cast<Derived*>(this), 2, "SPI",
1507 "AutoDetectChainLength: Found EXACT command pattern match at offset "
1508 "%zu (expected for n=%u), chain length = %u",
1509 loopback_offset, n, detected_length);
1510 break;
1511 }
1512 }
1513 }
1514
1515 // If exact match failed, log debug info to help diagnose
1516 if (detected_length == 0) {
1517 TMC5160_LOG_DEBUG(*static_cast<Derived*>(this), 1, "SPI",
1518 "AutoDetectChainLength: Exact command pattern not found in received data");
1519 TMC5160_LOG_DEBUG(*static_cast<Derived*>(this), 3, "SPI",
1520 "AutoDetectChainLength: Expected pattern: %02X %02X %02X %02X %02X", cmd_bytes[0], cmd_bytes[1],
1521 cmd_bytes[2], cmd_bytes[3], cmd_bytes[4]);
1522
1523 // Log first few potential loopback positions for debugging
1524 for (uint8_t n = 1; n <= 3 && n <= max_devices; ++n) {
1525 size_t loopback_offset = static_cast<size_t>(n) * 5;
1526 if (loopback_offset + 4 < rx_buf.size()) {
1527 TMC5160_LOG_DEBUG(*static_cast<Derived*>(this), 3, "SPI",
1528 "AutoDetectChainLength: At offset %zu (n=%u): %02X %02X %02X %02X %02X", loopback_offset, n,
1529 rx_buf[loopback_offset], rx_buf[loopback_offset + 1], rx_buf[loopback_offset + 2],
1530 rx_buf[loopback_offset + 3], rx_buf[loopback_offset + 4]);
1531 }
1532 }
1533 }
1534
1535 // If we detected a length, set it (but don't overwrite user-specified value yet)
1536 // The caller will handle verification and update
1537 if (detected_length > 0) {
1538 TMC5160_LOG_DEBUG(*static_cast<Derived*>(this), 2, "SPI", "AutoDetectChainLength: Detected chain length = %u",
1539 detected_length);
1540 // Only update if not user-specified, or if user-specified value matches
1541 if (user_specified_chain_length_ == 0 || user_specified_chain_length_ == detected_length) {
1542 total_chain_length_ = detected_length;
1543 }
1544 } else {
1545 TMC5160_LOG_DEBUG(*static_cast<Derived*>(this), 1, "SPI",
1546 "AutoDetectChainLength: Command pattern not found, assuming single chip");
1547 // Don't reset total_chain_length_ if it was already set (e.g., to 1 for single-chip mode)
1548 // Only reset if it was 0 and user hasn't specified a length
1549 if (user_specified_chain_length_ == 0 && total_chain_length_ == 0) {
1550 // Keep it at 0 - caller will handle single-chip case
1551 }
1552 }
