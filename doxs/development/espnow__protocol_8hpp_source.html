286inline uint16_t Crc16Ccitt(const uint8_t* data, size_t len) noexcept
287{
288 uint16_t crc = 0xFFFF;
289 for (size_t i = 0; i < len; ++i) {
290 crc ^= static_cast<uint16_t>(data[i]) << 8;
291 for (int j = 0; j < 8; ++j) {
292 if (crc & 0x8000) {
293 crc = (crc << 1) ^ 0x1021;
294 } else {
295 crc <<= 1;
296 }
297 }
298 }
299 return crc;
300}
