56inline bool CalculateMotorCurrent(const MotorSpec& motor_spec, uint32_t sense_resistor_mohm, uint32_t supply_voltage_mv,
57 uint16_t run_current_ma, uint16_t hold_current_ma, uint8_t& irun, uint8_t& ihold,
58 uint16_t& global_scaler) noexcept {
59 // Validate inputs
60 if (sense_resistor_mohm == 0 || supply_voltage_mv == 0) {
61 return false;
62 }
63
64 // Use motor rated current if run_current_ma is 0
65 if (run_current_ma == 0) {
66 run_current_ma = motor_spec.rated_current_ma;
67 }
68 if (run_current_ma == 0) {
69 return false; // No current specified
70 }
71
72 // Default hold current to 30% of run current if not specified
73 if (hold_current_ma == 0) {
74 float hold_current_float = run_current_ma * 0.3F;
75 hold_current_ma = static_cast<uint16_t>(hold_current_float);
76 }
77
78 // Use shared constants
79 constexpr float VFS_MV = MotorCalcConstants::VFS_MV;
80 constexpr float SQRT2 = MotorCalcConstants::SQRT2;
81
82 // Calculate maximum possible RMS current at full scale (GLOBAL_SCALER=256, CS=31)
83 // I_RMS_max (A) = (256/256) * ((31+1)/32) * (VFS_V/RSENSE_Ω) * (1/√2)
84 // = 1.0 * 1.0 * (VFS_V/RSENSE_Ω) * (1/√2)
85 // Converting from mV/mΩ to A: (VFS_mV/1000) / (RSENSE_mΩ/1000) = VFS_mV/RSENSE_mΩ (already in A)
86 // Then convert to mA: multiply by 1000
87 float i_rms_max_ma = ((VFS_MV / static_cast<float>(sense_resistor_mohm)) / SQRT2) * 1000.0F;
88
89 // Check if desired current exceeds maximum possible
90 if (static_cast<float>(run_current_ma) > i_rms_max_ma * 1.1F) { // Allow 10% tolerance
91 return false; // Desired current too high for sense resistor
92 }
93
94 // Strategy: Use IRUN in optimal range (16-31) and fine-tune with GLOBAL_SCALER
95 // Start with IRUN=31 (maximum CS) and calculate required GLOBAL_SCALER
96 // Then adjust IRUN down if GLOBAL_SCALER would be too low
97
98 // Calculate required GLOBAL_SCALER for IRUN=31
99 // I_RMS (mA) = (GLOBAL_SCALER/256) * ((31+1)/32) * (VFS_mV/RSENSE_mΩ) * (1/√2)
100 // Rearranged: GLOBAL_SCALER = I_RMS_ma * 256 * 32 / ((31+1) * (VFS_mV/RSENSE_mΩ) * (1/√2))
101 float global_scaler_float = (static_cast<float>(run_current_ma) * 256.0F * 32.0F) /
102 (32.0F * (VFS_MV / static_cast<float>(sense_resistor_mohm)) / SQRT2);
103
104 // Constrain GLOBAL_SCALER to valid range (32-256, where 0 = 256)
105 auto calculated_scaler = static_cast<uint16_t>(std::round(global_scaler_float));
106 calculated_scaler = std::max<uint16_t>(calculated_scaler, 32);
107 calculated_scaler = std::min<uint16_t>(calculated_scaler, 256);
108
109 // If GLOBAL_SCALER is at maximum (256), we can reduce IRUN for better precision
110 // Try to find optimal IRUN in range 16-31
111 uint8_t optimal_irun = 31;
112 uint16_t optimal_scaler = calculated_scaler;
113
114 if (calculated_scaler >= 200) {
115 // Try reducing IRUN to get GLOBAL_SCALER in better range (128-200)
116 for (uint8_t test_irun = 30; test_irun >= 16; --test_irun) {
117 float test_scaler_float = (static_cast<float>(run_current_ma) * 256.0F * 32.0F) /
118 (static_cast<float>(test_irun + 1) * (VFS_MV / static_cast<float>(sense_resistor_mohm)) / SQRT2);
119 auto test_scaler = static_cast<uint16_t>(std::round(test_scaler_float));
120
121 if (test_scaler >= 32 && test_scaler <= 200) {
122 optimal_irun = test_irun;
123 optimal_scaler = test_scaler;
124 break;
125 }
126 }
127 }
128
129 // Ensure IRUN meets minimum for automatic tuning (IRUN ≥ 8)
130 if (optimal_irun < 8) {
131 optimal_irun = 8;
132 // Recalculate scaler for IRUN=8
133 float scaler_float = (static_cast<float>(run_current_ma) * 256.0F * 32.0F) /
134 (9.0F * (VFS_MV / static_cast<float>(sense_resistor_mohm)) / SQRT2);
135 optimal_scaler = static_cast<uint16_t>(std::round(scaler_float));
136 optimal_scaler = std::max<uint16_t>(optimal_scaler, 32);
137 optimal_scaler = std::min<uint16_t>(optimal_scaler, 256);
138 }
139
140 // Calculate IHOLD using same method
141 // Calculate required IHOLD (use same scaler as IRUN)
142 // I_RMS (mA) = (GLOBAL_SCALER/256) * ((IHOLD+1)/32) * (VFS_mV/RSENSE_mΩ) * (1/√2)
143 // Rearranged: IHOLD = (I_RMS_ma * 256 * 32) / (GLOBAL_SCALER * (VFS_mV/RSENSE_mΩ) * (1/√2)) - 1
144 float ihold_float =
145 ((static_cast<float>(hold_current_ma) * 256.0F * 32.0F) /
146 (static_cast<float>(optimal_scaler) * (VFS_MV / static_cast<float>(sense_resistor_mohm)) / SQRT2)) - 1.0F;
147
148 auto calculated_ihold = static_cast<uint8_t>(std::round(ihold_float));
149
150 // Constrain IHOLD to valid range (0-31) and ensure it's less than IRUN
151 calculated_ihold = std::min<uint8_t>(calculated_ihold, 31);
152 if (calculated_ihold >= optimal_irun) {
153 calculated_ihold = (optimal_irun > 0) ? (optimal_irun - 1) : 0;
154 }
155
156 irun = optimal_irun;
157 ihold = calculated_ihold;
158 global_scaler = optimal_scaler;
159
160 return true;
161}

180inline uint16_t CalculateStealthChopLowerLimit(const MotorSpec& motor_spec, uint32_t supply_voltage_mv, uint8_t tbl,
181 uint8_t pwm_freq, uint32_t f_clk = 12000000U) noexcept {
182 if (motor_spec.winding_resistance_mohm == 0) {
183 return 0; // Cannot calculate without resistance
184 }
185
186 // Blank time in clock cycles based on TBL setting
187 constexpr std::array<uint8_t, 4> blank_times = {16, 24, 36, 54};
188 const uint8_t tbl_index = std::min(tbl, static_cast<uint8_t>(3));
189 uint8_t t_blank = blank_times[tbl_index];
190
191 // PWM frequency divider based on PWM_FREQ setting
192 constexpr std::array<uint32_t, 4> pwm_divisors = {1024, 683, 512, 410};
193 const uint8_t pwm_freq_index = std::min(pwm_freq, static_cast<uint8_t>(3));
194 uint32_t pwm_divisor = pwm_divisors[pwm_freq_index];
195
196 // Calculate PWM frequency: f_PWM = 2 / (pwm_divisor * t_CLK)
197 // Actually: f_PWM = 2 / (pwm_divisor * (1/f_CLK)) = 2 * f_CLK / pwm_divisor
198 float f_pwm = (2.0F * static_cast<float>(f_clk)) / static_cast<float>(pwm_divisor);
199
200 // Convert to Hz: f_PWM = 2 / (pwm_divisor * t_CLK) where t_CLK = 1/f_CLK
201 // So: f_PWM = 2 * f_CLK / pwm_divisor (already calculated above)
202
203 // Calculate lower limit: I (mA) = t_BLANK * f_PWM * V_M (mV) / R_COIL (mΩ)
204 // t_BLANK is in clock cycles, so we need: t_BLANK / f_CLK (time in seconds)
205 float t_blank_sec = static_cast<float>(t_blank) / static_cast<float>(f_clk);
206 float i_lower_ma = (t_blank_sec * f_pwm * static_cast<float>(supply_voltage_mv)) /
207 static_cast<float>(motor_spec.winding_resistance_mohm);
208
209 return static_cast<uint16_t>(i_lower_ma);
210}

250inline uint8_t CalculateS2VSLevel(uint16_t voltage_mv) noexcept {
251 if (voltage_mv == 0) {
252 return 6; // Default recommended value
253 }
254
255 // Constrain to valid range
256 if (voltage_mv < 400 || voltage_mv > 2000) {
257 return 0; // Invalid
258 }
259
260 // Datasheet typical values for interpolation
261 // S2VS_LEVEL=6: 625mV, S2VS_LEVEL=15: 1560mV
262 // Linear relationship: level = 6 + (voltage - 625) * (15-6) / (1560-625)
263 // Simplified: level = 6 + (voltage - 625) * 9 / 935
264
265 if (voltage_mv <= 625) {
266 // Below or at level 6 threshold - use linear interpolation from level 4 (approx 400mV) to level 6
267 // Level 4-6 range: approximate 400-625mV
268 if (voltage_mv < 400) {
269 return 4; // Minimum
270 }
271 float level = 4.0F + ((static_cast<float>(voltage_mv - 400) / 225.0F) * 2.0F); // 4 to 6
272 return static_cast<uint8_t>(std::round(level));
273 }
274 // Above level 6 threshold - interpolate from level 6 to level 15
275 float level = 6.0F + ((static_cast<float>(voltage_mv - 625) / 935.0F) * 9.0F); // 6 to 15
276 auto calculated = static_cast<uint8_t>(std::round(level));
277 return std::min(static_cast<uint8_t>(15), std::max(static_cast<uint8_t>(4), calculated));
278}

297inline uint8_t CalculateS2GLevel(uint16_t voltage_mv, uint32_t supply_voltage_mv = 0) noexcept {
298 if (voltage_mv == 0) {
299 return 6; // Default recommended value
300 }
301
302 // Constrain to valid range
303 if (voltage_mv < 400 || voltage_mv > 2000) {
304 return 0; // Invalid
305 }
306
307 // For VS>52V, enforce minimum 1200mV (S2G_LEVEL=12) to prevent false triggers
308 if (supply_voltage_mv > 52000 && voltage_mv < 1200) {
309 voltage_mv = 1200; // Enforce minimum
310 }
311
312 // Datasheet typical values for interpolation
313 // S2G_LEVEL=6: 625mV, S2G_LEVEL=15: 1560mV (VS<52V) or 850mV (VS<55V)
314 // Use VS<52V values for interpolation (more conservative)
315
316 if (voltage_mv <= 625) {
317 // Below or at level 6 threshold - use linear interpolation from level 2 (approx 400mV) to level 6
318 if (voltage_mv < 400) {
319 return 2; // Minimum
320 }
321 float level = 2.0F + ((static_cast<float>(voltage_mv - 400) / 225.0F) * 4.0F); // 2 to 6
322 return static_cast<uint8_t>(std::round(level));
323 }
324 // Above level 6 threshold - interpolate from level 6 to level 15
325 float level = 6.0F + ((static_cast<float>(voltage_mv - 625) / 935.0F) * 9.0F); // 6 to 15
326 auto calculated = static_cast<uint8_t>(std::round(level));
327 return std::min(static_cast<uint8_t>(15), std::max(static_cast<uint8_t>(2), calculated));

344inline uint8_t CalculateShortDelay(uint8_t delay_us_x10) noexcept {
345 if (delay_us_x10 == 0) {
346 return 0; // Default recommended value (0.85µs = shortdelay=0)
347 }
348
349 // Constrain to valid range (5-25 = 0.5-2.5µs)
350 if (delay_us_x10 < 5) {
351 delay_us_x10 = 5; // Minimum 0.5µs
352 } else if (delay_us_x10 > 25) {
353 delay_us_x10 = 25; // Maximum 2.5µs
354 }
355
356 // Threshold at ~1.0µs (10 in 0.1µs units)
357 // Below 1.0µs: use shortdelay=0 (normal)
358 // At or above 1.0µs: use shortdelay=1 (high delay)
359 return (delay_us_x10 >= 10) ? 1 : 0;
360}
